    61→  const sheets = useEditor((s) => s.sheets);
    62→  const childSheets = useEditor((s) => s.childSheets);
    63→  const showGrid = useEditor((s) => s.showGrid);
    64→  const setShowGrid = useEditor((s) => s.setShowGrid);
    65→  const snapToGrid = useEditor((s) => s.snapToGrid);
    66→  const setSnapToGrid = useEditor((s) => s.setSnapToGrid);
    67→  const past = useEditor((s) => s.past.length);
    68→  const future = useEditor((s) => s.future.length);
    69→  // KiCad-parity new state
    70→  const units = useEditor((s) => s.units);
    71→  const setUnits = useEditor((s) => s.setUnits);
    72→  const showPinNumbers = useEditor((s) => s.showPinNumbers);
    73→  const setShowPinNumbers = useEditor((s) => s.setShowPinNumbers);
    74→  const showPinNames = useEditor((s) => s.showPinNames);
    75→  const setShowPinNames = useEditor((s) => s.setShowPinNames);
    76→  const showPinElecTypes = useEditor((s) => s.showPinElecTypes);

    78→  const showRefdes = useEditor((s) => s.showRefdes);
    79→  const setShowRefdes = useEditor((s) => s.setShowRefdes);
    80→  const showValues = useEditor((s) => s.showValues);
    81→  const setShowValues = useEditor((s) => s.setShowValues);
    82→  const activeTool = useEditor((s) => s.activeTool);
    83→  const setActiveTool = useEditor((s) => s.setActiveTool);
    84→
    85→  const fileInputRef = useRef<HTMLInputElement | null>(null);
    86→  const schImportInputRef = useRef<HTMLInputElement | null>(null);
    87→  const [showMyCircuits, setShowMyCircuits] = useState(false);
    88→  const [showSpiceImport, setShowSpiceImport] = useState(false);
    89→  const [showSubCircuit, setShowSubCircuit] = useState(false);
    90→  const [showFindReplace, setShowFindReplace] = useState(false);
    91→  const [showViolations, setShowViolations] = useState(false);
    92→  const [showNetInspector, setShowNetInspector] = useState(false);

   108→    const handler = () => setShowFindReplace(true);
   109→    window.addEventListener('circuitlab:open-find-replace', handler);

   113→      if (detail) setCamera(detail);
   114→    };

   116→    // ask canvas for current camera
   117→    const askCamera = () => window.dispatchEvent(new CustomEvent('circuitlab:request-camera'));
   118→    askCamera();

   122→      window.removeEventListener('circuitlab:camera', camHandler);

   128→    const doc = serialize();
   129→    const blob = new Blob([JSON.stringify(doc, null, 2)], { type: 'application/json' });
   130→    const url = URL.createObjectURL(blob);
   131→    const a = document.createElement('a');
   132→    a.href = url;
   133→    a.download = `circuit_${Date.now()}.json`;
   134→    a.click();
   135→    URL.revokeObjectURL(url);
   136→  }, [serialize]);
   137→

   139→    const file = e.target.files?.[0];
   140→    if (!file) return;
   141→    const reader = new FileReader();
   142→    reader.onload = () => {
   143→      try {
   144→        const doc = JSON.parse(reader.result as string);
   145→        if (doc.version === 1 && Array.isArray(doc.components) && Array.isArray(doc.wires)) {
   146→          loadDocument(doc);
   147→        } else {
   148→          alert('Invalid circuit file');
   149→        }
   150→      } catch (err) {
   151→        alert('Failed to parse file: ' + (err as Error).message);
   152→      }
   153→    };
   154→    reader.readAsText(file);
   155→    e.target.value = '';
   156→  }, [loadDocument]);
   157→
   158→  // KiCad / Eagle schematic import
   159→  const handleSchImport = useCallback(async (e: React.ChangeEvent<HTMLInputElement>) => {
   160→    const file = e.target.files?.[0];
   161→    if (!file) return;
   162→    const reader = new FileReader();
   163→    reader.onload = () => {
   164→      try {
   165→        const text = reader.result as string;
   166→        const result = parseSchematicFile(text, file.name);
   167→        loadDocument(result.document);
   168→        if (result.unknownSymbols.length > 0) {
   169→          toast.warning(`Imported with ${result.unknownSymbols.length} unknown symbols skipped`, {
   170→            description: result.unknownSymbols.slice(0, 5).join('\n'),
   171→          });
   172→        } else {
   173→          toast.success(`Imported ${file.name}`);
   174→        }
   175→        if (result.warnings.length > 0) {
   176→          console.log('Import warnings:', result.warnings);
   177→        }
   178→      } catch (err) {
   179→        toast.error('Import failed: ' + (err as Error).message);
   180→      }
   181→    };
   182→    reader.readAsText(file);
   183→    e.target.value = '';
   184→  }, [loadDocument]);
   185→
   186→  const handlePlotSVG = useCallback(() => {
   187→    const doc = serialize();

   189→    downloadText(svg, `schematic_${Date.now()}.svg`, 'image/svg+xml');
   190→    toast.success('Exported SVG');
   191→  }, [serialize]);
   192→
   193→  const handlePlotPNG = useCallback(async () => {
   194→    try {
   195→      const doc = serialize();
   196→      const blob = await exportSchematicPNG(doc, 2);
   197→      downloadBlob(blob, `schematic_${Date.now()}.png`);
   198→      toast.success('Exported PNG');
   199→    } catch (err) {
   200→      toast.error('PNG export failed: ' + (err as Error).message);
   201→    }
   202→  }, [serialize]);
   203→
   204→  const handlePlotPDF = useCallback(async () => {
   205→    try {
   206→      const doc = serialize();
   207→      const blob = await exportSchematicPDF(doc);
   208→      downloadBlob(blob, `schematic_${Date.now()}.pdf`);
   209→      toast.success('Exported PDF');
   210→    } catch (err) {
   211→      toast.error('PDF export failed: ' + (err as Error).message);
   212→    }
   213→  }, [serialize]);
   214→

   230→    const doc = serialize();

   232→    if (format === 'csv') {
   233→      downloadText(exportBOMCSV(doc), `bom_${Date.now()}.csv`, 'text/csv');
   234→    } else if (format === 'html') {
   235→      downloadText(exportBOMHTML(doc), `bom_${Date.now()}.html`, 'text/html');
   236→    } else {
   237→      downloadText(exportBOMXML(doc), `bom_${Date.now()}.xml`, 'application/xml');

   408→          <TooltipTrigger asChild>
   409→            <Button size="sm" variant="ghost" onClick={() => setShowSubCircuit(true)} disabled={running}>
   410→              <Boxes size={14} />
   411→              <span className="ml-1 hidden lg:inline">Sub-Circuit</span>

   413→          </TooltipTrigger>

   415→        </Tooltip>

   428→              <ShieldCheck size={14} />
