   276→/** Get the reference designator prefix for a component type */
   277→function refdesPrefix(type: string): string {
   278→  switch (type) {
   279→    case 'resistor': return 'R';
   280→    case 'capacitor': return 'C';
   281→    case 'inductor': return 'L';
   282→    case 'led': return 'LED';
   283→    case 'diode': return 'D';
   284→    case 'zener': return 'DZ';
   285→    case 'schottky': return 'D';
   286→    case 'dcVoltage':
   287→    case 'acVoltage':
   288→    case 'pulseSource': return 'V';
   289→    case 'currentSource': return 'I';
   290→    case 'timer555':
   291→    case 'opamp':
   292→    case 'opampRails':
   293→    case 'voltageRegulator':
   294→    case 'vco': return 'U';
   295→    case 'npn':
   296→    case 'pnp':
   297→    case 'nmos':
   298→    case 'pmos': return 'Q';
   299→    case 'switch':
   300→    case 'pushButton': return 'SW';
   301→    case 'fuse': return 'F';
   302→    case 'crystal': return 'Y';
   303→    case 'transformer': return 'T';
   304→    case 'speaker':
   305→    case 'buzzer': return 'LS';
   306→    case 'dcMotor': return 'M';
   307→    case 'photoresistor': return 'LDR';
   308→    case 'sevenSegment': return 'DSP';
   309→    case 'ground':
   310→    case 'powerGND': return 'GND';
   311→    case 'junction': return 'J';
   312→    case 'oscilloscope':
   313→    case 'voltmeter':
   314→    case 'ammeter': return 'TP';
   315→    case 'arduino':
   316→    case 'arduinoReal':
   317→    case 'raspberryPi': return 'U';
   318→    case 'netLabel':
   319→    case 'busLabel': return 'NL';
   320→    case 'bus': return 'BUS';
   321→    default:
   322→      if (type.startsWith('7400_') || type.startsWith('7402_') || type.startsWith('7404_') ||
   323→          type.startsWith('7408_') || type.startsWith('7432_') || type.startsWith('7486_') ||
   324→          type.startsWith('7474_')) return 'U';
   325→      if (type.startsWith('custom_')) return 'U';
   326→      return 'U';
   327→  }
   328→}

   330→/** Generate the next refdes for a given type, based on existing components */
   331→function nextRefdes(type: string, components: CircuitComponent[]): string {
   332→  const prefix = refdesPrefix(type);
   333→  let maxNum = 0;
   334→  for (const c of components) {
   335→    if (refdesPrefix(c.type) === prefix) {
   336→      const ref = c.refdes ?? c.id;
   337→      const match = ref.match(/(\d+)$/);
   338→      if (match) {
   339→        const n = parseInt(match[1]);
   340→        if (n > maxNum) maxNum = n;
   341→      }
   342→    }
   343→  }
   344→  return `${prefix}${maxNum + 1}`;
   345→}

   347→function snapshot(s: {
   348→  components: CircuitComponent[];
   349→  wires: Wire[];
   350→  drawings?: DrawingPrimitive[];
   351→  noConnects?: NoConnectMarker[];
   352→  groups?: Group[];
   353→  sheets?: HierarchicalSheet[];
   354→  netClasses?: NetClass[];
   355→}) {
   356→  return {
   357→    components: s.components.map((c) => ({ ...c, parameters: { ...c.parameters }, simState: undefined, fields: c.fields ? c.fields.map((f) => ({ ...f })) : undefined })),
   358→    wires: s.wires.map((w) => ({ ...w })),
   359→    drawings: s.drawings ? s.drawings.map((d) => ({ ...d })) : [],
   360→    noConnects: s.noConnects ? s.noConnects.map((n) => ({ ...n })) : [],
   361→    groups: s.groups ? s.groups.map((g) => ({ ...g, componentIds: [...g.componentIds], wireIds: [...g.wireIds], drawingIds: [...g.drawingIds] })) : [],
   362→    sheets: s.sheets ? s.sheets.map((sh) => ({ ...sh, pins: sh.pins.map((p) => ({ ...p })) })) : [],
   363→    netClasses: s.netClasses ? s.netClasses.map((nc) => ({ ...nc, nets: [...nc.nets] })) : [],
   364→  };
   365→}

   369→// helper: get plugin defaults
   370→function defaultsFor(type: string): Record<string, number | string | boolean> {
   371→  const plugin = getPlugin(type);
   372→  if (!plugin) return {};
   373→  const defaults: Record<string, number | string | boolean> = {};
   374→  for (const p of plugin.parameters) defaults[p.key] = p.default;
   375→  return defaults;

   391→  traces: [],
   392→  maxTraceSamples: 500,
   393→  past: [],
   394→  future: [],
   395→  paused: true,
   396→  showGrid: true,
   397→  snapToGrid: true,
   398→  wireDraft: null,
   399→  // new doc state
   400→  sheets: [],
   401→  netClasses: [],
   402→  drawings: [],
   403→  noConnects: [],
   404→  groups: [],
   405→  pageSetup: { ...DEFAULT_PAGE_SETUP },
   406→  savedViews: [],
   407→  childSheets: {},
   408→  activeSheet: '',
   409→  metadata: { title: 'Untitled', revision: 'Rev 1', date: new Date().toISOString().slice(0, 10) },
   410→  // display
   411→  units: 'grid',
   412→  gridSize: 1,
   413→  showPinNumbers: false,
   414→  showPinNames: false,
   415→  showPinElecTypes: false,
   416→  showRefdes: true,
   417→  showValues: true,
   418→  showTitleBlock: false,
   419→  theme: 'dark',
   420→  customHotkeys: {},
   421→  activeTool: 'select',
   422→
   423→  addComponent: (type, position) => {
   424→    get().pushHistory();
   425→    const id = genId('comp');
   426→    const s = get();
   427→    const refdes = nextRefdes(type, s.components);
   428→    const comp: CircuitComponent = {
   429→      id,
   430→      type,
   431→      position: { ...position },
   432→      rotation: 0,
   433→      parameters: defaultsFor(type),
   434→      refdes,
   435→    };
   436→    set((s) => ({ components: [...s.components, comp], selection: { type: 'component', id } }));
   437→    return id;
   438→  },
   439→
   440→  moveComponent: (id, position) => {
   441→    set((s) => ({
   442→      components: s.components.map((c) =>
   443→        c.id === id ? { ...c, position: { ...position } } : c,
   444→      ),
   445→    }));
   446→  },
   447→
   448→  rotateComponent: (id) => {
   449→    get().pushHistory();
   450→    set((s) => ({
   451→      components: s.components.map((c) =>
   452→        c.id === id
   453→          ? { ...c, rotation: (((c.rotation + 1) % 4) as 0 | 1 | 2 | 3) }
   454→          : c,
   455→      ),
   456→    }));
   457→  },
   458→
   459→  mirrorComponent: (id, axis) => {
   460→    get().pushHistory();
   461→    set((s) => ({
   462→      components: s.components.map((c) =>
   463→        c.id === id
   464→          ? { ...c, [axis === 'x' ? 'mirrorX' : 'mirrorY']: !c[axis === 'x' ? 'mirrorX' : 'mirrorY'] }
   465→          : c,
   466→      ),
   467→    }));
   468→  },
   469→
   470→  rotateComponentFree: (id, degrees) => {
   471→    // Snap to 15° increments (24 distinct angles). Stored in `rotationDeg`.
   472→    // The canvas's render loop should use rotationDeg if present, otherwise fall back to rotation*90.
   473→    get().pushHistory();
   474→    const snapped = Math.round(degrees / 15) * 15;
   475→    set((s) => ({
   476→      components: s.components.map((c) =>
   477→        c.id === id ? { ...c, rotationDeg: snapped } : c,
   478→      ),
   479→    }));
   480→  },
   481→
   482→  toggleDeMorgan: (id) => {
   483→    const comp = get().components.find((c) => c.id === id);
   484→    if (!comp) return;
   485→    const plugin = getPlugin(comp.type);
   486→    if (!plugin?.hasAlternateBody) return;
   487→    get().pushHistory();
   488→    set((s) => ({
   489→      components: s.components.map((c) =>
   490→        c.id === id ? { ...c, convert: c.convert === 2 ? 1 : 2 } : c,
   491→      ),
   492→    }));
   493→  },
   494→
   495→  swapPins: (id, pinIdA, pinIdB) => {
   496→    const comp = get().components.find((c) => c.id === id);
   497→    if (!comp) return false;
   498→    const plugin = getPlugin(comp.type);
   499→    if (!plugin?.pinSwapGroups) return false;
   500→    // Check if both pins are in the same swap group
   501→    const inSameGroup = plugin.pinSwapGroups.some((group) =>
   502→      group.includes(pinIdA) && group.includes(pinIdB),
   503→    );
   504→    if (!inSameGroup) return false;
   505→    // Swap: we re-emit all wires that referenced pinIdA to pinIdB and vice versa.
   506→    // This requires walking the wires array and swapping terminal IDs.
   507→    get().pushHistory();
   508→    set((s) => ({
   509→      wires: s.wires.map((w) => {
   510→        if (w.from.componentId === id && w.from.terminalId === pinIdA) {
   511→          return { ...w, from: { ...w.from, terminalId: pinIdB } };
   512→        }
   513→        if (w.from.componentId === id && w.from.terminalId === pinIdB) {
   514→          return { ...w, from: { ...w.from, terminalId: pinIdA } };
   515→        }
   516→        if (w.to.componentId === id && w.to.terminalId === pinIdA) {
   517→          return { ...w, to: { ...w.to, terminalId: pinIdB } };
   518→        }
   519→        if (w.to.componentId === id && w.to.terminalId === pinIdB) {

   940→      // when navigating back to root, just restore the saved snapshot below
   941→      // when going elsewhere).
   942→    }
   943→
   944→    if (!fileName) {
   945→      // Navigating back to root — restore root's saved snapshot if we have one.
   946→      // Root's components/wires are kept as the "current" state when on root,
   947→      // so if we're already on root there's nothing to do. If we're returning
   948→      // from a sub-sheet, the root snapshot is in `childSheets['__root__']`
   949→      // (a special key we use to preserve root state across sub-sheet edits).
   950→      const rootSnap = (s.childSheets as any)['__root__'];

   954→          components: rootSnap.components,
   955→          wires: rootSnap.wires,

   964→      }

   967→

   969→    if (!currentSheetFileName) {

   972→        childSheets: {
   973→          ...st.childSheets,
   974→          __root__: rootSnap as any,
   975→        },
   976→      }));
   977→    }

   983→        activeSheet: fileName,
   984→        components: targetDoc.components ?? [],
   985→        wires: targetDoc.wires ?? [],

   990→      set({
   991→        activeSheet: fileName,
   992→        components: [],
   993→        wires: [],
   994→        sheets: [],
   995→      });
   996→    }

  1299→  step: () => {
  1300→    const s = get();
  1301→    if (s.components.length === 0 && s.sheets.length === 0) return;
  1302→
  1303→    try {
  1304→
  1305→    // ── Cross-sheet simulation ────────────────────────────────────────────
  1306→    // If we're inside a sub-sheet OR the root has sub-sheets, we need to
  1307→    // flatten the hierarchy into a single (components, wires) pair before
  1308→    // simulating. We do this on every step so sub-sheet edits are picked up
  1309→    // immediately. The flattened components have prefixed IDs (e.g. "amp.R1")
  1310→    // so they don't collide with the active sheet's IDs.
  1311→    let simComponents = s.components;
  1312→    let simWires = s.wires;
  1313→    let usingHierarchy = false;
  1314→    if (s.activeSheet || s.sheets.length > 0) {
  1315→      usingHierarchy = true;
  1316→      // Save the current sheet back into childSheets so flattenHierarchy sees it
  1317→      let childSheets = s.childSheets;
  1318→      if (s.activeSheet) {
  1319→        childSheets = {
  1320→          ...childSheets,
  1321→          [s.activeSheet]: snapshotSheet(s.components, s.wires, s.sheets),
  1322→        };
  1323→      }
  1324→      // Build root document from either the active root state or the saved __root__
  1325→      const rootDoc = s.activeSheet
  1326→        ? (childSheets as any).__root__ ?? { version: 1 as const, components: [], wires: [], sheets: [] }
  1327→        : { version: 1 as const, components: s.components, wires: s.wires, sheets: s.sheets };
  1328→      const flat = flattenHierarchy(rootDoc as any, childSheets as any);
  1329→      simComponents = flat.components;
  1330→      simWires = flat.wires;
  1331→    }
  1332→
  1333→    const plugins = new Map<string, any>();
  1334→    for (const c of simComponents) {
  1335→      const p = getPlugin(c.type);
  1336→      if (p) plugins.set(c.type, p);
  1337→    }
  1338→    // Keep a persistent state map across steps so capacitors/inductors/555 remember their state
  1339→    const persistentState = s.simContext?.state ?? {};
  1340→    const prev = s.simContext
  1341→      ? {
  1342→          nodeVoltage: s.simContext.nodeVoltage,
  1343→          branchCurrent: s.simContext.branchCurrent,
  1344→          time: s.simContext.time,
  1345→          state: persistentState,
  1346→        }
  1347→      : { nodeVoltage: new Float64Array(0), branchCurrent: new Float64Array(0), time: 0, state: persistentState };
  1348→    // run sub-steps based on speed
  1349→    // For speed >= 1: run floor(speed) sub-steps per frame (e.g., 4x = 4 steps)
  1350→    // For speed < 1: run 1 step every Nth frame, where N = ceil(1/speed)
  1351→    //   This is tracked via simStepCounter in the store state
  1352→    const speed = s.speed;
  1353→    let subSteps: number;
  1354→    if (speed >= 1) {
  1355→      subSteps = Math.max(1, Math.floor(speed));
  1356→    } else {
  1357→      // Sub-real-time: skip steps to slow down the simulation
  1358→      const skipInterval = Math.ceil(1 / speed); // e.g., 0.5x → skip every 2nd frame
  1359→      const stepCount = (s as any).__stepCount ?? 0;
  1360→      (s as any).__stepCount = stepCount + 1;
  1361→      if (stepCount % skipInterval !== 0) {
  1362→        return; // skip this frame to slow down
  1363→      }
  1364→      subSteps = 1;
  1365→    }
  1366→    const dt = s.dt;
  1367→    // Build simOptions from the editor's settings (temperature, .IC, .NODESET)
  1368→    const simOpts = {
  1369→      initialConditions: s.simOptions?.initialConditions,
  1370→      nodeSets: s.simOptions?.nodeSets,
  1371→    };
  1372→    let result: { sim: SimContext; branchCurrentSize: number; nodeMap: any } | null = null;
  1373→    for (let i = 0; i < subSteps; i++) {
  1374→      result = simulateStep(simComponents, simWires, plugins, prev, dt, simOpts);
  1375→      if (!result) break;
  1376→      prev.nodeVoltage = result.sim.nodeVoltage;
  1377→      prev.branchCurrent = result.sim.branchCurrent;
  1378→      prev.time = result.sim.time;
  1379→      prev.state = result.sim.state;
  1380→
  1381→      // Digital fast-forward: advance sim.time toward the next clock edge.
  1382→      // Handles two types of clock sources:
  1383→      //   1. pulseSource (crystal) — when no capacitors/inductors present
  1384→      //   2. timer555 in astable mode — computes output from sim.time + R/C params
  1385→      // Without fast-forward, a 1Hz clock would take ~10000 steps per cycle.
  1386→      const hasCapacitor = simComponents.some(c => c.type === 'capacitor' || c.type === 'inductor');
  1387→      const hasPulseSource = simComponents.some(c => c.type === 'pulseSource');
  1388→      const hasArduino = simComponents.some(c => c.type === 'arduinoReal' || c.type === 'arduino');
  1389→      const hasAstable555 = simComponents.some(c =>
  1390→        c.type === 'timer555' && (c.parameters.astable as boolean) === true);
  1391→
  1392→      // Compute the minimum clock frequency across all digital clock sources
  1393→      let minFreq = Infinity;
  1394→
  1395→      // Pulse sources
  1396→      if (hasPulseSource && !hasCapacitor && !hasArduino) {
  1397→        for (const c of simComponents) {
  1398→          if (c.type === 'pulseSource') {
  1399→            const f = c.parameters.frequency as number;
  1400→            if (f > 0 && f < minFreq) minFreq = f;
  1401→          }
  1402→        }
  1403→      }
  1404→
  1405→      // Astable 555 timers: frequency = 1 / (0.693 * (R1 + 2*R2) * C)
  1406→      if (hasAstable555) {
  1407→        for (const c of simComponents) {
  1408→          if (c.type === 'timer555' && (c.parameters.astable as boolean) === true) {
  1409→            const r1 = (c.parameters.r1 as number) || 47000;
  1410→            const r2 = (c.parameters.r2 as number) || 47000;
  1411→            const cap = (c.parameters.c as number) || 1e-5;
  1412→            const period = 0.693 * (r1 + 2 * r2) * cap;
  1413→            const f = 1 / period;
  1414→            if (f > 0 && f < minFreq) minFreq = f;
  1415→          }
  1416→        }
  1417→      }
  1418→
  1419→      if (minFreq !== Infinity && minFreq > 0) {
  1420→        const currentTime = result.sim.time;
  1421→        const phase = (currentTime * minFreq) % 1;
  1422→        const timeToRisingEdge = (1.0 - phase) / minFreq;
  1423→        const advance = Math.min(0.016, timeToRisingEdge + 0.001);
  1424→        if (advance > dt) {
  1425→          const newTime = currentTime + advance;
  1426→          prev.time = newTime;
  1427→          result.sim.time = newTime;
  1428→        }
  1429→      }
  1430→    }
  1431→    if (!result) {
  1432→      // Solver failed — likely singular matrix (no ground, or conflicting voltage sources)
  1433→      const hasGround = simComponents.some(c => c.type === 'ground');
  1434→      const errorMsg = !hasGround
  1435→        ? 'Simulation failed: No ground reference found. Add a Ground component to your circuit.'
  1436→        : 'Simulation failed: Singular matrix. Check for conflicting voltage sources or short circuits.';
  1437→      set({ running: false, paused: true, simError: errorMsg });
  1438→      return;
  1439→    }
  1440→    // update traces for oscilloscope components — when using hierarchy, the
  1441→    // active sheet's oscilloscope IDs match the flattened IDs (no prefix when
  1442→    // on root). When inside a sub-sheet, oscilloscope IDs need prefix lookup.
  1443→    const traces = [...s.traces];
  1444→    for (const comp of simComponents) {
  1445→      if (comp.type !== 'oscilloscope') continue;
  1446→      const plugin = plugins.get('oscilloscope');
  1447→      if (!plugin) continue;
  1448→      const terminals = getTerminalsForComponent(comp, plugin, result.nodeMap);
  1449→      const measurements = plugin.measure(comp.parameters, terminals, result.sim);
  1450→      const v = parseFloat(measurements[0]?.value ?? '0');
  1451→      // .PRINT directive: if printNodes is set, log matching node voltages to console
  1452→      if (s.simOptions.printNodes && s.simOptions.printNodes.length > 0) {
  1453→        for (const termKey of s.simOptions.printNodes) {
  1454→          const nodeId = result.nodeMap.terminalNode.get(termKey);
  1455→          if (nodeId != null && nodeId > 0) {
  1456→            const nodeV = result.sim.nodeVoltage[nodeId - 1] ?? 0;
  1457→            // eslint-disable-next-line no-console
  1458→            console.log(`[.PRINT t=${result.sim.time.toFixed(6)}] V(${termKey}) = ${nodeV.toFixed(6)} V`);
  1459→          }
  1460→        }
  1461→      }
  1462→      // When using hierarchy, the trace's componentId is the active sheet's local ID
  1463→      // (without prefix). Map back: if comp.id contains a '.', strip the prefix.
  1464→      const traceKey = usingHierarchy && comp.id.includes('.') ? comp.id.split('.').slice(1).join('.') : comp.id;
  1465→      let traceIdx = traces.findIndex((t) => t.componentId === traceKey);
  1466→      if (traceIdx < 0) {
  1467→        traces.push({
  1468→          componentId: traceKey,
  1469→          color: (comp.parameters.color as string) || '#22d3ee',
  1470→          label: (comp.parameters.label as string) || 'CH',
  1471→          samples: [],
  1472→        });
  1473→        traceIdx = traces.length - 1;
  1474→      }
  1475→      const trace = traces[traceIdx];
  1476→      trace.samples.push({ time: result.sim.time, voltage: v });
  1477→      if (trace.samples.length > s.maxTraceSamples) {
  1478→        trace.samples = trace.samples.slice(-s.maxTraceSamples);
  1479→      }
  1480→      traces[traceIdx] = { ...trace };
  1481→    }
  1482→    // Run physics validation (debug — catches simulation bugs).
  1483→    // Only runs when a special debug flag is set to avoid perf overhead in production.
  1484→    let physicsViolations: PhysicsViolation[] = s.physicsViolations;
  1485→    if (typeof window !== 'undefined' && (window as any).__PHYSICS_DEBUG__) {
  1486→      try {
  1487→        const validation = validatePhysics(simComponents, simWires, plugins, result.sim);
  1488→        physicsViolations = validation.violations;
  1489→        if (!validation.passed) {
  1490→          // eslint-disable-next-line no-console
  1491→          console.warn('[Physics] Violations detected:', validation.violations.length);
  1492→        }
  1493→      } catch (e) {
  1494→        // validation errors should never break the simulation
  1495→      }
  1496→    }
  1497→    set({ simContext: result.sim, traces, physicsViolations });
  1498→    } catch (err) {
  1499→      // Catch any unexpected errors from the simulation engine, plugins, or
  1500→      // hierarchy flattening. Prevents a single throwing plugin from killing
  1501→      // the RAF loop silently.
  1502→      const errorMsg = `Simulation error: ${(err as Error).message || 'Unknown error'}`;
  1503→      set({ running: false, paused: true, simError: errorMsg });
  1504→    }
  1505→  },

  1527→  runAnalysis: (config) => {
  1528→    const s = get();
  1529→    const plugins = new Map<string, any>();
  1530→    for (const c of s.components) {
