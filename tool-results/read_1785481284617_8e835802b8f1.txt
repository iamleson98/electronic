   205→    const vds = sim.nodeVoltage[d] - sim.nodeVoltage[s];

   207→    const key = `nmos_${d}_${g}_${s}`;

   213→      // gate is high-Z
   214→      sys.stampConductance(g, s, 1e-12);
   215→      return;
   216→    }
   217→    // On: if vds > (vgs - vth): saturation -> current source Id = Kp * (vgs-vth)^2
   218→    // Else: linear region -> approximately a small resistor (ron)
   219→    // For simplicity, model as a small resistor when on (linear-region switch model)
   220→    // OR as a VCCS in saturation. Pick based on vds.

   223→      // saturation: Id = Kp * vov^2 (current from d to s)
   224→      const kp = params.kp as number;
   225→      const id = kp * vov * vov;
   226→      sys.stampCurrentSource(d, s, id);
   227→      // also output resistance (channel modulation) — skip for simplicity
   228→    } else {
   229→      // linear: small resistor
   230→      sys.stampConductance(d, s, 1 / ron);
   231→    }
   232→    // gate high-Z
   233→    sys.stampConductance(g, s, 1e-12);

   834→    const minCurrent = 0.0001; // 0.1 mA minimum
   835→    const activity = isSimRunning && absCurrent > minCurrent
   836→      ? Math.min(1, absCurrent / 0.1) // full activity at 100mA
   837→      : 0;
   838→    // sound waves — animated when active, amplitude scales with current
   839→    const waveOffset = activity > 0.01 ? (sim!.time * 8) % 1 : 0;
   840→    const waveAmp = activity; // 0..1
   841→    ctx.strokeStyle = activity > 0.01 ? '#22d3ee' : '#475569';
   842→    ctx.lineWidth = 1.5 + activity * 1.5;
   843→    ctx.beginPath();
   844→    const r1 = 5 + waveAmp * Math.sin(waveOffset * Math.PI * 2) * 2;
   845→    const r2 = 9 + waveAmp * Math.sin(waveOffset * Math.PI * 2 + 1) * 2.5;
   846→    ctx.arc(2 * cellSize, cellSize, r1, -Math.PI / 3, Math.PI / 3);
   847→    ctx.moveTo(2 * cellSize + r2 * Math.cos(Math.PI / 3), cellSize - r2 * Math.sin(Math.PI / 3));
   848→    ctx.arc(2 * cellSize, cellSize, r2, -Math.PI / 3, Math.PI / 3);
   849→    ctx.stroke();
   850→    if (activity > 0.01) {
   851→      ctx.shadowColor = '#22d3ee';
   852→      ctx.shadowBlur = 4 + activity * 6;
   853→      ctx.stroke();
   854→      ctx.shadowBlur = 0;
   855→    }
   856→    // wire to right
   857→    ctx.beginPath();
   858→    ctx.moveTo(2 * cellSize, cellSize);
   859→    ctx.lineTo(3 * cellSize, cellSize);
   860→    ctx.strokeStyle = '#e2e8f0';
   861→    ctx.stroke();
   862→  },
   863→  stamp(params, terminals, sys) {
   864→    const r = Math.max(0.1, params.impedance as number);
   865→    const a = terminals.find((t) => t.terminalId === 'a')!.nodeId;
   866→    const b = terminals.find((t) => t.terminalId === 'b')!.nodeId;
   867→    sys.stampConductance(a, b, 1 / r);
   868→  },
   869→  measure(params, terminals, sim) {
   870→    const a = terminals.find((t) => t.terminalId === 'a')!.nodeId;
   871→    const b = terminals.find((t) => t.terminalId === 'b')!.nodeId;
   872→    const v = sim.nodeVoltage[a] - sim.nodeVoltage[b];
   873→    const r = Math.max(0.1, params.impedance as number);
   874→    const i = v / r;
   875→    const p = v * i;
   876→    return [
   877→      { label: 'V', value: v.toFixed(3), unit: 'V' },
   878→      { label: 'I', value: (i * 1000).toFixed(2), unit: 'mA' },
   879→      { label: 'P', value: (p * 1000).toFixed(2), unit: 'mW' },
   880→    ];
   881→  },
   882→};

   886→  type: 'photoresistor',
   887→  name: 'Photoresistor',
   888→  category: 'passive',
   889→  description: 'Light-dependent resistor. Resistance = darkR - (darkR - lightR) * light (0..1).',
   890→  symbol: 'LDR',
   891→  boundingBox: { width: 4, height: 2 },
   892→  terminals: [
   893→    { id: 'a', label: 'A', position: { x: 0, y: 1 } },
   894→    { id: 'b', label: 'B', position: { x: 4, y: 1 } },
   895→  ],
   896→  parameters: [
   897→    { key: 'darkR', label: 'Dark Resistance', type: 'number', default: 1000000, unit: 'Ω', min: 1, max: 1e9, step: 1000 },
   898→    { key: 'lightR', label: 'Light Resistance', type: 'number', default: 1000, unit: 'Ω', min: 0.1, max: 1e6, step: 100 },
   899→    { key: 'light', label: 'Light Level', type: 'number', default: 0.5, unit: '', min: 0, max: 1, step: 0.05 },
   900→  ],
   901→  render(ctx, params, cellSize) {
   902→    ctx.beginPath();
   903→    ctx.moveTo(0, cellSize);
   904→    ctx.lineTo(cellSize, cellSize);
   905→    ctx.moveTo(3 * cellSize, cellSize);
   906→    ctx.lineTo(4 * cellSize, cellSize);
   907→    ctx.stroke();
   908→    ctx.translate(2 * cellSize, cellSize);
   909→    // resistor body in dashed circle (LDR convention)
   910→    ctx.beginPath();
   911→    ctx.moveTo(-cellSize, 0);
   912→    for (let i = 0; i < 6; i++) {
   913→      const x = -cellSize + (i + 0.5) * (2 * cellSize / 6);
   914→      const y = (i % 2 === 0 ? -1 : 1) * 6;
   915→      ctx.lineTo(x, y);
   916→    }
   917→    ctx.lineTo(cellSize, 0);
   918→    ctx.stroke();
   919→    // arrows pointing in (light)
   920→    ctx.strokeStyle = '#fbbf24';
   921→    ctx.beginPath();
   922→    ctx.moveTo(-10, -14);
   923→    ctx.lineTo(-4, -8);
   924→    ctx.moveTo(-4, -8);
   925→    ctx.lineTo(-7, -8);
   926→    ctx.moveTo(-4, -8);
   927→    ctx.lineTo(-4, -11);
   928→    ctx.moveTo(-3, -16);
   929→    ctx.lineTo(3, -10);
   930→    ctx.moveTo(3, -10);
   931→    ctx.lineTo(0, -10);
   932→    ctx.moveTo(3, -10);
   933→    ctx.lineTo(3, -13);
   934→    ctx.stroke();
   935→    drawLabel(ctx, `${((params.light as number) * 100).toFixed(0)}%`, 0, 14);
   936→  },
   937→  stamp(params, terminals, sys) {
   938→    const darkR = params.darkR as number;
   939→    const lightR = params.lightR as number;
   940→    const light = Math.max(0, Math.min(1, params.light as number));
   941→    const r = darkR + (lightR - darkR) * light;
   942→    const a = terminals.find((t) => t.terminalId === 'a')!.nodeId;
   943→    const b = terminals.find((t) => t.terminalId === 'b')!.nodeId;
   944→    sys.stampConductance(a, b, 1 / Math.max(1e-6, r));
   945→  },
   946→  measure(params, terminals, sim) {
   947→    const darkR = params.darkR as number;
   948→    const lightR = params.lightR as number;
   949→    const light = Math.max(0, Math.min(1, params.light as number));
   950→    const r = darkR + (lightR - darkR) * light;
   951→    return [{ label: 'R', value: r >= 1000 ? `${(r / 1000).toFixed(2)}k` : r.toFixed(1), unit: 'Ω' }];
   952→  },
