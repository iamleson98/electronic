    13→    13→/** Cache key prefix for default (registered) models. */
    14→    14→const DEFAULT_KEY_PREFIX = 'default:';
    15→    15→/** Sentinel stored in the geometry cache to mark a model-URL load that failed. */
    16→    16→const FAILED_SENTINEL: unique symbol = Symbol('failed');
    17→    17→
    18→    18→/**
    19→    19→ * Look up (and lazily build) the default geometry for a component type.
    20→    20→ * Returns `null` if no default model is registered, or `{ failed: true }`
    21→    21→ * if parsing threw.
    22→    22→ */
    23→    23→function getDefaultGeometry(
    24→    24→  type: string,
    25→    25→  cache: Map<string, THREE.BufferGeometry | typeof FAILED_SENTINEL>,
    26→    26→): THREE.BufferGeometry | null {
    27→    27→  const key = `${DEFAULT_KEY_PREFIX}${type}`;
    28→    28→  const cached = cache.get(key);
    29→    29→  if (cached === FAILED_SENTINEL) return null;
    30→    30→  if (cached) return cached;
    31→    31→  const entry = DEFAULT_MODELS.get(type);
    32→    32→  if (!entry) return null;
    33→    33→  try {
    34→    34→    const model: LoadedModel = parseSTL(entry.stlAscii);
    35→    35→    const geo = modelToGeometry(model);
    36→    36→    cache.set(key, geo);
    37→    37→    return geo;
    38→    38→  } catch (err) {
    39→    39→    console.warn(`[PCB3DViewer] Failed to parse default model for ${type}:`, err);
    40→    40→    cache.set(key, FAILED_SENTINEL);
    41→    41→    return null;
    42→    42→  }
    43→    43→}

    45→    45→/** Fetch and parse a model from a URL, caching the resulting geometry. */
    46→    46→async function fetchModelGeometry(
    47→    47→  url: string,
    48→    48→  cache: Map<string, THREE.BufferGeometry | typeof FAILED_SENTINEL>,
    49→    49→): Promise<void> {
    50→    50→  const cached = cache.get(url);
    51→    51→  if (cached) return;
    52→    52→  try {
    53→    53→    const res = await fetch(url);
    54→    54→    if (!res.ok) throw new Error(`HTTP ${res.status} ${res.statusText}`);
    55→    55→    const buf = await res.arrayBuffer();
    56→    56→    // Parse by inferring the format from the URL's file extension.
    57→    57→    const lower = url.toLowerCase();
    58→    58→    let model: LoadedModel;
    59→    59→    if (lower.endsWith('.stl')) {
    60→    60→      model = parseSTL(buf);
    61→    61→    } else if (lower.endsWith('.wrl') || lower.endsWith('.vrml') || lower.endsWith('.x3dv')) {
    62→    62→      model = parseVRML(new TextDecoder().decode(buf));
    63→    63→    } else if (lower.endsWith('.obj')) {
    64→    64→      model = parseOBJ(new TextDecoder().decode(buf));
    65→    65→    } else if (lower.endsWith('.step') || lower.endsWith('.stp')) {
    66→    66→      model = parseModel(url, buf);
    67→    67→    } else {
    68→    68→      throw new Error(`Unknown model extension for ${url}`);
    69→    69→    }
    70→    70→    cache.set(url, modelToGeometry(model));
    71→    71→  } catch (err) {
    72→    72→    console.warn(`[PCB3DViewer] Failed to load model ${url}:`, err);
    73→    73→    cache.set(url, FAILED_SENTINEL);
    74→    74→  }
    75→    75→}

    77→    77→/** Compute the parametric body height used by the box fallback. */
    78→    78→function parametricBodyHeight(componentType: string): number {
    79→    79→  switch (componentType) {
    80→    80→    case 'dcVoltage':
    81→    81→    case 'acVoltage':
    82→    82→      return 5.0;
    83→    83→    case 'timer555':
    84→    84→    case 'opamp':
    85→    85→      return 3.5;
    86→    86→    case 'arduino':
    87→    87→    case 'arduinoReal':
    88→    88→      return 1.5;
    89→    89→    case 'led':
    90→    90→      return 0.6;
    91→    91→    default:
    92→    92→      return 1.0;
    93→    93→  }
    94→    94→}

    96→    96→/** Position a mesh so its model sits flush on the board surface. */
    97→    97→function positionMeshOnBoard(
    98→    98→  mesh: THREE.Mesh,
    99→    99→  geo: THREE.BufferGeometry,
   100→   100→  fp: Footprint,
   101→   101→  boardThickness: number,
   102→   102→  padHeight: number,
   103→   103→): void {
   104→   104→  geo.computeBoundingBox();
   105→   105→  const bb = geo.boundingBox;
   106→   106→  if (!bb) return;
   107→   107→  const minY = bb.min.y;
   108→   108→  const maxY = bb.max.y;
   109→   109→  if (fp.side === 'top') {
   110→   110→    // Bottom of model aligns with top of pads.
   111→   111→    mesh.position.y = padHeight - minY;
   112→   112→  } else {
   113→   113→    // Top of model aligns with bottom of pads (under the board).
   114→   114→    mesh.position.y = -boardThickness - padHeight - maxY;
   115→   115→  }
   116→   116→  mesh.position.x = fp.position.x;
   117→   117→  mesh.position.z = fp.position.y;
   118→   118→  mesh.rotation.y = (fp.rotation * Math.PI) / 180;
   119→   119→}

   337→   337→            });
   338→   338→          }

   344→   344→              if (mat instanceof THREE.MeshStandardMaterial && mat.emissiveIntensity > 0) {
   345→   345→                mat.emissive.setRGB(0, 0, 0);
   346→   346→                mat.emissiveIntensity = 0;
   347→   347→              }

   374→   374→      };

   376→   376→
   377→   377→      return () => {
   378→   378→        cancelAnimationFrame(raf);
   379→   379→        window.removeEventListener('resize', handleResize);
   380→   380→        if (rendererRef.current) {
   381→   381→          rendererRef.current.dispose();
   382→   382→          if (rendererRef.current.domElement.parentNode) {
   383→   383→            rendererRef.current.domElement.parentNode.removeChild(rendererRef.current.domElement);
   384→   384→          }
   385→   385→        }
   386→   386→      };
   387→   387→    };
   388→   388→
   389→   389→    let cleanupFn: (() => void) | null = null;
   390→   390→    initScene().then(fn => { if (typeof fn === 'function') cleanupFn = fn; });
   391→   391→
   392→   392→    return () => {
   393→   393→      cancelled = true;
   394→   394→      if (cleanupFn) cleanupFn();
   395→   395→    };
   396→   396→  }, []);

   398→   398→  // Update PCB 3D model when data changes
   399→   399→  useEffect(() => {
   400→   400→    if (!sceneRef.current || footprints.length === 0) return;
   401→   401→
   402→   402→    if (pcbGroupRef.current) {
   403→   403→      sceneRef.current.remove(pcbGroupRef.current);

   408→   408→          else child.material.dispose();

   410→   410→      });
   411→   411→    }

   413→   413→    const group = new THREE.Group();

   415→   415→    // Board substrate — realistic FR4 with gradient

   427→   427→    // Silkscreen border on top of board
   428→   428→    const silkGeo = new THREE.BoxGeometry(board.width - 2, 0.02, board.height - 2);
   429→   429→    const silkMat = new THREE.MeshStandardMaterial({ color: 0xe8e8e8, roughness: 0.9, metalness: 0.0 });
   430→   430→    const silkMesh = new THREE.Mesh(silkGeo, silkMat);
   431→   431→    silkMesh.position.set(board.width / 2, 0.01, board.height / 2);
   432→   432→    (silkMesh as any).__baseY = 0.01;
   433→   433→    group.add(silkMesh);
   434→   434→
   435→   435→    const PAD_HEIGHT = 0.05;
   436→   436→    const TRACE_HEIGHT = 0.05; // raised traces — visible embossing
   437→   437→
   438→   438→    // Traces — rendered as raised copper strips with metallic material
   439→   439→    for (const trace of traces) {
   440→   440→      for (let si = 0; si < trace.segments.length; si++) {
   441→   441→        const seg = trace.segments[si];
   442→   442→        const dx = seg.end.x - seg.start.x;
   443→   443→        const dy = seg.end.y - seg.start.y;
   444→   444→        const length = Math.hypot(dx, dy);
   445→   445→        if (length < 0.01) continue;
   446→   446→        const geo = new THREE.BoxGeometry(length, TRACE_HEIGHT, seg.width);
   447→   447→        // Copper material with high metalness for realistic look
   448→   448→        const mat = new THREE.MeshStandardMaterial({
   449→   449→          color: 0xb87333, roughness: 0.3, metalness: 0.9,
   450→   450→        });
   451→   451→        const mesh = new THREE.Mesh(geo, mat);
   452→   452→        const midX = (seg.start.x + seg.end.x) / 2;
   453→   453→        const midZ = (seg.start.y + seg.end.y) / 2;
   454→   454→        const y = trace.layer === 'top' ? TRACE_HEIGHT / 2 : -BOARD_THICKNESS - TRACE_HEIGHT / 2;
   455→   455→        mesh.position.set(midX, y, midZ);
   456→   456→        mesh.rotation.y = -Math.atan2(dy, dx);
   457→   457→        (mesh as any).__isTrace = true;
   458→   458→        (mesh as any).__flowOffset = si * 0.5;
   459→   459→        (mesh as any).__baseY = y;
   460→   460→        group.add(mesh);
   461→   461→      }
   462→   462→    }
   463→   463→
   464→   464→    // Vias — realistic copper-plated holes
   465→   465→    for (const via of vias) {
   466→   466→      const r = via.diameter / 2;
   467→   467→      const drillR = via.drill / 2;
   468→   468→      // Outer copper barrel
   469→   469→      const viaGeo = new THREE.CylinderGeometry(r, r, BOARD_THICKNESS + 0.02, 20);
   470→   470→      const viaMat = new THREE.MeshStandardMaterial({ color: 0xc8a020, roughness: 0.25, metalness: 0.95 });
   471→   471→      const viaMesh = new THREE.Mesh(viaGeo, viaMat);
   472→   472→      viaMesh.position.set(via.position.x, 0, via.position.y);
   473→   473→      viaMesh.rotation.x = Math.PI / 2;
   474→   474→      (viaMesh as any).__baseY = 0;
   475→   475→      group.add(viaMesh);
   476→   476→
   477→   477→      // Inner hole (dark)
   478→   478→      if (drillR > 0) {
   479→   479→        const holeGeo = new THREE.CylinderGeometry(drillR, drillR, BOARD_THICKNESS + 0.03, 16);
   480→   480→        const holeMat = new THREE.MeshStandardMaterial({ color: 0x0a0a0a, roughness: 1.0, metalness: 0.0 });
   481→   481→        const holeMesh = new THREE.Mesh(holeGeo, holeMat);
   482→   482→        holeMesh.position.set(via.position.x, 0, via.position.y);
   483→   483→        holeMesh.rotation.x = Math.PI / 2;
   484→   484→        (holeMesh as any).__baseY = 0;
   485→   485→        group.add(holeMesh);
   486→   486→      }
   487→   487→    }
   488→   488→
   489→   489→    // Footprints (components + pads) — realistic rendering
   490→   490→    const componentColors: Record<string, number> = {
   491→   491→      resistor: 0x2a2a3e, capacitor: 0x1a1a1a, led: 0xff3333, diode: 0x1a1a1a,
   492→   492→      timer555: 0x1a1a2e, opamp: 0x1a1a2e, npn: 0x2a2a2a, pnp: 0x2a2a2a,
   493→   493→      nmos: 0x2a2a2a, pmos: 0x2a2a2a, bsim3nmos: 0x2a2a2a, bsim3pmos: 0x2a2a2a,
   494→   494→      bsim4nmos: 0x2a2a2a, bsim4pmos: 0x2a2a2a,
   495→   495→      arduino: 0x005533, arduinoReal: 0x005533, raspberryPi: 0x7c2d12,
   496→   496→      switch: 0x333333, pushButton: 0x333333,
   497→   497→      crystal: 0x888888, inductor: 0x333333, speaker: 0x222222,
   498→   498→      transformer: 0x444444, vco: 0x1a1a2e, and: 0x1a1a2e, or: 0x1a1a2e,
   499→   499→      nand: 0x1a1a2e, nor: 0x1a1a2e, xor: 0x1a1a2e, not: 0x1a1a2e,
   500→   500→    };
   501→   501→
   502→   502→    /** Track which URLs we've already started fetching this render pass. */
   503→   503→    const inFlight = new Set<string>();
   504→   504→
   505→   505→    for (const fp of footprints) {
   506→   506→      const color = componentColors[fp.componentType] ?? 0x111111;
   507→   507→      const mat = new THREE.MeshStandardMaterial({
   508→   508→        color, roughness: 0.6, metalness: 0.2,
   509→   509→      });
   510→   510→
   511→   511→      // 1) External modelUrl takes precedence — look it up in the cache.
   512→   512→      //    If not cached yet, kick off an async fetch (handled below).
   513→   513→      let geo: THREE.BufferGeometry | null = null;
   514→   514→      const modelUrl = fp.modelUrl;
   515→   515→      if (modelUrl) {
   516→   516→        const cached = modelCacheRef.current.get(modelUrl);
   517→   517→        if (cached && cached !== FAILED_SENTINEL) {
   518→   518→          geo = cached;
   519→   519→        } else if (!cached && !inFlight.has(modelUrl)) {
   520→   520→          inFlight.add(modelUrl);
   521→   521→          // Fire-and-forget; when it resolves we bump modelVersion to trigger
   522→   522→          // a re-render, then the cached geometry is picked up synchronously.
   523→   523→          fetchModelGeometry(modelUrl, modelCacheRef.current).finally(bumpModelVersion);
   524→   524→        }
   525→   525→      }
   526→   526→
   527→   527→      // 2) Fall back to the default registered model for this type.
   528→   528→      if (!geo) {
   529→   529→        geo = getDefaultGeometry(fp.componentType, modelCacheRef.current);
   530→   530→      }
   531→   531→
   532→   532→      // 3) Final fallback: parametric box (matches the pre-model rendering).
   533→   533→      if (geo) {
   534→   534→        const mesh = new THREE.Mesh(geo, mat);
   535→   535→        positionMeshOnBoard(mesh, geo, fp, BOARD_THICKNESS, PAD_HEIGHT);
   536→   536→        group.add(mesh);
   537→   537→      } else {
   538→   538→        const bodyH = parametricBodyHeight(fp.componentType);
   539→   539→        const boxGeo = new THREE.BoxGeometry(fp.bodySize.width, bodyH, fp.bodySize.height);
   540→   540→        const mesh = new THREE.Mesh(boxGeo, mat);
   541→   541→        const y = fp.side === 'top'
   542→   542→          ? PAD_HEIGHT + bodyH / 2
   543→   543→          : -BOARD_THICKNESS - PAD_HEIGHT - bodyH / 2;
   544→   544→        mesh.position.set(fp.position.x, y, fp.position.y);
   545→   545→        mesh.rotation.y = (fp.rotation * Math.PI) / 180;
   546→   546→        // Tag for explosion/assembly animation
   547→   547→        (mesh as any).__baseY = y;
   548→   548→        group.add(mesh);
   549→   549→      }
   550→   550→
   551→   551→      // Pads — rendered as metallic discs/cylinders on the board surface
   552→   552→      for (const pad of fp.pads) {
   553→   553→        const r = Math.max(pad.size.width, pad.size.height) / 2;
   554→   554→        const padH = PAD_HEIGHT;
   555→   555→        // Use more segments for smoother circles
   556→   556→        const padGeo = pad.shape === 'circle'
   557→   557→          ? new THREE.CylinderGeometry(r, r, padH, 24)
   558→   558→          : new THREE.BoxGeometry(pad.size.width, padH, pad.size.height);
   559→   559→        // Metallic silver-gold pad material
   560→   560→        const padMat = new THREE.MeshStandardMaterial({
   561→   561→          color: 0xd4a020, roughness: 0.2, metalness: 0.95,
   562→   562→        });
   563→   563→        const padMesh = new THREE.Mesh(padGeo, padMat);
   564→   564→        const padY = fp.side === 'top' ? padH / 2 : -BOARD_THICKNESS - padH / 2;
   565→   565→        padMesh.position.set(pad.position.x, padY, pad.position.y);
   566→   566→        if (pad.shape === 'circle') padMesh.rotation.x = Math.PI / 2;
   567→   567→        (padMesh as any).__baseY = padY;
   568→   568→        group.add(padMesh);
   569→   569→
   570→   570→        // Drill hole for THT pads
   571→   571→        if (pad.drill && pad.drill > 0) {
   572→   572→          const drillR = pad.drill / 2;
   573→   573→          const drillGeo = new THREE.CylinderGeometry(drillR, drillR, BOARD_THICKNESS + 0.05, 16);
   574→   574→          const drillMat = new THREE.MeshStandardMaterial({ color: 0x0a0a0a, roughness: 1.0, metalness: 0.0 });
   575→   575→          const drillMesh = new THREE.Mesh(drillGeo, drillMat);
   576→   576→          drillMesh.position.set(pad.position.x, 0, pad.position.y);
   577→   577→          drillMesh.rotation.x = Math.PI / 2;
   578→   578→          (drillMesh as any).__baseY = 0;
   579→   579→          group.add(drillMesh);
   580→   580→        }
   581→   581→      }
   582→   582→    }
   583→   583→
   584→   584→    sceneRef.current.add(group);
   585→   585→    pcbGroupRef.current = group;
   586→   586→
   587→   587→    // Auto-frame camera
   588→   588→    if (cameraRef.current && controlsRef.current) {
   589→   589→      const centerX = board.width / 2;
   590→   590→      const centerZ = board.height / 2;
   591→   591→      const maxDim = Math.max(board.width, board.height);
   592→   592→      const dist = maxDim * 1.5;
   593→   593→      cameraRef.current.position.set(centerX + dist * 0.5, dist * 0.8, centerZ + dist * 0.5);
   594→   594→      controlsRef.current.target.set(centerX, 0, centerZ);
   595→   595→      controlsRef.current.update();
   596→   596→    }
   597→   597→  }, [footprints, traces, vias, board, padNets, modelVersion, bumpModelVersion]);
   598→   598→
   599→   599→  return (
   600→   600→    <div className="relative h-full w-full overflow-hidden bg-[#0a1628]"
   601→   601→      onContextMenu={(e) => e.preventDefault()}>
   602→   602→      <div ref={containerRef} className="absolute inset-0" />
   603→   603→      {loading && (
   604→   604→        <div className="absolute inset-0 flex items-center justify-center">
   605→   605→          <div className="text-sm text-slate-400">Loading 3D viewer...</div>
   606→   606→        </div>
   607→   607→      )}
   608→   608→      {error && (
   609→   609→        <div className="absolute inset-0 flex items-center justify-center">
   610→   610→          <div className="text-sm text-rose-400">{error}</div>
   611→   611→        </div>
   612→   612→      )}
   613→   613→      {footprints.length === 0 && !loading && !error && (
   614→   614→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-3">
   615→   615→          <div className="text-lg font-semibold text-slate-300">3D PCB Preview</div>
   616→   616→          <div className="text-sm text-slate-500">
   617→   617→            Import a schematic in PCB Layout mode first, then switch to 3D View.
   618→   618→          </div>
   619→   619→        </div>
   620→   620→      )}
   621→   621→      {!loading && !error && footprints.length > 0 && (
   622→   622→        <>
   623→   623→          {/* 3D Viewer control panel — top-right */}
   624→   624→          <div className="absolute right-2 top-2 flex flex-col gap-1 rounded-md bg-slate-900/90 p-2 text-xs">
   625→   625→            <button
   626→   626→              onClick={() => setCrossSection(!crossSection)}
   627→   627→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   628→   628→                crossSection ? 'bg-cyan-500/30 text-cyan-300 border border-cyan-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   629→   629→              }`}
   630→   630→            >
   631→   631→              🔪 Cross-section
   632→   632→            </button>
   633→   633→            {crossSection && (
   634→   634→              <input
   635→   635→                type="range" min={-2} max={5} step={0.1}
   636→   636→                value={crossSectionY}
   637→   637→                onChange={(e) => setCrossSectionY(parseFloat(e.target.value))}
   638→   638→                className="w-full"
   639→   639→              />
   640→   640→            )}
   641→   641→            <button
   642→   642→              onClick={() => setHighQuality(!highQuality)}
   643→   643→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   644→   644→                highQuality ? 'bg-amber-500/30 text-amber-300 border border-amber-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   645→   645→              }`}
   646→   646→            >
   647→   647→              ✨ High-quality render
   648→   648→            </button>
   649→   649→            <button
   650→   650→              onClick={() => setShowCurrentFlow(!showCurrentFlow)}
   651→   651→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   652→   652→                showCurrentFlow ? 'bg-emerald-500/30 text-emerald-300 border border-emerald-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   653→   653→              }`}
   654→   654→            >
   655→   655→              ⚡ Current flow {running ? '' : '(run sim)'}
   656→   656→            </button>
   657→   657→            <button
   658→   658→              onClick={() => setShowVoltageProbes(!showVoltageProbes)}
   659→   659→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   660→   660→                showVoltageProbes ? 'bg-blue-500/30 text-blue-300 border border-blue-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   661→   661→              }`}
   662→   662→            >
   663→   663→              📊 Voltage probes
   664→   664→            </button>
   665→   665→            <div className="flex items-center gap-1 px-2 py-1 text-slate-300">
   666→   666→              💥 Explode
   667→   667→              <input
   668→   668→                type="range" min={0} max={1} step={0.05}
   669→   669→                value={explosionFactor}

   693→   693→              className="flex items-center gap-1 rounded px-2 py-1 text-purple-300 hover:bg-slate-800 border border-transparent"
   694→   694→            >
   695→   695→              ▶ Play assembly

   697→   697→          </div>
   698→   698→
   699→   699→          {/* Voltage probe labels — floating HTML at projected 3D positions */}

   705→   705→            const x = (pos.x + 1) / 2 * (rendererRef.current.domElement.width / window.devicePixelRatio);

   721→   721→              </div>

   724→   724→
   725→   725→          {/* Original bottom overlay */}
   726→   726→          <div className="pointer-events-none absolute bottom-2 left-2 rounded-md bg-[#0a1628]/80 px-3 py-1.5 text-xs font-mono text-slate-400">
   727→   727→            Left-drag: rotate · Right-drag: pan · Scroll: zoom
   728→   728→            {showCurrentFlow && running && <span className="ml-2 text-emerald-300">· ⚡ current flowing</span>}
   729→   729→          </div>
   730→   730→        </>
   731→   731→      )}
   732→   732→    </div>
   733→   733→  );
