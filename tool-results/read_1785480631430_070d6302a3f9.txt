    52→    52→/** Get the orthogonal path points for a wire.
    53→    53→ *  fromPos and toPos are in SCREEN coords. Waypoints are in GRID coords and
    54→    54→ *  are converted to screen coords using the provided converter. */
    55→    55→function getWirePath(
    56→    56→  wire: Wire,
    57→    57→  fromPos: Vec2,
    58→    58→  toPos: Vec2,
    59→    59→  gridToScreenFn: (gx: number, gy: number) => Vec2,
    60→    60→  use45: boolean = false,
    61→    61→): Vec2[] {
    62→    62→  const points: Vec2[] = [fromPos];
    63→    63→  if (wire.waypoints && wire.waypoints.length > 0) {
    64→    64→    for (const wp of wire.waypoints) {
    65→    65→      points.push(gridToScreenFn(wp.x, wp.y));
    66→    66→    }
    67→    67→  } else if (use45) {
    68→    68→    // 45-degree routing: L-shape with 45° diagonal in the middle
    69→    69→    const dx = toPos.x - fromPos.x;
    70→    70→    const dy = toPos.y - fromPos.y;
    71→    71→    const absDx = Math.abs(dx);
    72→    72→    const absDy = Math.abs(dy);
    73→    73→    if (absDx < 1 || absDy < 1) {
    74→    74→      // Nearly straight — just go direct
    75→    75→      points.push({ x: toPos.x, y: fromPos.y });
    76→    76→    } else if (absDx > absDy) {
    77→    77→      // More horizontal: go horizontal, then 45° diagonal, then horizontal
    78→    78→      const diagLen = absDy;
    79→    79→      const sign = Math.sign(dy);
    80→    80→      const horizSign = Math.sign(dx);
    81→    81→      const midX1 = fromPos.x + horizSign * (absDx - diagLen) / 2;
    82→    82→      const midX2 = midX1 + horizSign * diagLen;
    83→    83→      points.push({ x: midX1, y: fromPos.y });
    84→    84→      points.push({ x: midX2, y: fromPos.y + sign * diagLen });
    85→    85→      points.push({ x: midX2, y: toPos.y });
    86→    86→    } else {
    87→    87→      // More vertical: go vertical, then 45° diagonal, then vertical
    88→    88→      const diagLen = absDx;
    89→    89→      const sign = Math.sign(dx);
    90→    90→      const vertSign = Math.sign(dy);
    91→    91→      const midY1 = fromPos.y + vertSign * (absDy - diagLen) / 2;
    92→    92→      const midY2 = midY1 + vertSign * diagLen;
    93→    93→      points.push({ x: fromPos.x, y: midY1 });
    94→    94→      points.push({ x: fromPos.x + sign * diagLen, y: midY2 });
    95→    95→      points.push({ x: toPos.x, y: midY2 });
    96→    96→    }
    97→    97→  } else {
    98→    98→    // default orthogonal routing: go to midpoint X, then to target
    99→    99→    const midX = (fromPos.x + toPos.x) / 2;
   100→   100→    points.push({ x: midX, y: fromPos.y });
   101→   101→    points.push({ x: midX, y: toPos.y });
   102→   102→  }
   103→   103→  points.push(toPos);
   104→   104→  return points;
   105→   105→}

   107→   107→/** Compute the midpoint of a segment for drag handle detection */
   108→   108→function segmentMidpoint(a: Vec2, b: Vec2): Vec2 {
   109→   109→  return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
   110→   110→}

   112→   112→/** Angle (in degrees, 0-360) from a center point to a target point */
   113→   113→function angleFromCenter(cx: number, cy: number, x: number, y: number): number {
   114→   114→  const angle = Math.atan2(y - cy, x - cx) * 180 / Math.PI;
   115→   115→  return (angle + 360) % 360;
   116→   116→}

   118→   118→/**
   119→   119→ * Re-route a wire so a dragged segment follows the cursor freely in 2D.
   120→   120→ *
   121→   121→ * The wire is ALWAYS kept orthogonal (only horizontal/vertical segments).
   122→   122→ * Terminal positions (fromPos, toPos) are NEVER moved — they're fixed to
   123→   123→ * component terminals. Only waypoints change.
   124→   124→ *
   125→   125→ * Path layout: [fromPos, wp1, wp2, ..., toPos]
   126→   126→ *   segIndex 0 = fromPos→wp1, segIndex 1 = wp1→wp2, ..., last = wpN→toPos
   127→   127→ *
   128→   128→ * When the user drags a segment handle, we move that segment to the cursor
   129→   129→ * position. The dragged segment stays horizontal or vertical (whichever it
   130→   130→ * was), and its perpendicular axis snaps to the cursor. The parallel axis
   131→   131→ * also follows the cursor when possible (for interior segments with waypoints
   132→   132→ * on both ends). Adjacent segments naturally stretch/shrink to stay connected.
   133→   133→ */
   134→   134→function rerouteWireForDrag(
   135→   135→  fromPos: Vec2,
   136→   136→  toPos: Vec2,
   137→   137→  segIndex: number,
   138→   138→  cursorGrid: Vec2,
   139→   139→  existingWaypoints: Vec2[],
   140→   140→): Vec2[] {
   141→   141→  // Build waypoint copy (never touch fromPos/toPos)
   142→   142→  const wps: Vec2[] = existingWaypoints.length > 0
   143→   143→    ? existingWaypoints.map((w) => ({ ...w }))
   144→   144→    : [
   145→   145→        { x: (fromPos.x + toPos.x) / 2, y: fromPos.y },
   146→   146→        { x: (fromPos.x + toPos.x) / 2, y: toPos.y },
   147→   147→      ];
   148→   148→
   149→   149→  // Helper: get path point by index (0=fromPos, 1..n=waypoints, n+1=toPos)
   150→   150→  const getPt = (i: number): Vec2 => {
   151→   151→    if (i === 0) return fromPos;
   152→   152→    if (i === wps.length + 1) return toPos;
   153→   153→    return wps[i - 1];
   154→   154→  };
   155→   155→  // Helper: set a waypoint by path index (only waypoints are settable)
   156→   156→  const setPt = (i: number, val: Vec2) => {
   157→   157→    if (i > 0 && i <= wps.length) wps[i - 1] = val;
   158→   158→  };
   159→   159→
   160→   160→  const a = getPt(segIndex);
   161→   161→  const b = getPt(segIndex + 1);
   162→   162→  const isHorizontal = Math.abs(b.y - a.y) < Math.abs(b.x - a.x);
   163→   163→
   164→   164→  if (isHorizontal) {
   165→   165→    // Horizontal segment: perpendicular axis = Y. Cursor Y becomes the new Y.
   166→   166→    // Parallel axis = X: shift segment along X toward cursor (only waypoints move).
   167→   167→    const newY = cursorGrid.y;
   168→   168→    const midX = (a.x + b.x) / 2;
   169→   169→    const deltaX = cursorGrid.x - midX;
   170→   170→    const aIdx = segIndex;
   171→   171→    const bIdx = segIndex + 1;
   172→   172→    const aIsFixed = (aIdx === 0);           // fromPos can't move
   173→   173→    const bIsFixed = (bIdx === wps.length + 1); // toPos can't move
   174→   174→    // Set Y on both endpoints (waypoints only; fixed terminals keep their Y)
   175→   175→    const newA = { ...getPt(aIdx) };
   176→   176→    const newB = { ...getPt(bIdx) };
   177→   177→    if (!aIsFixed) newA.y = newY;
   178→   178→    if (!bIsFixed) newB.y = newY;
   179→   179→    // Shift X: both endpoints if both are waypoints; only one if the other is fixed
   180→   180→    if (!aIsFixed && !bIsFixed) {
   181→   181→      newA.x += deltaX;
   182→   182→      newB.x += deltaX;
   183→   183→    } else if (aIsFixed && !bIsFixed) {
   184→   184→      newB.x += deltaX;
   185→   185→    } else if (!aIsFixed && bIsFixed) {
   186→   186→      newA.x += deltaX;
   187→   187→    }
   188→   188→    setPt(aIdx, newA);
   189→   189→    setPt(bIdx, newB);
   190→   190→  } else {
   191→   191→    // Vertical segment: perpendicular axis = X. Cursor X becomes the new X.
   192→   192→    // Parallel axis = Y: shift segment along Y toward cursor.
   193→   193→    const newX = cursorGrid.x;
   194→   194→    const midY = (a.y + b.y) / 2;
   195→   195→    const deltaY = cursorGrid.y - midY;
   196→   196→    const aIdx = segIndex;
   197→   197→    const bIdx = segIndex + 1;
   198→   198→    const aIsFixed = (aIdx === 0);
   199→   199→    const bIsFixed = (bIdx === wps.length + 1);
   200→   200→    const newA = { ...getPt(aIdx) };
   201→   201→    const newB = { ...getPt(bIdx) };
   202→   202→    if (!aIsFixed) newA.x = newX;
   203→   203→    if (!bIsFixed) newB.x = newX;
   204→   204→    if (!aIsFixed && !bIsFixed) {
   205→   205→      newA.y += deltaY;
   206→   206→      newB.y += deltaY;
   207→   207→    } else if (aIsFixed && !bIsFixed) {
   208→   208→      newB.y += deltaY;
   209→   209→    } else if (!aIsFixed && bIsFixed) {
   210→   210→      newA.y += deltaY;
   211→   211→    }
   212→   212→    setPt(aIdx, newA);
   213→   213→    setPt(bIdx, newB);
   214→   214→  }
   215→   215→
   216→   216→  // After moving waypoints, the path may have diagonal segments adjacent to
   217→   217→  // fixed terminals. Re-orthogonalize by inserting elbow waypoints.
   218→   218→  return orthogonalizePath(fromPos, toPos, wps);
   219→   219→}
   220→   220→
   221→   221→/**
   222→   222→ * Ensure a wire path is fully orthogonal (no diagonal segments).
   223→   223→ * If moving a waypoint created a diagonal segment adjacent to a fixed terminal,
   224→   224→ * insert an extra elbow waypoint to break it into two orthogonal segments.
   225→   225→ *
   226→   226→ * This is called after rerouteWireForDrag to clean up any diagonals.
   227→   227→ */
   228→   228→function orthogonalizePath(fromPos: Vec2, toPos: Vec2, wps: Vec2[]): Vec2[] {
   229→   229→  const result: Vec2[] = [];
   230→   230→  const full: Vec2[] = [fromPos, ...wps, toPos];
   231→   231→  for (let i = 0; i < full.length - 1; i++) {
   232→   232→    const a = full[i];
   233→   233→    const b = full[i + 1];
   234→   234→    result.push({ ...a });
   235→   235→    // If segment is diagonal, insert an elbow (go horizontal first, then vertical)
   236→   236→    if (Math.abs(a.x - b.x) > 0.01 && Math.abs(a.y - b.y) > 0.01) {
   237→   237→      // Insert elbow at (b.x, a.y) — horizontal first, then vertical
   238→   238→      result.push({ x: b.x, y: a.y });
   239→   239→    }
   240→   240→  }
   241→   241→  result.push({ ...toPos });
   242→   242→  // Convert back to waypoints (exclude fromPos and toPos)
   243→   243→  return result.slice(1, -1);
   244→   244→}

  1190→  1190→      // Screen angle: 0=right, 90=down, 180=left, 270=up.

  1192→  1192→      // But this mapping is relative to the component's "forward" direction. A simpler

  1194→  1194→      // and add to startRotation.

  1196→  1196→      // normalize to -180..180

  1198→  1198→      while (deltaDeg < -180) deltaDeg += 360;

  1209→  1209→          rotateComponent(comp.id);
  1210→  1210→        }

  1232→  1232→      // This gives smooth, predictable behavior: the segment follows the cursor
  1233→  1233→      // absolutely, and the path is rebuilt fresh each frame from the original.

  1235→  1235→        { ...fromPos },
  1236→  1236→        { ...toPos },
  1237→  1237→        wd.segIndex,
  1238→  1238→        g,

  1242→  1242→      return;
  1243→  1243→    }
  1244→  1244→
  1245→  1245→    if (dragRef.current) {

  1249→  1249→          x: g.x - (dragRef.current.lastGrid?.x ?? g.x),
  1250→  1250→          y: g.y - (dragRef.current.lastGrid?.y ?? g.y),
  1251→  1251→        };

  1253→  1253→          useEditor.getState().moveSelectedComponents(delta);
  1254→  1254→        }
  1255→  1255→        dragRef.current.lastGrid = { ...g };
  1256→  1256→        return;
  1257→  1257→      }

  1259→  1259→      moveComponent(dragRef.current.componentId, newPos);
  1260→  1260→      return;
  1261→  1261→    }
  1262→  1262→
  1263→  1263→    if (wireDraft) {
  1264→  1264→      updateWireCursor(g);
  1265→  1265→    }
  1266→  1266→

  1269→  1269→    if (term) {
  1270→  1270→      setHover({ componentId: term.componentId, terminal: term, wireId: null, wireHandle: null, rotateHandle: null });
  1271→  1271→      return;
  1272→  1272→    }

  1275→  1275→      const comp = components.find((c) => c.id === selection.id);

  1280→  1280→          const dy = sy - handlePos.y;

  1282→  1282→            setHover({ componentId: comp.id, terminal: null, wireId: null, wireHandle: null, rotateHandle: comp.id });
  1283→  1283→            return;
  1284→  1284→          }
  1285→  1285→        }
  1286→  1286→      }
  1287→  1287→    }

  1289→  1289→    if (!running) {
  1290→  1290→      const wh = findWireHandle(sx, sy);
  1291→  1291→      if (wh) {

  1293→  1293→        return;
  1294→  1294→      }

  1296→  1296→    // wire

  1300→  1300→      return;
  1301→  1301→    }
  1302→  1302→    // component
  1303→  1303→    const comp = findComponentAt(g.x, g.y);

  1322→  1322→    }
  1323→  1323→    if (dragRef.current) {
  1324→  1324→      // History was already pushed at drag start

  1330→  1330→    const rect = canvasRef.current!.getBoundingClientRect();
  1331→  1331→    const sx = e.clientX - rect.left;
  1332→  1332→    const sy = e.clientY - rect.top;
  1333→  1333→    const factor = e.deltaY < 0 ? 1.1 : 1 / 1.1;
  1334→  1334→    const newZoom = Math.max(0.4, Math.min(4, zoom * factor));
  1335→  1335→    // Keep the grid point under the cursor stationary during zoom.

  1382→  1382→      }

  1384→  1384→    }

  1387→  1387→    if (hover.wireHandle) return 'move';
  1388→  1388→    if (hover.wireId) return 'pointer';
  1389→  1389→    if (hover.componentId) return 'move';

  1391→  1391→  };

  1405→  1405→        } else if (s.selection.type === 'wire') {

  1407→  1407→        }
  1408→  1408→      } else if (e.key === 'r' || e.key === 'R') {
  1409→  1409→        if (running) return;

  1412→  1412→      } else if (e.key === 'Escape') {

  1414→  1414→        setSelection({ type: null, id: null });

  1417→  1417→        if (running) return;

  1422→  1422→        e.preventDefault();
  1423→  1423→        useEditor.getState().redo();

  1426→  1426→        e.preventDefault();

  1430→  1430→        e.preventDefault();
  1431→  1431→        useEditor.getState().paste();
  1432→  1432→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'd') {

  1434→  1434→        e.preventDefault();
  1435→  1435→        useEditor.getState().duplicate();

  1481→  1481→      <div className="pointer-events-none absolute bottom-2 right-2 rounded-md bg-slate-900/80 px-2 py-1 text-xs font-mono text-slate-400">

  1484→  1484→      {/* zoom controls */}

  1489→  1489→        >+</button>

  1491→  1491→          onClick={() => setZoom((z) => Math.max(0.4, z / 1.2))}
  1492→  1492→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700"
  1493→  1493→        >−</button>

  1497→  1497→        >⌂</button>
  1498→  1498→      </div>
  1499→  1499→    </div>
  1500→  1500→  );
  1501→  1501→}

  1503→  1503→/** Distance from point (px, py) to segment (ax,ay)-(bx,by) */
  1504→  1504→function pointToSegmentDist(px: number, py: number, ax: number, ay: number, bx: number, by: number): number {
  1505→  1505→  const dx = bx - ax;
  1506→  1506→  const dy = by - ay;
  1507→  1507→  const lenSq = dx * dx + dy * dy;
  1508→  1508→  if (lenSq === 0) return Math.hypot(px - ax, py - ay);
  1509→  1509→  let t = ((px - ax) * dx + (py - ay) * dy) / lenSq;
  1510→  1510→  t = Math.max(0, Math.min(1, t));
  1511→  1511→  const cx = ax + t * dx;
  1512→  1512→  const cy = ay + t * dy;
  1513→  1513→  return Math.hypot(px - cx, py - cy);
  1514→  1514→}
  1515→  1515→
