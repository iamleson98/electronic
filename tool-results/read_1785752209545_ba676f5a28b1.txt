   274→   274→/** Get the reference designator prefix for a component type */
   275→   275→function refdesPrefix(type: string): string {
   276→   276→  switch (type) {
   277→   277→    case 'resistor': return 'R';
   278→   278→    case 'capacitor': return 'C';
   279→   279→    case 'inductor': return 'L';
   280→   280→    case 'led': return 'LED';
   281→   281→    case 'diode': return 'D';
   282→   282→    case 'zener': return 'DZ';
   283→   283→    case 'schottky': return 'D';
   284→   284→    case 'dcVoltage':
   285→   285→    case 'acVoltage':
   286→   286→    case 'pulseSource': return 'V';
   287→   287→    case 'currentSource': return 'I';
   288→   288→    case 'timer555':
   289→   289→    case 'opamp':
   290→   290→    case 'opampRails':
   291→   291→    case 'voltageRegulator':
   292→   292→    case 'vco': return 'U';
   293→   293→    case 'npn':
   294→   294→    case 'pnp':
   295→   295→    case 'nmos':
   296→   296→    case 'pmos': return 'Q';
   297→   297→    case 'switch':
   298→   298→    case 'pushButton': return 'SW';
   299→   299→    case 'fuse': return 'F';
   300→   300→    case 'crystal': return 'Y';
   301→   301→    case 'transformer': return 'T';
   302→   302→    case 'speaker':
   303→   303→    case 'buzzer': return 'LS';
   304→   304→    case 'dcMotor': return 'M';
   305→   305→    case 'photoresistor': return 'LDR';
   306→   306→    case 'sevenSegment': return 'DSP';
   307→   307→    case 'ground':
   308→   308→    case 'powerGND': return 'GND';
   309→   309→    case 'junction': return 'J';
   310→   310→    case 'oscilloscope':
   311→   311→    case 'voltmeter':
   312→   312→    case 'ammeter': return 'TP';
   313→   313→    case 'arduino':
   314→   314→    case 'arduinoReal':
   315→   315→    case 'raspberryPi': return 'U';
   316→   316→    case 'netLabel':
   317→   317→    case 'busLabel': return 'NL';
   318→   318→    case 'bus': return 'BUS';
   319→   319→    default:
   320→   320→      if (type.startsWith('7400_') || type.startsWith('7402_') || type.startsWith('7404_') ||
   321→   321→          type.startsWith('7408_') || type.startsWith('7432_') || type.startsWith('7486_') ||
   322→   322→          type.startsWith('7474_')) return 'U';
   323→   323→      if (type.startsWith('custom_')) return 'U';
   324→   324→      return 'U';
   325→   325→  }
   326→   326→}

   328→   328→/** Generate the next refdes for a given type, based on existing components */
   329→   329→function nextRefdes(type: string, components: CircuitComponent[]): string {
   330→   330→  const prefix = refdesPrefix(type);
   331→   331→  let maxNum = 0;
   332→   332→  for (const c of components) {
   333→   333→    if (refdesPrefix(c.type) === prefix) {
   334→   334→      const ref = c.refdes ?? c.id;
   335→   335→      const match = ref.match(/(\d+)$/);
   336→   336→      if (match) {
   337→   337→        const n = parseInt(match[1]);
   338→   338→        if (n > maxNum) maxNum = n;
   339→   339→      }
   340→   340→    }
   341→   341→  }
   342→   342→  return `${prefix}${maxNum + 1}`;
   343→   343→}

   345→   345→function snapshot(s: {
   346→   346→  components: CircuitComponent[];
   347→   347→  wires: Wire[];
   348→   348→  drawings?: DrawingPrimitive[];
   349→   349→  noConnects?: NoConnectMarker[];
   350→   350→  groups?: Group[];
   351→   351→  sheets?: HierarchicalSheet[];
   352→   352→  netClasses?: NetClass[];
   353→   353→}) {
   354→   354→  return {
   355→   355→    components: s.components.map((c) => ({ ...c, parameters: { ...c.parameters }, simState: undefined, fields: c.fields ? c.fields.map((f) => ({ ...f })) : undefined })),
   356→   356→    wires: s.wires.map((w) => ({ ...w })),
   357→   357→    drawings: s.drawings ? s.drawings.map((d) => ({ ...d })) : [],
   358→   358→    noConnects: s.noConnects ? s.noConnects.map((n) => ({ ...n })) : [],
   359→   359→    groups: s.groups ? s.groups.map((g) => ({ ...g, componentIds: [...g.componentIds], wireIds: [...g.wireIds], drawingIds: [...g.drawingIds] })) : [],
   360→   360→    sheets: s.sheets ? s.sheets.map((sh) => ({ ...sh, pins: sh.pins.map((p) => ({ ...p })) })) : [],
   361→   361→    netClasses: s.netClasses ? s.netClasses.map((nc) => ({ ...nc, nets: [...nc.nets] })) : [],
   362→   362→  };
   363→   363→}

   368→   368→function defaultsFor(type: string): Record<string, number | string | boolean> {
   369→   369→  const plugin = getPlugin(type);
   370→   370→  if (!plugin) return {};
   371→   371→  const defaults: Record<string, number | string | boolean> = {};
   372→   372→  for (const p of plugin.parameters) defaults[p.key] = p.default;
   373→   373→  return defaults;
