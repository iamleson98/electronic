    52→/** Get the orthogonal path points for a wire.
    53→ *  fromPos and toPos are in SCREEN coords. Waypoints are in GRID coords and
    54→ *  are converted to screen coords using the provided converter. */
    55→function getWirePath(
    56→  wire: Wire,
    57→  fromPos: Vec2,
    58→  toPos: Vec2,
    59→  gridToScreenFn: (gx: number, gy: number) => Vec2,
    60→  use45: boolean = false,
    61→): Vec2[] {
    62→  const points: Vec2[] = [fromPos];
    63→  if (wire.waypoints && wire.waypoints.length > 0) {
    64→    for (const wp of wire.waypoints) {
    65→      points.push(gridToScreenFn(wp.x, wp.y));
    66→    }
    67→  } else if (use45) {
    68→    // 45-degree routing: L-shape with 45° diagonal in the middle
    69→    const dx = toPos.x - fromPos.x;
    70→    const dy = toPos.y - fromPos.y;
    71→    const absDx = Math.abs(dx);
    72→    const absDy = Math.abs(dy);
    73→    if (absDx < 1 || absDy < 1) {
    74→      // Nearly straight — just go direct
    75→      points.push({ x: toPos.x, y: fromPos.y });
    76→    } else if (absDx > absDy) {
    77→      // More horizontal: go horizontal, then 45° diagonal, then horizontal
    78→      const diagLen = absDy;
    79→      const sign = Math.sign(dy);
    80→      const horizSign = Math.sign(dx);
    81→      const midX1 = fromPos.x + horizSign * (absDx - diagLen) / 2;
    82→      const midX2 = midX1 + horizSign * diagLen;
    83→      points.push({ x: midX1, y: fromPos.y });
    84→      points.push({ x: midX2, y: fromPos.y + sign * diagLen });
    85→      points.push({ x: midX2, y: toPos.y });
    86→    } else {
    87→      // More vertical: go vertical, then 45° diagonal, then vertical
    88→      const diagLen = absDx;
    89→      const sign = Math.sign(dx);
    90→      const vertSign = Math.sign(dy);
    91→      const midY1 = fromPos.y + vertSign * (absDy - diagLen) / 2;
    92→      const midY2 = midY1 + vertSign * diagLen;
    93→      points.push({ x: fromPos.x, y: midY1 });
    94→      points.push({ x: fromPos.x + sign * diagLen, y: midY2 });
    95→      points.push({ x: toPos.x, y: midY2 });
    96→    }
    97→  } else {
    98→    // default orthogonal routing: go to midpoint X, then to target
    99→    const midX = (fromPos.x + toPos.x) / 2;
   100→    points.push({ x: midX, y: fromPos.y });
   101→    points.push({ x: midX, y: toPos.y });
   102→  }
   103→  points.push(toPos);
   104→  return points;
   105→}

   107→/** Compute the midpoint of a segment for drag handle detection */
   108→function segmentMidpoint(a: Vec2, b: Vec2): Vec2 {
   109→  return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
   110→}

   112→/** Angle (in degrees, 0-360) from a center point to a target point */
   113→function angleFromCenter(cx: number, cy: number, x: number, y: number): number {
   114→  const angle = Math.atan2(y - cy, x - cx) * 180 / Math.PI;
   115→  return (angle + 360) % 360;
   116→}

   118→/**
   119→ * Re-route a wire so a dragged segment follows the cursor freely in 2D.
   120→ *
   121→ * The wire is ALWAYS kept orthogonal (only horizontal/vertical segments).
   122→ * Terminal positions (fromPos, toPos) are NEVER moved — they're fixed to
   123→ * component terminals. Only waypoints change.
   124→ *
   125→ * Path layout: [fromPos, wp1, wp2, ..., toPos]
   126→ *   segIndex 0 = fromPos→wp1, segIndex 1 = wp1→wp2, ..., last = wpN→toPos
   127→ *
   128→ * When the user drags a segment handle, we move that segment to the cursor
   129→ * position. The dragged segment stays horizontal or vertical (whichever it
   130→ * was), and its perpendicular axis snaps to the cursor. The parallel axis
   131→ * also follows the cursor when possible (for interior segments with waypoints
   132→ * on both ends). Adjacent segments naturally stretch/shrink to stay connected.
   133→ */
   134→function rerouteWireForDrag(
   135→  fromPos: Vec2,
   136→  toPos: Vec2,
   137→  segIndex: number,
   138→  cursorGrid: Vec2,
   139→  existingWaypoints: Vec2[],
   140→): Vec2[] {
   141→  // Build waypoint copy (never touch fromPos/toPos)
   142→  const wps: Vec2[] = existingWaypoints.length > 0
   143→    ? existingWaypoints.map((w) => ({ ...w }))
   144→    : [
   145→        { x: (fromPos.x + toPos.x) / 2, y: fromPos.y },
   146→        { x: (fromPos.x + toPos.x) / 2, y: toPos.y },
   147→      ];
   148→
   149→  // Helper: get path point by index (0=fromPos, 1..n=waypoints, n+1=toPos)
   150→  const getPt = (i: number): Vec2 => {
   151→    if (i === 0) return fromPos;
   152→    if (i === wps.length + 1) return toPos;
   153→    return wps[i - 1];
   154→  };
   155→  // Helper: set a waypoint by path index (only waypoints are settable)
   156→  const setPt = (i: number, val: Vec2) => {
   157→    if (i > 0 && i <= wps.length) wps[i - 1] = val;
   158→  };
   159→
   160→  const a = getPt(segIndex);
   161→  const b = getPt(segIndex + 1);
   162→  const isHorizontal = Math.abs(b.y - a.y) < Math.abs(b.x - a.x);
   163→
   164→  if (isHorizontal) {
   165→    // Horizontal segment: perpendicular axis = Y. Cursor Y becomes the new Y.
   166→    // Parallel axis = X: shift segment along X toward cursor (only waypoints move).
   167→    const newY = cursorGrid.y;
   168→    const midX = (a.x + b.x) / 2;
   169→    const deltaX = cursorGrid.x - midX;
   170→    const aIdx = segIndex;
   171→    const bIdx = segIndex + 1;
   172→    const aIsFixed = (aIdx === 0);           // fromPos can't move
   173→    const bIsFixed = (bIdx === wps.length + 1); // toPos can't move
   174→    // Set Y on both endpoints (waypoints only; fixed terminals keep their Y)
   175→    const newA = { ...getPt(aIdx) };
   176→    const newB = { ...getPt(bIdx) };
   177→    if (!aIsFixed) newA.y = newY;
   178→    if (!bIsFixed) newB.y = newY;
   179→    // Shift X: both endpoints if both are waypoints; only one if the other is fixed
   180→    if (!aIsFixed && !bIsFixed) {
   181→      newA.x += deltaX;
   182→      newB.x += deltaX;
   183→    } else if (aIsFixed && !bIsFixed) {
   184→      newB.x += deltaX;
   185→    } else if (!aIsFixed && bIsFixed) {
   186→      newA.x += deltaX;
   187→    }
   188→    setPt(aIdx, newA);
   189→    setPt(bIdx, newB);
   190→  } else {
   191→    // Vertical segment: perpendicular axis = X. Cursor X becomes the new X.
   192→    // Parallel axis = Y: shift segment along Y toward cursor.
   193→    const newX = cursorGrid.x;
   194→    const midY = (a.y + b.y) / 2;
   195→    const deltaY = cursorGrid.y - midY;
   196→    const aIdx = segIndex;
   197→    const bIdx = segIndex + 1;
   198→    const aIsFixed = (aIdx === 0);
   199→    const bIsFixed = (bIdx === wps.length + 1);
   200→    const newA = { ...getPt(aIdx) };
   201→    const newB = { ...getPt(bIdx) };
   202→    if (!aIsFixed) newA.x = newX;
   203→    if (!bIsFixed) newB.x = newX;
   204→    if (!aIsFixed && !bIsFixed) {
   205→      newA.y += deltaY;
   206→      newB.y += deltaY;
   207→    } else if (aIsFixed && !bIsFixed) {
   208→      newB.y += deltaY;
   209→    } else if (!aIsFixed && bIsFixed) {
   210→      newA.y += deltaY;
   211→    }
   212→    setPt(aIdx, newA);
   213→    setPt(bIdx, newB);
   214→  }
   215→
   216→  // After moving waypoints, the path may have diagonal segments adjacent to
   217→  // fixed terminals. Re-orthogonalize by inserting elbow waypoints.
   218→  return orthogonalizePath(fromPos, toPos, wps);
   219→}
   220→
   221→/**
   222→ * Ensure a wire path is fully orthogonal (no diagonal segments).
   223→ * If moving a waypoint created a diagonal segment adjacent to a fixed terminal,
   224→ * insert an extra elbow waypoint to break it into two orthogonal segments.
   225→ *
   226→ * This is called after rerouteWireForDrag to clean up any diagonals.
   227→ */
   228→function orthogonalizePath(fromPos: Vec2, toPos: Vec2, wps: Vec2[]): Vec2[] {
   229→  const result: Vec2[] = [];
   230→  const full: Vec2[] = [fromPos, ...wps, toPos];
   231→  for (let i = 0; i < full.length - 1; i++) {
   232→    const a = full[i];
   233→    const b = full[i + 1];
   234→    result.push({ ...a });
   235→    // If segment is diagonal, insert an elbow (go horizontal first, then vertical)
   236→    if (Math.abs(a.x - b.x) > 0.01 && Math.abs(a.y - b.y) > 0.01) {
   237→      // Insert elbow at (b.x, a.y) — horizontal first, then vertical
   238→      result.push({ x: b.x, y: a.y });
   239→    }
   240→  }
   241→  result.push({ ...toPos });
   242→  // Convert back to waypoints (exclude fromPos and toPos)
   243→  return result.slice(1, -1);
   244→}

  1378→        // only increments by 1, we call it the right number of times.
  1379→        let diff = (targetRotation - comp.rotation + 4) % 4;
  1380→        for (let i = 0; i < diff; i++) {
  1381→          rotateComponent(comp.id);
  1382→        }
  1383→      }
  1384→      return;
  1385→    }
  1386→
  1387→    // wire segment dragging — free 2D drag in any direction
  1388→    if (wireDragRef.current) {
  1389→      const wd = wireDragRef.current;
  1390→      const wire = wires.find((w) => w.id === wd.wireId);
  1391→      if (!wire) return;
  1392→      const fromComp = components.find((c) => c.id === wire.from.componentId);
  1393→      const toComp = components.find((c) => c.id === wire.to.componentId);
  1394→      if (!fromComp || !toComp) return;
  1395→      const fromPlugin = getPlugin(fromComp.type);
  1396→      const toPlugin = getPlugin(toComp.type);
  1397→      if (!fromPlugin || !toPlugin) return;
  1398→      const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
  1399→      const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
  1400→      if (!fromT || !toT) return;
  1401→      const fromPos = getTerminalPos(fromComp, fromT);
  1402→      const toPos = getTerminalPos(toComp, toT);
  1403→      // Always start from the ORIGINAL waypoints captured at drag start.
  1404→      // This gives smooth, predictable behavior: the segment follows the cursor
  1405→      // absolutely, and the path is rebuilt fresh each frame from the original.
  1406→      const newWaypoints = rerouteWireForDrag(
  1407→        { ...fromPos },
  1408→        { ...toPos },
  1409→        wd.segIndex,
  1410→        g,
  1411→        wd.originalWaypoints,
  1412→      );
  1413→      setWireWaypoints(wd.wireId, newWaypoints);
  1414→      return;
  1415→    }
  1416→
  1417→    if (dragRef.current) {
  1418→      if (running) return;
  1419→      if (dragRef.current.isGroupDrag) {
  1420→        const delta = {
  1421→          x: g.x - (dragRef.current.lastGrid?.x ?? g.x),
  1422→          y: g.y - (dragRef.current.lastGrid?.y ?? g.y),
  1423→        };
  1424→        if (delta.x !== 0 || delta.y !== 0) {
  1425→          useEditor.getState().moveSelectedComponents(delta);
  1426→        }
  1427→        dragRef.current.lastGrid = { ...g };
  1428→        return;
  1429→      }
  1430→      const newPos = { x: g.x - dragRef.current.offset.x, y: g.y - dragRef.current.offset.y };
  1431→      moveComponent(dragRef.current.componentId, newPos);
  1432→      return;
  1433→    }
  1434→
  1435→    if (wireDraft) {
  1436→      updateWireCursor(g);
  1437→    }
  1438→
  1439→    // hover detection: priority: terminal > rotate handle > wire handle > wire > component
  1440→    const term = findTerminalAt(g.x, g.y);
  1441→    if (term) {
  1442→      setHover({ componentId: term.componentId, terminal: term, wireId: null, wireHandle: null, rotateHandle: null });
  1443→      return;
  1444→    }
  1445→    // rotate handle
  1446→    if (!running && selection.type === 'component') {
  1447→      const comp = components.find((c) => c.id === selection.id);
  1448→      if (comp) {
  1449→        const handlePos = getRotateHandlePos(comp);
  1450→        if (handlePos) {
  1451→          const dx = sx - handlePos.x;
  1452→          const dy = sy - handlePos.y;
  1453→          if (dx * dx + dy * dy < 144) { // 12px radius hit zone
  1454→            setHover({ componentId: comp.id, terminal: null, wireId: null, wireHandle: null, rotateHandle: comp.id });
  1455→            return;
  1456→          }
  1457→        }
  1458→      }
  1459→    }
  1460→    // wire handle
  1461→    if (!running) {
  1462→      const wh = findWireHandle(sx, sy);
  1463→      if (wh) {
  1464→        setHover({ componentId: null, terminal: null, wireId: wh.wireId, wireHandle: wh, rotateHandle: null });
  1465→        return;
  1466→      }
  1467→    }
  1468→    // wire
  1469→    const wireId = findWireAt(sx, sy);
  1470→    if (wireId) {
  1471→      setHover({ componentId: null, terminal: null, wireId, wireHandle: null, rotateHandle: null });
  1472→      return;
  1473→    }
  1474→    // component
  1475→    const comp = findComponentAt(g.x, g.y);
  1476→    setHover({ componentId: comp?.id ?? null, terminal: null, wireId: null, wireHandle: null, rotateHandle: null });
  1477→  };
  1478→
  1479→  const onMouseUp = (e: React.MouseEvent) => {
  1480→    if (e.button === 1 || e.button === 2) {
  1481→      panRef.current = null;
  1482→      return;
  1483→    }
  1484→    if (rotateDragRef.current) {
  1485→      // History was already pushed at drag start
  1486→      rotateDragRef.current = null;
  1487→      setRotateDrag(null);
  1488→      return;
  1489→    }
  1490→    if (wireDragRef.current) {
  1491→      // History was already pushed at drag start
  1492→      wireDragRef.current = null;
  1493→      return;
  1494→    }
  1495→    if (dragRef.current) {
  1496→      // History was already pushed at drag start
  1497→      dragRef.current = null;
  1498→    }
  1499→  };
  1500→
  1501→  const onWheel = (e: React.WheelEvent) => {
  1502→    const rect = canvasRef.current!.getBoundingClientRect();
  1503→    const sx = e.clientX - rect.left;
  1504→    const sy = e.clientY - rect.top;
  1505→    const factor = e.deltaY < 0 ? 1.1 : 1 / 1.1;
  1506→    const newZoom = Math.max(0.4, Math.min(4, zoom * factor));
  1507→    // Keep the grid point under the cursor stationary during zoom.
  1508→    // Grid point under cursor before zoom: (sx - pan.x) / (CELL_SIZE * zoom)
  1509→    // After zoom, we want: sx = gridX * CELL_SIZE * newZoom + newPan.x
  1510→    // => newPan.x = sx - gridX * CELL_SIZE * newZoom
  1511→    const gridX = (sx - pan.x) / (CELL_SIZE * zoom);
  1512→    const gridY = (sy - pan.y) / (CELL_SIZE * zoom);
  1513→    const newPanX = sx - gridX * CELL_SIZE * newZoom;
  1514→    const newPanY = sy - gridY * CELL_SIZE * newZoom;
  1515→    setZoom(newZoom);
  1516→    setPan({ x: newPanX, y: newPanY });
  1517→  };
  1518→
  1519→  const onDrop = (e: React.DragEvent) => {
  1520→    if (running) return;
  1521→    e.preventDefault();
  1522→    const type = e.dataTransfer.getData('application/x-circuit-type');
  1523→    if (!type) return;
  1524→    const rect = canvasRef.current!.getBoundingClientRect();
  1525→    const sx = e.clientX - rect.left;
  1526→    const sy = e.clientY - rect.top;
  1527→    const g = screenToGrid(sx, sy);
  1528→    addComponent(type, g);
  1529→  };
  1530→
  1531→  const onDragOver = (e: React.DragEvent) => {
  1532→    if (running) return;
  1533→    e.preventDefault();
  1534→    e.dataTransfer.dropEffect = 'copy';
  1535→  };
  1536→
  1537→  const onDoubleClick = (e: React.MouseEvent) => {
  1538→    if (running) return;
  1539→    const rect = canvasRef.current!.getBoundingClientRect();
  1540→    const sx = e.clientX - rect.left;
  1541→    const sy = e.clientY - rect.top;
  1542→    const g = screenToGrid(sx, sy);
  1543→    const comp = findComponentAt(g.x, g.y);
  1544→    if (comp) rotateComponent(comp.id);
  1545→  };
  1546→
  1547→  // cursor style based on hover state
  1548→  const getCursorStyle = (): string => {
  1549→    if (rotateDrag) return 'grabbing';
  1550→    if (running) {
  1551→      if (hover.componentId) {
  1552→        const comp = components.find((c) => c.id === hover.componentId);
  1553→        if (comp && TOGGLEABLE_TYPES.has(comp.type)) return 'pointer';
  1554→      }
  1555→      return 'default';
  1556→    }
  1557→    if (hover.terminal) return 'crosshair';
  1558→    if (hover.rotateHandle) return 'grab';
  1559→    if (hover.wireHandle) return 'move';
  1560→    if (hover.wireId) return 'pointer';
  1561→    if (hover.componentId) return 'move';
  1562→    return 'crosshair';
  1563→  };
  1564→
  1565→  // keyboard
  1566→  useEffect(() => {
  1567→    const onKey = (e: KeyboardEvent) => {
  1568→      const target = e.target as HTMLElement;
  1569→      if (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.tagName === 'SELECT') return;
  1570→      if (e.key === 'Delete' || e.key === 'Backspace') {
  1571→        if (running) return;
  1572→        const s = useEditor.getState();
  1573→        if (s.multiSelection.components.size > 0 || s.multiSelection.wires.size > 0) {
  1574→          s.deleteSelected();
  1575→        } else if (s.selection.type === 'component') {
  1576→          deleteComponent(s.selection.id!);
  1577→        } else if (s.selection.type === 'wire') {
  1578→          s.deleteWire(s.selection.id!);
  1579→        }
  1580→      } else if (e.key === 'r' || e.key === 'R') {
  1581→        if (running) return;
  1582→        const s = useEditor.getState().selection;
  1583→        if (s.type === 'component') rotateComponent(s.id!);
  1584→      } else if (e.key === 'x' || e.key === 'X') {
  1585→        // Mirror X (vertical flip)
  1586→        if (running) return;
  1587→        e.preventDefault();
  1588→        const s = useEditor.getState();
  1589→        if (s.selection.type === 'component' && s.selection.id) s.mirrorComponent(s.selection.id, 'x');
  1590→        else if (s.multiSelection.components.size > 0) s.mirrorSelected('x');
  1591→      } else if (e.key === 'y' || e.key === 'Y') {
  1592→        // Mirror Y (horizontal flip)
  1593→        if (running) return;
  1594→        e.preventDefault();
  1595→        const s = useEditor.getState();
  1596→        if (s.selection.type === 'component' && s.selection.id) s.mirrorComponent(s.selection.id, 'y');
  1597→        else if (s.multiSelection.components.size > 0) s.mirrorSelected('y');
  1598→      } else if (e.key === 'l' || e.key === 'L') {
  1599→        // Lock/unlock component
  1600→        if (running) return;
  1601→        e.preventDefault();
  1602→        const s = useEditor.getState();
  1603→        if (s.selection.type === 'component' && s.selection.id) s.toggleLock(s.selection.id);
  1604→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'f') {
  1605→        // Find / Replace — dispatch a custom event the Toolbar listens for
  1606→        if (running) return;
  1607→        e.preventDefault();
  1608→        window.dispatchEvent(new CustomEvent('circuitlab:open-find-replace'));
  1609→      } else if (e.key === 'n' || e.key === 'N') {
  1610→        // Add no-connect marker to hovered terminal
  1611→        if (running) return;
  1612→        e.preventDefault();
  1613→        if (hover.terminal) {
  1614→          const s = useEditor.getState();
  1615→          // toggle: remove if exists, add if not
  1616→          const exists = s.noConnects.find((nc) => nc.componentId === hover.terminal!.componentId && nc.terminalId === hover.terminal!.terminalId);
  1617→          if (exists) s.removeNoConnect(hover.terminal!.componentId, hover.terminal!.terminalId);
  1618→          else s.addNoConnect(hover.terminal!.componentId, hover.terminal!.terminalId);
  1619→        }
  1620→      } else if (e.key === 'Escape') {
  1621→        cancelWire();
  1622→        setSelection({ type: null, id: null });
  1623→        useEditor.getState().clearMultiSelection();
  1624→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'z' && !e.shiftKey) {
  1625→        if (running) return;
  1626→        e.preventDefault();
  1627→        useEditor.getState().undo();
  1628→      } else if ((e.ctrlKey || e.metaKey) && (e.key === 'y' || (e.key === 'z' && e.shiftKey))) {
  1629→        if (running) return;
  1630→        e.preventDefault();
  1631→        useEditor.getState().redo();
  1632→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'c') {
  1633→        if (running) return;
  1634→        e.preventDefault();
  1635→        useEditor.getState().copySelection();
  1636→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'v') {
  1637→        if (running) return;
  1638→        e.preventDefault();
  1639→        useEditor.getState().paste();
  1640→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'd') {
  1641→        if (running) return;
  1642→        e.preventDefault();
  1643→        useEditor.getState().duplicate();
  1644→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'a') {
  1645→        if (running) return;
  1646→        e.preventDefault();
  1647→        const s = useEditor.getState();
  1648→        s.setMultiSelection({
  1649→          components: new Set(s.components.map((c) => c.id)),
  1650→          wires: new Set(s.wires.map((w) => w.id)),
  1651→        });
  1652→      } else if (e.key === ' ') {
  1653→        e.preventDefault();
  1654→        const s = useEditor.getState();
  1655→        s.setRunning(!s.running);
  1656→      } else if (e.key === '\\' || e.key === '|') {
  1657→        e.preventDefault();
  1658→        setUse45Routing((v) => !v);
  1659→      }
  1660→    };
  1661→    window.addEventListener('keydown', onKey);
  1662→    return () => window.removeEventListener('keydown', onKey);
  1663→  }, [deleteComponent, rotateComponent, cancelWire, setSelection, running]);
  1664→
  1665→  return (
  1666→    <div
  1667→      ref={containerRef}
  1668→      className="relative h-full w-full overflow-hidden bg-slate-950"
  1669→      onContextMenu={(e) => e.preventDefault()}
  1670→    >
  1671→      <canvas
  1672→        ref={canvasRef}
  1673→        className="absolute inset-0"
  1674→        style={{ cursor: getCursorStyle() }}
  1675→        onMouseDown={onMouseDown}
  1676→        onMouseMove={onMouseMove}
  1677→        onMouseUp={onMouseUp}
  1678→        onMouseLeave={() => { dragRef.current = null; panRef.current = null; wireDragRef.current = null; rotateDragRef.current = null; setRotateDrag(null); }}
  1679→        onWheel={onWheel}
  1680→        onDrop={onDrop}
  1681→        onDragOver={onDragOver}
  1682→        onDoubleClick={onDoubleClick}
  1683→      />
  1684→      {/* status overlay */}
  1685→      <div className="pointer-events-none absolute bottom-2 left-2 rounded-md bg-slate-900/80 px-2 py-1 text-xs font-mono text-slate-400">
  1686→        ({cursor.x.toFixed(1)}, {cursor.y.toFixed(1)})  zoom: {zoom.toFixed(2)}x  {running ? '▶ running' : '⏸ paused'}
  1687→        {running && <span className="ml-2 text-amber-300">· click switches to toggle</span>}
  1688→      </div>
  1689→      <div className="pointer-events-none absolute bottom-2 right-2 rounded-md bg-slate-900/80 px-2 py-1 text-xs font-mono text-slate-400">
  1690→        {running ? 'Click switches to toggle • Space to pause' : 'Drag from left • Double-click/R to rotate • Del delete • Space play/pause'}
  1691→      </div>
  1692→      {/* zoom controls */}
  1693→      <div className="absolute right-2 top-2 flex flex-col gap-1">
  1694→        <button
  1695→          onClick={() => setZoom((z) => Math.min(4, z * 1.2))}
  1696→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700"
  1697→        >+</button>
  1698→        <button
  1699→          onClick={() => setZoom((z) => Math.max(0.4, z / 1.2))}
  1700→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700"
  1701→        >−</button>
  1702→        <button
  1703→          onClick={() => { setZoom(1); setPan({ x: 0, y: 0 }); }}
  1704→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700 text-xs"
  1705→        >⌂</button>
  1706→      </div>
  1707→    </div>
  1708→  );
  1709→}

  1711→/** Distance from point (px, py) to segment (ax,ay)-(bx,by) */
  1712→function pointToSegmentDist(px: number, py: number, ax: number, ay: number, bx: number, by: number): number {
  1713→  const dx = bx - ax;
  1714→  const dy = by - ay;
  1715→  const lenSq = dx * dx + dy * dy;
  1716→  if (lenSq === 0) return Math.hypot(px - ax, py - ay);
  1717→  let t = ((px - ax) * dx + (py - ay) * dy) / lenSq;
  1718→  t = Math.max(0, Math.min(1, t));
  1719→  const cx = ax + t * dx;
  1720→  const cy = ay + t * dy;
  1721→  return Math.hypot(px - cx, py - cy);
  1722→}
