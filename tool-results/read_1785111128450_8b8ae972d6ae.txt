    50→/** Get the orthogonal path points for a wire.
    51→ *  fromPos and toPos are in SCREEN coords. Waypoints are in GRID coords and
    52→ *  are converted to screen coords using the provided converter. */
    53→function getWirePath(
    54→  wire: Wire,
    55→  fromPos: Vec2,
    56→  toPos: Vec2,
    57→  gridToScreenFn: (gx: number, gy: number) => Vec2,
    58→): Vec2[] {
    59→  const points: Vec2[] = [fromPos];
    60→  if (wire.waypoints && wire.waypoints.length > 0) {
    61→    for (const wp of wire.waypoints) {
    62→      points.push(gridToScreenFn(wp.x, wp.y));
    63→    }
    64→  } else {
    65→    // default orthogonal routing: go to midpoint X, then to target
    66→    const midX = (fromPos.x + toPos.x) / 2;
    67→    points.push({ x: midX, y: fromPos.y });
    68→    points.push({ x: midX, y: toPos.y });
    69→  }
    70→  points.push(toPos);
    71→  return points;
    72→}
    73→
    74→/** Compute the midpoint of a segment for drag handle detection */
    75→function segmentMidpoint(a: Vec2, b: Vec2): Vec2 {
    76→  return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
    77→}

    79→/** Angle (in degrees, 0-360) from a center point to a target point */
    80→function angleFromCenter(cx: number, cy: number, x: number, y: number): number {
    81→  const angle = Math.atan2(y - cy, x - cx) * 180 / Math.PI;
    82→  return (angle + 360) % 360;
    83→}

    85→/**
    86→ * Re-route a wire so a dragged segment follows the cursor freely in 2D.
    87→ *
    88→ * The wire is ALWAYS kept orthogonal (only horizontal/vertical segments).
    89→ * Terminal positions (fromPos, toPos) are NEVER moved — they're fixed to
    90→ * component terminals. Only waypoints change.
    91→ *
    92→ * Path layout: [fromPos, wp1, wp2, ..., toPos]
    93→ *   segIndex 0 = fromPos→wp1, segIndex 1 = wp1→wp2, ..., last = wpN→toPos
    94→ *
    95→ * When the user drags a segment handle, we move that segment to the cursor
    96→ * position. The dragged segment stays horizontal or vertical (whichever it
    97→ * was), and its perpendicular axis snaps to the cursor. The parallel axis
    98→ * also follows the cursor when possible (for interior segments with waypoints
    99→ * on both ends). Adjacent segments naturally stretch/shrink to stay connected.
   100→ */
   101→function rerouteWireForDrag(
   102→  fromPos: Vec2,
   103→  toPos: Vec2,
   104→  segIndex: number,
   105→  cursorGrid: Vec2,
   106→  existingWaypoints: Vec2[],
   107→): Vec2[] {
   108→  // Build waypoint copy (never touch fromPos/toPos)
   109→  const wps: Vec2[] = existingWaypoints.length > 0
   110→    ? existingWaypoints.map((w) => ({ ...w }))
   111→    : [
   112→        { x: (fromPos.x + toPos.x) / 2, y: fromPos.y },
   113→        { x: (fromPos.x + toPos.x) / 2, y: toPos.y },
   114→      ];
   115→
   116→  // Helper: get path point by index (0=fromPos, 1..n=waypoints, n+1=toPos)
   117→  const getPt = (i: number): Vec2 => {
   118→    if (i === 0) return fromPos;
   119→    if (i === wps.length + 1) return toPos;
   120→    return wps[i - 1];
   121→  };
   122→  // Helper: set a waypoint by path index (only waypoints are settable)
   123→  const setPt = (i: number, val: Vec2) => {
   124→    if (i > 0 && i <= wps.length) wps[i - 1] = val;
   125→  };
   126→
   127→  const a = getPt(segIndex);
   128→  const b = getPt(segIndex + 1);
   129→  const isHorizontal = Math.abs(b.y - a.y) < Math.abs(b.x - a.x);
   130→
   131→  if (isHorizontal) {
   132→    // Horizontal segment: perpendicular axis = Y. Cursor Y becomes the new Y.
   133→    // Parallel axis = X: shift segment along X toward cursor (only waypoints move).
   134→    const newY = cursorGrid.y;
   135→    const midX = (a.x + b.x) / 2;
   136→    const deltaX = cursorGrid.x - midX;
   137→    const aIdx = segIndex;
   138→    const bIdx = segIndex + 1;
   139→    const aIsFixed = (aIdx === 0);           // fromPos can't move
   140→    const bIsFixed = (bIdx === wps.length + 1); // toPos can't move
   141→    // Set Y on both endpoints (waypoints only; fixed terminals keep their Y)
   142→    const newA = { ...getPt(aIdx) };
   143→    const newB = { ...getPt(bIdx) };
   144→    if (!aIsFixed) newA.y = newY;
   145→    if (!bIsFixed) newB.y = newY;
   146→    // Shift X: both endpoints if both are waypoints; only one if the other is fixed
   147→    if (!aIsFixed && !bIsFixed) {
   148→      newA.x += deltaX;
   149→      newB.x += deltaX;
   150→    } else if (aIsFixed && !bIsFixed) {
   151→      newB.x += deltaX;
   152→    } else if (!aIsFixed && bIsFixed) {
   153→      newA.x += deltaX;
   154→    }
   155→    setPt(aIdx, newA);
   156→    setPt(bIdx, newB);
   157→  } else {
   158→    // Vertical segment: perpendicular axis = X. Cursor X becomes the new X.
   159→    // Parallel axis = Y: shift segment along Y toward cursor.
   160→    const newX = cursorGrid.x;
   161→    const midY = (a.y + b.y) / 2;
   162→    const deltaY = cursorGrid.y - midY;
   163→    const aIdx = segIndex;
   164→    const bIdx = segIndex + 1;
   165→    const aIsFixed = (aIdx === 0);
   166→    const bIsFixed = (bIdx === wps.length + 1);
   167→    const newA = { ...getPt(aIdx) };
   168→    const newB = { ...getPt(bIdx) };
   169→    if (!aIsFixed) newA.x = newX;
   170→    if (!bIsFixed) newB.x = newX;
   171→    if (!aIsFixed && !bIsFixed) {
   172→      newA.y += deltaY;
   173→      newB.y += deltaY;
   174→    } else if (aIsFixed && !bIsFixed) {
   175→      newB.y += deltaY;
   176→    } else if (!aIsFixed && bIsFixed) {
   177→      newA.y += deltaY;
   178→    }
   179→    setPt(aIdx, newA);
   180→    setPt(bIdx, newB);
   181→  }
   182→
   183→  // After moving waypoints, the path may have diagonal segments adjacent to
   184→  // fixed terminals. Re-orthogonalize by inserting elbow waypoints.
   185→  return orthogonalizePath(fromPos, toPos, wps);
   186→}
   187→
   188→/**
   189→ * Ensure a wire path is fully orthogonal (no diagonal segments).
   190→ * If moving a waypoint created a diagonal segment adjacent to a fixed terminal,
   191→ * insert an extra elbow waypoint to break it into two orthogonal segments.
   192→ *
   193→ * This is called after rerouteWireForDrag to clean up any diagonals.
   194→ */
   195→function orthogonalizePath(fromPos: Vec2, toPos: Vec2, wps: Vec2[]): Vec2[] {
   196→  const result: Vec2[] = [];
   197→  const full: Vec2[] = [fromPos, ...wps, toPos];
   198→  for (let i = 0; i < full.length - 1; i++) {
   199→    const a = full[i];
   200→    const b = full[i + 1];
   201→    result.push({ ...a });
   202→    // If segment is diagonal, insert an elbow (go horizontal first, then vertical)
   203→    if (Math.abs(a.x - b.x) > 0.01 && Math.abs(a.y - b.y) > 0.01) {
   204→      // Insert elbow at (b.x, a.y) — horizontal first, then vertical
   205→      result.push({ x: b.x, y: a.y });
   206→    }
   207→  }
   208→  result.push({ ...toPos });
   209→  // Convert back to waypoints (exclude fromPos and toPos)
   210→  return result.slice(1, -1);
   211→}

   251→  const deleteComponent = useEditor((s) => s.deleteComponent);

   273→  // SINGLE unified animation+simulation loop.
   274→  // Both the simulation step and the flow-dot phase advance happen in the same
   275→  // requestAnimationFrame callback, so there's only ONE re-render per frame.
   276→  // This eliminates the blinking/lag caused by two competing RAF loops.
   277→  // The flow dot speed is linked to the simulation speed setting so they match.
   278→  const [animTick, setAnimTick] = useState(0);
   279→  useEffect(() => {
   280→    if (!running) return;
   281→    let raf = 0;
   282→    let lastSimTime = performance.now();
   283→    const SIM_INTERVAL = 16; // ms between simulation steps (~60Hz)
   284→    const loop = (now: number) => {

   288→      flowPhaseRef.current = (flowPhaseRef.current + 0.012 * Math.max(0.5, Math.min(4, simSpeed))) % 1;
   289→      // Run simulation step at fixed interval

   292→        lastSimTime = now;
   293→      }
   294→      // Single re-render per frame — covers both sim state and animation
   295→      setAnimTick((t) => (t + 1) % 1000000);
   296→      raf = requestAnimationFrame(loop);
   297→    };

   314→  const getTerminalPos = useCallback((comp: CircuitComponent, terminal: TerminalDef): Vec2 => {
   315→    const plugin = getPlugin(comp.type);
   316→    if (!plugin) return { x: 0, y: 0 };
   317→    const rotated = rotateTerminal(terminal, comp.rotation, plugin.boundingBox);
   318→    return {
   319→      x: comp.position.x + rotated.position.x,
   320→      y: comp.position.y + rotated.position.y,

   324→  const findTerminalAt = useCallback((gx: number, gy: number) => {
   325→    for (const comp of components) {
   326→      const plugin = getPlugin(comp.type);
   327→      if (!plugin) continue;
   328→      for (const t of plugin.terminals) {
   329→        const pos = getTerminalPos(comp, t);
   330→        const dx = pos.x - gx;
   331→        const dy = pos.y - gy;
   332→        if (dx * dx + dy * dy < 0.5 * 0.5) {
   333→          return { componentId: comp.id, terminalId: t.id, pos };
   334→        }
   335→      }
   336→    }
   337→    return null;
   338→  }, [components, getTerminalPos]);
   339→
   340→  const findComponentAt = useCallback((gx: number, gy: number): CircuitComponent | null => {
   341→    for (let i = components.length - 1; i >= 0; i--) {
   342→      const comp = components[i];
   343→      const plugin = getPlugin(comp.type);
   344→      if (!plugin) continue;
   345→      const bb = plugin.boundingBox;
   346→      const cx = bb.width / 2;
   347→      const cy = bb.height / 2;
   348→      const dx = gx - (comp.position.x + cx);
   349→      const dy = gy - (comp.position.y + cy);
   350→      let rx: number, ry: number;
   351→      switch (comp.rotation) {
   352→        case 0: rx = dx; ry = dy; break;
   353→        case 1: rx = -dy; ry = dx; break;
   354→        case 2: rx = -dx; ry = -dy; break;
   355→        case 3: rx = dy; ry = -dx; break;
   356→      }
   357→      if (Math.abs(rx) <= bb.width / 2 && Math.abs(ry) <= bb.height / 2) {
   358→        return comp;
   359→      }
   360→    }
   361→    return null;
   362→  }, [components]);
   363→
   364→  // find wire under cursor (hit-test against wire path)
   365→  const findWireAt = useCallback((sx: number, sy: number): string | null => {
   366→    for (const wire of wires) {
   367→      const fromComp = components.find((c) => c.id === wire.from.componentId);
   368→      const toComp = components.find((c) => c.id === wire.to.componentId);
   369→      if (!fromComp || !toComp) continue;
   370→      const fromPlugin = getPlugin(fromComp.type);
   371→      const toPlugin = getPlugin(toComp.type);
   372→      if (!fromPlugin || !toPlugin) continue;
   373→      const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
   374→      const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
   375→      if (!fromT || !toT) continue;
   376→      const fromPos = gridToScreen(getTerminalPos(fromComp, fromT).x, getTerminalPos(fromComp, fromT).y);
   377→      const toPos = gridToScreen(getTerminalPos(toComp, toT).x, getTerminalPos(toComp, toT).y);
   378→      const path = getWirePath(wire, fromPos, toPos, gridToScreen);
   379→      for (let i = 0; i < path.length - 1; i++) {
   380→        const a = path[i];
   381→        const b = path[i + 1];
   382→        const dist = pointToSegmentDist(sx, sy, a.x, a.y, b.x, b.y);
   383→        if (dist < 5) return wire.id;
   384→      }
   385→    }
   386→    return null;
   387→  }, [wires, components, gridToScreen, getTerminalPos]);
   388→
   389→  // find a draggable wire segment midpoint handle
   390→  const findWireHandle = useCallback((sx: number, sy: number): HoverState['wireHandle'] => {
   391→    for (const wire of wires) {
   392→      const fromComp = components.find((c) => c.id === wire.from.componentId);
   393→      const toComp = components.find((c) => c.id === wire.to.componentId);
   394→      if (!fromComp || !toComp) continue;
   395→      const fromPlugin = getPlugin(fromComp.type);
   396→      const toPlugin = getPlugin(toComp.type);
   397→      if (!fromPlugin || !toPlugin) continue;
   398→      const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
   399→      const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
   400→      if (!fromT || !toT) continue;
   401→      const fromPos = gridToScreen(getTerminalPos(fromComp, fromT).x, getTerminalPos(fromComp, fromT).y);
   402→      const toPos = gridToScreen(getTerminalPos(toComp, toT).x, getTerminalPos(toComp, toT).y);
   403→      const path = getWirePath(wire, fromPos, toPos, gridToScreen);
   404→      for (let i = 0; i < path.length - 1; i++) {
   405→        const a = path[i];
   406→        const b = path[i + 1];
   407→        const mid = segmentMidpoint(a, b);
   408→        const dx = sx - mid.x;
   409→        const dy = sy - mid.y;
   410→        if (dx * dx + dy * dy < 36) {
   411→          return { wireId: wire.id, segIndex: i, pos: mid };
   412→        }
   413→      }
   414→    }
   415→    return null;
   416→  }, [wires, components, gridToScreen, getTerminalPos]);
   417→
   418→  // compute rotation handle position for a component (in screen coords)
   419→  const getRotateHandlePos = useCallback((comp: CircuitComponent): Vec2 | null => {
   420→    const plugin = getPlugin(comp.type);
   421→    if (!plugin) return null;
   422→    const bb = plugin.boundingBox;
   423→    const centerGrid = { x: comp.position.x + bb.width / 2, y: comp.position.y + bb.height / 2 };
   424→    const centerScreen = gridToScreen(centerGrid.x, centerGrid.y);
   425→    return {
   426→      x: centerScreen.x,
   427→      y: centerScreen.y - (bb.height / 2 * CELL_SIZE * zoom) - 18,
   428→    };
   429→  }, [gridToScreen, zoom]);

   434→    if (!canvas) return;

   437→    canvas.height = size.height * dpr;
   438→    canvas.style.width = `${size.width}px`;
   439→    canvas.style.height = `${size.height}px`;
   440→    const ctx = canvas.getContext('2d');
   441→    if (!ctx) return;
   442→    ctx.save();
   443→    ctx.scale(dpr, dpr);
   444→    // bg
   445→    ctx.fillStyle = '#0f172a';
   446→    ctx.fillRect(0, 0, size.width, size.height);
   447→
   448→    // grid
   449→    if (showGrid) {
   450→      ctx.strokeStyle = '#1e293b';
   451→      ctx.lineWidth = 1;
   452→      const stepPx = CELL_SIZE * zoom;
   453→      const startX = pan.x % stepPx;
   454→      const startY = pan.y % stepPx;
   455→      ctx.beginPath();
   456→      for (let x = startX; x < size.width; x += stepPx) {
   457→        ctx.moveTo(x, 0);
   458→        ctx.lineTo(x, size.height);
   459→      }
   460→      for (let y = startY; y < size.height; y += stepPx) {
   461→        ctx.moveTo(0, y);
   462→        ctx.lineTo(size.width, y);
   463→      }
   464→      ctx.stroke();
   465→      ctx.fillStyle = '#475569';
   466→      ctx.fillRect(pan.x - 1, pan.y - 1, 3, 3);
   467→    }
   468→
   469→    // compute wire + component currents for flow animation — ONLY when running AND simContext exists.
   470→    const isAnimating = running && simContext != null;
   471→    const pluginsMap = new Map(plugins.map((p) => [p.type, p]));
   472→    const wireCurrents = isAnimating ? computeWireCurrents(
   473→      components, wires, pluginsMap, simContext!,
   474→    ) : new Map<string, number>();
   475→    const componentCurrents = isAnimating ? computeComponentCurrents(
   476→      components, wires, pluginsMap, simContext!,
   477→    ) : new Map<string, number>();
   478→
   479→    // Build a map of node -> number of attached wires (for junction coloring)
   480→    const nodeWireCount = new Map<string, number>();
   481→    for (const wire of wires) {
   482→      const key1 = `${wire.from.componentId}:${wire.from.terminalId}`;
   483→      const key2 = `${wire.to.componentId}:${wire.to.terminalId}`;
   484→      nodeWireCount.set(key1, (nodeWireCount.get(key1) ?? 0) + 1);
   485→      nodeWireCount.set(key2, (nodeWireCount.get(key2) ?? 0) + 1);
   486→    }
   487→
   488→    // Build node network: which terminals share the same electrical node.
   489→    // When a wire is selected or hovered, all terminals and wires on the same
   490→    // node should highlight, making it easy to see what connects to what.
   491→    const pluginsMapForNodes = new Map(plugins.map((p) => [p.type, p]));
   492→    const nodeMap = buildNodeMap(components, wires, pluginsMapForNodes);
   493→    // Map: terminalKey ("compId:termId") -> nodeId
   494→    // Map: nodeId -> Set of terminalKeys
   495→    // Map: nodeId -> Set of wireIds
   496→    const nodeToTerminals = new Map<number, Set<string>>();
   497→    const nodeToWires = new Map<number, Set<string>>();
   498→    for (const [termKey, nodeId] of nodeMap.terminalNode) {
   499→      if (!nodeToTerminals.has(nodeId)) nodeToTerminals.set(nodeId, new Set());
   500→      nodeToTerminals.get(nodeId)!.add(termKey);
   501→    }
   502→    for (const wire of wires) {
   503→      const fromKey = `${wire.from.componentId}:${wire.from.terminalId}`;
   504→      const toKey = `${wire.to.componentId}:${wire.to.terminalId}`;
   505→      const fromNode = nodeMap.terminalNode.get(fromKey) ?? -1;
   506→      const toNode = nodeMap.terminalNode.get(toKey) ?? -2;
   507→      // The wire connects fromNode and toNode (which should be the same after union-find)
   508→      const nodeId = fromNode; // they're the same node
   509→      if (!nodeToWires.has(nodeId)) nodeToWires.set(nodeId, new Set());
   510→      nodeToWires.get(nodeId)!.add(wire.id);
   511→      void toNode;
   512→    }
   513→
   514→    // Determine which node is "active" (selected or hovered wire's node)
   515→    let activeNodeId: number | null = null;
   516→    if (selection.type === 'wire' && selection.id) {
   517→      const selWire = wires.find((w) => w.id === selection.id);
   518→      if (selWire) {
   519→        const fromKey = `${selWire.from.componentId}:${selWire.from.terminalId}`;
   520→        activeNodeId = nodeMap.terminalNode.get(fromKey) ?? null;
   521→      }
   522→    } else if (hover.wireId) {
   523→      const hovWire = wires.find((w) => w.id === hover.wireId);
   524→      if (hovWire) {
   525→        const fromKey = `${hovWire.from.componentId}:${hovWire.from.terminalId}`;
   526→        activeNodeId = nodeMap.terminalNode.get(fromKey) ?? null;
   527→      }
   528→    }
   529→    // Set of terminal keys and wire IDs that should highlight
   530→    const activeTerminals = activeNodeId != null ? (nodeToTerminals.get(activeNodeId) ?? new Set<string>()) : new Set<string>();
   531→    const activeWires = activeNodeId != null ? (nodeToWires.get(activeNodeId) ?? new Set<string>()) : new Set<string>();
   532→
   533→    // ---- Draw wires FIRST (below components) ----
   534→    for (const wire of wires) {
   535→      const fromComp = components.find((c) => c.id === wire.from.componentId);
   536→      const toComp = components.find((c) => c.id === wire.to.componentId);
   537→      if (!fromComp || !toComp) continue;
   538→      const fromPlugin = getPlugin(fromComp.type);
   539→      const toPlugin = getPlugin(toComp.type);
   540→      if (!fromPlugin || !toPlugin) continue;
   541→      const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
   542→      const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
   543→      if (!fromT || !toT) continue;
   544→      const fromPos = gridToScreen(getTerminalPos(fromComp, fromT).x, getTerminalPos(fromComp, fromT).y);
   545→      const toPos = gridToScreen(getTerminalPos(toComp, toT).x, getTerminalPos(toComp, toT).y);
   546→      const path = getWirePath(wire, fromPos, toPos, gridToScreen);
   547→      const isSelected = selection.type === 'wire' && selection.id === wire.id;
   548→      const isHover = hover.wireId === wire.id;
   549→      const isOnActiveNode = activeWires.has(wire.id);
   550→
   551→      // wire — highlight all wires on the same electrical node
   552→      ctx.strokeStyle = isSelected ? '#fbbf24' : (isHover ? '#fde047' : (isOnActiveNode ? '#cbd5e1' : '#94a3b8'));
   553→      ctx.lineWidth = isSelected ? 3.5 : (isHover ? 3 : (isOnActiveNode ? 2.5 : 2));
   554→      ctx.lineCap = 'round';
   555→      ctx.lineJoin = 'round';
   556→      ctx.beginPath();
   557→      ctx.moveTo(path[0].x, path[0].y);
   558→      for (let i = 1; i < path.length; i++) ctx.lineTo(path[i].x, path[i].y);
   559→      ctx.stroke();
   560→
   561→      // draw wire segment midpoint handles (only when not running, for editing)
   562→      if (!running) {
   563→        for (let i = 0; i < path.length - 1; i++) {
   564→          const a = path[i];
   565→          const b = path[i + 1];
   566→          const mid = segmentMidpoint(a, b);
   567→          const isHandleHot = hover.wireHandle?.wireId === wire.id && hover.wireHandle?.segIndex === i;
   568→          ctx.beginPath();
   569→          ctx.arc(mid.x, mid.y, isHandleHot ? 5 : 3, 0, Math.PI * 2);
   570→          ctx.fillStyle = isHandleHot ? '#fbbf24' : '#475569';
   571→          ctx.fill();
   572→          ctx.strokeStyle = '#0f172a';
   573→          ctx.lineWidth = 1;
   574→          ctx.stroke();
   575→        }
   576→      }
   577→
   578→      // draw animated current flow dots — ONLY when running AND simContext is active.
   579→      // When paused or editing, absolutely no dots are drawn.
   580→      if (isAnimating) {
   581→        const current = wireCurrents.get(wire.id) ?? 0;
   582→        const absCurrent = Math.abs(current);
   583→        if (absCurrent > 1e-9) {
   584→          const dir = current >= 0 ? 1 : -1;
   585→          const speed = Math.min(1, Math.max(0.1, Math.log10(absCurrent * 1000 + 1) / 3));
   586→          const dotSpacing = 24;
   587→          let totalLen = 0;
   588→          const segLens: number[] = [];
   589→          for (let i = 0; i < path.length - 1; i++) {
   590→            const a = path[i];
   591→            const b = path[i + 1];
   592→            const len = Math.hypot(b.x - a.x, b.y - a.y);
   593→            segLens.push(len);
   594→            totalLen += len;
   595→          }
   596→          if (totalLen > 0) {
   597→            const numDots = Math.max(2, Math.floor(totalLen / dotSpacing));
   598→            const phase = flowPhaseRef.current * speed;
   599→            // Same bright yellow as component dots, with glow
   600→            ctx.fillStyle = '#fde047';
   601→            ctx.shadowColor = '#fde047';
   602→            ctx.shadowBlur = 6;
   603→            for (let n = 0; n < numDots; n++) {
   604→              let distAlong = (n / numDots + phase * dir) * totalLen;
   605→              while (distAlong < 0) distAlong += totalLen;
   606→              while (distAlong >= totalLen) distAlong -= totalLen;
   607→              let acc = 0;
   608→              for (let i = 0; i < segLens.length; i++) {
   609→                if (acc + segLens[i] >= distAlong) {
   610→                  const t = (distAlong - acc) / segLens[i];
   611→                  const a = path[i];
   612→                  const b = path[i + 1];
   613→                  const x = a.x + (b.x - a.x) * t;
   614→                  const y = a.y + (b.y - a.y) * t;
   615→                  ctx.beginPath();
   616→                  ctx.arc(x, y, 3, 0, Math.PI * 2);
   617→                  ctx.fill();
   618→                  break;
   619→                }
   620→                acc += segLens[i];
   621→              }
   622→            }
   623→            ctx.shadowBlur = 0; // reset glow after dots
   624→          }
   625→        }
   626→      }
   627→    }
   628→
   629→    // draw wire draft
   630→    if (wireDraft) {
   631→      const fromComp = components.find((c) => c.id === wireDraft.from.componentId);
   632→      if (fromComp) {
   633→        const plugin = getPlugin(fromComp.type);
   634→        if (plugin) {
   635→          const t = plugin.terminals.find((tt) => tt.id === wireDraft.from.terminalId);
   636→          if (t) {
   637→            const fromPos = gridToScreen(getTerminalPos(fromComp, t).x, getTerminalPos(fromComp, t).y);
   638→            const toPos = gridToScreen(wireDraft.cursor.x, wireDraft.cursor.y);
   639→            ctx.strokeStyle = '#fbbf24';
   640→            ctx.lineWidth = 2;
   641→            ctx.setLineDash([4, 4]);
   642→            ctx.beginPath();
   643→            ctx.moveTo(fromPos.x, fromPos.y);
   644→            ctx.lineTo(toPos.x, toPos.y);
   645→            ctx.stroke();
   646→            ctx.setLineDash([]);
   647→          }
   648→        }
   649→      }
   650→    }
   651→
   652→    // ---- Draw components ON TOP of wires ----
   653→    for (const comp of components) {
   654→      const plugin = getPlugin(comp.type);
   655→      if (!plugin) continue;
   656→      const isSelected = selection.type === 'component' && selection.id === comp.id;
   657→      const isHover = hover.componentId === comp.id;
   658→      const origin = gridToScreen(comp.position.x, comp.position.y);
   659→      ctx.save();
   660→      ctx.translate(origin.x, origin.y);
   661→      ctx.scale(zoom, zoom);
   662→      // Rotate around the component's CENTER (not top-left corner).
   663→      // Translate to center, rotate, translate back.
   664→      const bbCx = plugin.boundingBox.width / 2;
   665→      const bbCy = plugin.boundingBox.height / 2;
   666→      ctx.translate(bbCx * CELL_SIZE, bbCy * CELL_SIZE);
   667→      ctx.rotate((comp.rotation * Math.PI) / 2);
   668→      ctx.translate(-bbCx * CELL_SIZE, -bbCy * CELL_SIZE);
   669→      // selection halo
   670→      if (isSelected || isHover) {
   671→        ctx.save();
   672→        ctx.fillStyle = isSelected ? 'rgba(251, 191, 36, 0.18)' : 'rgba(148, 163, 184, 0.12)';
   673→        ctx.strokeStyle = isSelected ? '#fbbf24' : '#64748b';
   674→        ctx.lineWidth = 1.5;
   675→        ctx.beginPath();
   676→        ctx.rect(-2, -2, plugin.boundingBox.width * CELL_SIZE + 4, plugin.boundingBox.height * CELL_SIZE + 4);
   677→        ctx.fill();
   678→        ctx.stroke();
   679→        ctx.restore();
   680→      }
   681→      ctx.strokeStyle = '#e2e8f0';
   682→      ctx.fillStyle = '#e2e8f0';
   683→      ctx.lineWidth = 1.5;
   684→      // Create a temporary instance with __current attached so render() can
   685→      // use it for brightness/volume effects. We don't mutate the store state.
   686→      const renderInstance = isAnimating
   687→        ? { ...comp, simState: { ...(comp.simState ?? {}), __current: componentCurrents.get(comp.id) ?? 0 } }
   688→        : comp;
   689→      try {
   690→        plugin.render(ctx, comp.parameters, CELL_SIZE, simContext ?? undefined, renderInstance);
   691→      } catch (e) {
   692→        console.error(`render error in ${comp.type}:`, e);
   693→      }
   694→      ctx.restore();
   695→
   696→      // draw terminals (in screen coords) — color by connected state and active node
   697→      for (const t of plugin.terminals) {
   698→        const tpos = getTerminalPos(comp, t);
   699→        const sp = gridToScreen(tpos.x, tpos.y);
   700→        const isHot = hover.terminal?.componentId === comp.id && hover.terminal?.terminalId === t.id;
   701→        const connKey = `${comp.id}:${t.id}`;
   702→        const isConnected = (nodeWireCount.get(connKey) ?? 0) > 0;
   703→        const isOnActiveNode = activeTerminals.has(connKey);
   704→        // Junctions are "live" (green glow) only while running; otherwise show connected state plainly.
   705→        const isLive = isConnected && running && simContext != null;
   706→        let color = '#475569'; // unconnected: dark gray
   707→        let radius = 3;
   708→        let glow = 0;
   709→        if (isHot) { color = '#fbbf24'; radius = 5; }
   710→        else if (isOnActiveNode) { color = '#fde047'; radius = 5; glow = 6; } // active node: bright yellow
   711→        else if (isLive) { color = '#22c55e'; radius = 4; glow = 8; }
   712→        else if (isConnected) { color = '#cbd5e1'; }
   713→        ctx.beginPath();
   714→        ctx.arc(sp.x, sp.y, radius, 0, Math.PI * 2);
   715→        ctx.fillStyle = color;
   716→        ctx.fill();
   717→        if (glow > 0) {
   718→          ctx.shadowColor = color;
   719→          ctx.shadowBlur = glow;
   720→          ctx.fill();
   721→          ctx.shadowBlur = 0;
   722→        }
   723→        ctx.strokeStyle = '#0f172a';
   724→        ctx.lineWidth = 1;
   725→        ctx.stroke();
   726→      }
   727→
   728→      // draw animated current flow dots THROUGH the component body
   729→      if (isAnimating && plugin.getFlowPath) {
   730→        const current = componentCurrents.get(comp.id) ?? 0;
   731→        const absCurrent = Math.abs(current);
   732→        if (absCurrent > 1e-9) {
   733→          const dir = current >= 0 ? 1 : -1;
   734→          // Speed: proportional to current magnitude.
   735→          // Physics: I = V/R (Ohm's law). Higher current = faster flow.
   736→          // For inductor: I(t) = V/R·(1-e^(-Rt/L)) — starts at 0, ramps up.
   737→          // For capacitor: I = C·dV/dt — strong when charging, 0 when fully charged.
   738→          // The actual current value already reflects these physics because we
   739→          // compute it from the solved node voltages and component state.
   740→          // We map current to a visual speed: log-scaled so wide current ranges
   741→          // are visible. 1mA → ~0.33, 10mA → ~0.67, 100mA → ~1.0
   742→          const speed = Math.min(1.5, Math.max(0.05, Math.log10(absCurrent * 1000 + 1) / 3));
   743→          // Get the flow path in grid coords (relative to component origin, pre-rotation)
   744→          const flowGridPath = plugin.getFlowPath(comp.parameters, simContext ?? undefined, comp);
   745→          if (flowGridPath && flowGridPath.length >= 2) {
   746→            // Transform each point: apply rotation, then translate to component position, then to screen
   747→            const flowScreenPath: Vec2[] = flowGridPath.map((gp) => {
   748→              // Apply rotation (0,1,2,3 = 0°,90°,180°,270°) around bounding box center
   749→              const bb = plugin.boundingBox;
   750→              const cx = bb.width / 2;
   751→              const cy = bb.height / 2;
   752→              const dx = gp.x - cx;
   753→              const dy = gp.y - cy;
   754→              let rx: number, ry: number;
   755→              switch (comp.rotation) {
   756→                case 0: rx = dx; ry = dy; break;
   757→                case 1: rx = -dy; ry = dx; break;
   758→                case 2: rx = -dx; ry = -dy; break;
   759→                case 3: rx = dy; ry = -dx; break;
   760→              }
   761→              const gridX = comp.position.x + cx + rx;
   762→              const gridY = comp.position.y + cy + ry;
   763→              return gridToScreen(gridX, gridY);
   764→            });
   765→            // Compute total path length
   766→            let totalLen = 0;
   767→            const segLens: number[] = [];
   768→            for (let i = 0; i < flowScreenPath.length - 1; i++) {
   769→              const a = flowScreenPath[i];
   770→              const b = flowScreenPath[i + 1];
   771→              const len = Math.hypot(b.x - a.x, b.y - a.y);
   772→              segLens.push(len);
   773→              totalLen += len;
   774→            }
   775→            if (totalLen > 0) {
   776→              const dotSpacing = 18;
   777→              const numDots = Math.max(2, Math.floor(totalLen / dotSpacing));
   778→              const phase = flowPhaseRef.current * speed;
   779→              ctx.fillStyle = '#fde047';
   780→              ctx.shadowColor = '#fde047';
   781→              ctx.shadowBlur = 6;
   782→              for (let n = 0; n < numDots; n++) {
   783→                let distAlong = (n / numDots + phase * dir) * totalLen;
   784→                while (distAlong < 0) distAlong += totalLen;
   785→                while (distAlong >= totalLen) distAlong -= totalLen;
   786→                let acc = 0;
   787→                for (let i = 0; i < segLens.length; i++) {
   788→                  if (acc + segLens[i] >= distAlong) {
   789→                    const t = (distAlong - acc) / segLens[i];
   790→                    const a = flowScreenPath[i];
   791→                    const b = flowScreenPath[i + 1];
   792→                    const x = a.x + (b.x - a.x) * t;
   793→                    const y = a.y + (b.y - a.y) * t;
   794→                    ctx.beginPath();
   795→                    ctx.arc(x, y, 3.5, 0, Math.PI * 2);
   796→                    ctx.fill();
   797→                    break;
   798→                  }
   799→                  acc += segLens[i];
   800→                }
   801→              }
   802→              ctx.shadowBlur = 0;
   803→            }
   804→          }
   805→        }
   806→      }
   807→
   808→      // draw rotation handle on selected component (only when not running)
   809→      if (isSelected && !running) {
   810→        const bb = plugin.boundingBox;
   811→        const centerGrid = { x: comp.position.x + bb.width / 2, y: comp.position.y + bb.height / 2 };
   812→        const centerScreen = gridToScreen(centerGrid.x, centerGrid.y);
   813→        const handlePos = getRotateHandlePos(comp);
   814→        if (handlePos) {
   815→          const isHandleHot = hover.rotateHandle === comp.id || rotateDragRef.current?.componentId === comp.id;
   816→          // line from box to handle
   817→          ctx.strokeStyle = '#fbbf24';
   818→          ctx.lineWidth = 1;
   819→          ctx.setLineDash([2, 2]);
   820→          ctx.beginPath();
   821→          ctx.moveTo(centerScreen.x, centerScreen.y - (bb.height / 2 * CELL_SIZE * zoom) - 2);
   822→          ctx.lineTo(handlePos.x, handlePos.y + 9);
   823→          ctx.stroke();
   824→          ctx.setLineDash([]);
   825→          // handle circle with rotate icon
   826→          ctx.beginPath();
   827→          ctx.arc(handlePos.x, handlePos.y, isHandleHot ? 10 : 9, 0, Math.PI * 2);
   828→          ctx.fillStyle = isHandleHot ? '#fbbf24' : '#1e293b';
   829→          ctx.fill();
   830→          ctx.strokeStyle = '#fbbf24';
   831→          ctx.lineWidth = 1.5;
   832→          ctx.stroke();
   833→          // draw a small rotate arrow inside
   834→          ctx.strokeStyle = isHandleHot ? '#0f172a' : '#fbbf24';
   835→          ctx.lineWidth = 1.4;
   836→          ctx.beginPath();
   837→          ctx.arc(handlePos.x, handlePos.y, 4, -Math.PI * 0.2, Math.PI * 1.1);
   838→          ctx.stroke();
   839→          // arrowhead
   840→          ctx.beginPath();
   841→          ctx.moveTo(handlePos.x + 4.5, handlePos.y - 1);
   842→          ctx.lineTo(handlePos.x + 2.8, handlePos.y - 4.5);
   843→          ctx.lineTo(handlePos.x + 5.5, handlePos.y - 4.5);
   844→          ctx.closePath();
   845→          ctx.fillStyle = isHandleHot ? '#0f172a' : '#fbbf24';
   846→          ctx.fill();
   847→
   848→          // While dragging rotation, show a visual indicator of the target snap angle
   849→          if (rotateDragRef.current?.componentId === comp.id) {
   850→            const cursorScreen = gridToScreen(cursor.x, cursor.y);
   851→            const curAngle = angleFromCenter(centerScreen.x, centerScreen.y, cursorScreen.x, cursorScreen.y);
   852→            // Snap to nearest 90°: 0/90/180/270. We map cursor angle to rotation increment.
   853→            // Up = 270°, Right = 0°, Down = 90°, Left = 180° (screen y is down)
   854→            // We snap based on which quadrant the cursor is in relative to center.
   855→            // Show the snap direction as a thick arrow from center.
   856→            let snapRot: 0 | 1 | 2 | 3 = 0;
   857→            // Convert cursor angle to a rotation increment (0=right, 1=down, 2=left, 3=up)
   858→            // We want: cursor up (angle ~270) => rotation 3 (or whichever makes component point up)
   859→            // Use the angle to determine nearest of 4 directions
   860→            const normalized = (curAngle + 45) % 360;
   861→            if (normalized < 90) snapRot = 0;
   862→            else if (normalized < 180) snapRot = 1;
   863→            else if (normalized < 270) snapRot = 2;
   864→            else snapRot = 3;
   865→            void snapRot;
   866→            // Draw a dashed line from center to cursor
   867→            ctx.strokeStyle = 'rgba(251, 191, 36, 0.5)';
   868→            ctx.lineWidth = 1;
   869→            ctx.setLineDash([4, 4]);
   870→            ctx.beginPath();
   871→            ctx.moveTo(centerScreen.x, centerScreen.y);
   872→            ctx.lineTo(cursorScreen.x, cursorScreen.y);
   873→            ctx.stroke();
   874→            ctx.setLineDash([]);
   875→            // Draw the snap angle text
   876→            ctx.fillStyle = '#fbbf24';
   877→            ctx.font = 'bold 11px ui-monospace, monospace';
   878→            ctx.textAlign = 'center';
   879→            ctx.textBaseline = 'middle';
   880→            const snapDeg = (snapRot * 90) % 360;
   881→            ctx.fillText(`${snapDeg}°`, cursorScreen.x, cursorScreen.y - 14);
   882→          }
   883→        }
   884→      }
   885→
   886→      // "click to toggle" hint for switches during simulation
   887→      if (running && TOGGLEABLE_TYPES.has(comp.type) && (isHover || isSelected)) {
   888→        const bb = plugin.boundingBox;
   889→        const centerGrid = { x: comp.position.x + bb.width / 2, y: comp.position.y + bb.height / 2 };
   890→        const centerScreen = gridToScreen(centerGrid.x, centerGrid.y);
   891→        const label = comp.type === 'switch'
   892→          ? (comp.parameters.closed ? 'OPEN' : 'CLOSE')
   893→          : (comp.parameters.pressed ? 'RELEASE' : 'PRESS');
   894→        ctx.fillStyle = '#fbbf24';
   895→        ctx.font = 'bold 10px ui-monospace, monospace';
   896→        ctx.textAlign = 'center';
   897→        ctx.textBaseline = 'middle';
   898→        const tw = ctx.measureText(label).width + 10;
   899→        ctx.fillStyle = 'rgba(15, 23, 42, 0.9)';
   900→        ctx.fillRect(centerScreen.x - tw / 2, centerScreen.y + (bb.height / 2 * CELL_SIZE * zoom) + 4, tw, 16);
   901→        ctx.strokeStyle = '#fbbf24';
   902→        ctx.lineWidth = 1;
   903→        ctx.strokeRect(centerScreen.x - tw / 2, centerScreen.y + (bb.height / 2 * CELL_SIZE * zoom) + 4, tw, 16);
   904→        ctx.fillStyle = '#fbbf24';
   905→        ctx.fillText(label, centerScreen.x, centerScreen.y + (bb.height / 2 * CELL_SIZE * zoom) + 12);
   906→      }
   907→    }
   908→
   909→    // draw cursor crosshair (when no drag)
   910→    if (!dragRef.current && !panRef.current && !wireDragRef.current && !rotateDragRef.current) {
   911→      const sp = gridToScreen(cursor.x, cursor.y);
   912→      ctx.strokeStyle = 'rgba(251, 191, 36, 0.4)';
   913→      ctx.lineWidth = 1;
   914→      ctx.beginPath();
   915→      ctx.moveTo(sp.x - 6, sp.y);
   916→      ctx.lineTo(sp.x + 6, sp.y);
   917→      ctx.moveTo(sp.x, sp.y - 6);
   918→      ctx.lineTo(sp.x, sp.y + 6);
   919→      ctx.stroke();
   920→    }
   921→
   922→    // ---- Wire overlay: redraw wires ON TOP of components so they're always visible ----
   923→    // This prevents components from hiding wires. We draw a subtle dark background
   924→    // under each wire for contrast, then the wire color on top.
   925→    for (const wire of wires) {
   926→      const fromComp = components.find((c) => c.id === wire.from.componentId);
   927→      const toComp = components.find((c) => c.id === wire.to.componentId);
   928→      if (!fromComp || !toComp) continue;
   929→      const fromPlugin = getPlugin(fromComp.type);
   930→      const toPlugin = getPlugin(toComp.type);
   931→      if (!fromPlugin || !toPlugin) continue;
   932→      const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
   933→      const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
   934→      if (!fromT || !toT) continue;
   935→      const fromPos = gridToScreen(getTerminalPos(fromComp, fromT).x, getTerminalPos(fromComp, fromT).y);
   936→      const toPos = gridToScreen(getTerminalPos(toComp, toT).x, getTerminalPos(toComp, toT).y);
   937→      const path = getWirePath(wire, fromPos, toPos, gridToScreen);
   938→      const isSelected = selection.type === 'wire' && selection.id === wire.id;
   939→      const isHover = hover.wireId === wire.id;
   940→      const isOnActiveNode = activeWires.has(wire.id);
   941→
   942→      // Only redraw wires that are selected, hovered, or on the active node
   943→      // (to avoid overdrawing every wire on top of every component)
   944→      if (isSelected || isHover || isOnActiveNode) {
   945→        // Draw with glow for highlighted wires
   946→        ctx.strokeStyle = isSelected ? '#fbbf24' : (isHover ? '#fde047' : '#cbd5e1');
   947→        ctx.lineWidth = isSelected ? 3.5 : (isHover ? 3 : 2.5);
   948→        ctx.lineCap = 'round';
   949→        ctx.lineJoin = 'round';
   950→        if (isOnActiveNode && !isSelected) {
   951→          ctx.shadowColor = '#fde047';
   952→          ctx.shadowBlur = 4;
   953→        }
   954→        ctx.beginPath();
   955→        ctx.moveTo(path[0].x, path[0].y);
   956→        for (let i = 1; i < path.length; i++) ctx.lineTo(path[i].x, path[i].y);
   957→        ctx.stroke();
   958→        ctx.shadowBlur = 0;
   959→      }
   960→    }
   961→
   962→    ctx.restore();
   963→  }, [size, pan, zoom, components, wires, selection, hover, cursor, simContext, showGrid, wireDraft, running, gridToScreen, getTerminalPos, plugins, animTick, getRotateHandlePos]);
   964→
   965→  // ----- Mouse handlers -----
   966→  const onMouseDown = (e: React.MouseEvent) => {
   967→    const rect = canvasRef.current!.getBoundingClientRect();
   968→    const sx = e.clientX - rect.left;
   969→    const sy = e.clientY - rect.top;
   970→    const g = screenToGrid(sx, sy);
   971→
   972→    // middle or right button: pan
   973→    if (e.button === 1 || e.button === 2) {
   974→      panRef.current = { start: { x: sx, y: sy }, origin: { ...pan } };
   975→      return;
   976→    }
   977→
   978→    // If simulation is running, check for toggleable components FIRST
   979→    // Use getState() for the latest running state
   980→    const isRunningNow = useEditor.getState().running;
   981→    if (isRunningNow) {
   982→      const term = findTerminalAt(g.x, g.y);
   983→      if (term) return; // no wire drawing during simulation
   984→      const comp = findComponentAt(g.x, g.y);
   985→      if (comp && TOGGLEABLE_TYPES.has(comp.type)) {
   986→        toggleSwitch(comp.id);
   987→        return;
   988→      }
   989→      if (comp) {
   990→        setSelection({ type: 'component', id: comp.id });
   991→        return;
   992→      }
   993→      setSelection({ type: null, id: null });
   994→      return;
   995→    }
   996→
   997→    // check rotation handle on selected component — start a rotate drag
   998→    // Use getState() for the latest selection (the `selection` from the hook may be stale)
   999→    const currentSelection = useEditor.getState().selection;
  1000→    if (!isRunningNow && currentSelection.type === 'component') {
  1001→      const comp = components.find((c) => c.id === currentSelection.id);
  1002→      if (comp) {
  1003→        const plugin = getPlugin(comp.type);
  1004→        if (plugin) {
  1005→          const handlePos = getRotateHandlePos(comp);
  1006→          if (handlePos) {
  1007→            const dx = sx - handlePos.x;
  1008→            const dy = sy - handlePos.y;
  1009→            const distSq = dx * dx + dy * dy;
  1010→            if (distSq < 400) { // 20px radius hit zone
  1011→              const bb = plugin.boundingBox;
  1012→              const centerGrid = { x: comp.position.x + bb.width / 2, y: comp.position.y + bb.height / 2 };
  1013→              const centerScreen = gridToScreen(centerGrid.x, centerGrid.y);
  1014→              const newDrag: RotateDragState = {
  1015→                componentId: comp.id,
  1016→                center: centerScreen,
  1017→                startAngle: angleFromCenter(centerScreen.x, centerScreen.y, sx, sy),
  1018→                startRotation: comp.rotation,
  1019→              };
  1020→              rotateDragRef.current = newDrag;
  1021→              setRotateDrag(newDrag);
  1022→              return;
  1023→            }
  1024→          }
  1025→        }
  1026→      }
  1027→    }
  1028→
  1029→    // check wire segment handle (for dragging)
  1030→    if (hover.wireHandle) {
  1031→      const wire = wires.find((w) => w.id === hover.wireHandle.wireId);
  1032→      if (wire) {
  1033→        const fromComp = components.find((c) => c.id === wire.from.componentId);
  1034→        const toComp = components.find((c) => c.id === wire.to.componentId);
  1035→        if (fromComp && toComp) {
  1036→          const fromPlugin = getPlugin(fromComp.type);
  1037→          const toPlugin = getPlugin(toComp.type);
  1038→          if (fromPlugin && toPlugin) {
  1039→            const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
  1040→            const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
  1041→            if (fromT && toT) {
  1042→              const fromPos = getTerminalPos(fromComp, fromT);
  1043→              const toPos = getTerminalPos(toComp, toT);
  1044→              const originalWaypoints = wire.waypoints && wire.waypoints.length > 0
  1045→                ? wire.waypoints.map((w) => ({ ...w }))
  1046→                : [
  1047→                    { x: (fromPos.x + toPos.x) / 2, y: fromPos.y },
  1048→                    { x: (fromPos.x + toPos.x) / 2, y: toPos.y },
  1049→                  ];
  1050→              wireDragRef.current = {
  1051→                wireId: hover.wireHandle.wireId,
  1052→                segIndex: hover.wireHandle.segIndex,
  1053→                startGrid: g,
  1054→                originalWaypoints,
  1055→              };
  1056→              return;
  1057→            }
  1058→          }
  1059→        }
  1060→      }
  1061→    }
  1062→
  1063→    // check terminal first
  1064→    const term = findTerminalAt(g.x, g.y);
  1065→    if (term) {
  1066→      if (wireDraft) {
  1067→        completeWire({ componentId: term.componentId, terminalId: term.terminalId });
  1068→      } else {
  1069→        startWire({ componentId: term.componentId, terminalId: term.terminalId }, g);
  1070→      }
  1071→      return;
  1072→    }
  1073→
  1074→    // check component
  1075→    const comp = findComponentAt(g.x, g.y);
  1076→    if (comp) {
  1077→      setSelection({ type: 'component', id: comp.id });
  1078→      dragRef.current = {
  1079→        componentId: comp.id,
  1080→        offset: { x: g.x - comp.position.x, y: g.y - comp.position.y },
  1081→      };
  1082→      return;
  1083→    }
  1084→
  1085→    // check wire (for selection)
  1086→    const wireId = findWireAt(sx, sy);
  1087→    if (wireId) {
  1088→      setSelection({ type: 'wire', id: wireId });
  1089→      return;
  1090→    }
  1091→
  1092→    // empty space: clear selection, cancel wire
  1093→    setSelection({ type: null, id: null });
  1094→    if (wireDraft) cancelWire();
  1095→  };
  1096→
  1097→  const onMouseMove = (e: React.MouseEvent) => {
  1098→    const rect = canvasRef.current!.getBoundingClientRect();
  1099→    const sx = e.clientX - rect.left;
  1100→    const sy = e.clientY - rect.top;
  1101→    const g = screenToGrid(sx, sy);
  1102→    setCursor(g);
  1103→
  1104→    if (panRef.current) {
  1105→      setPan({
  1106→        x: panRef.current.origin.x + (sx - panRef.current.start.x),
  1107→        y: panRef.current.origin.y + (sy - panRef.current.start.y),
  1108→      });
  1109→      return;
  1110→    }
  1111→
  1112→    // rotation drag — compute snap rotation from cursor angle around component center
  1113→    if (rotateDragRef.current) {
  1114→      const rd = rotateDragRef.current;
  1115→      const curAngle = angleFromCenter(rd.center.x, rd.center.y, sx, sy);
  1116→      // Determine target rotation by snapping cursor angle to nearest 90°.
  1117→      // Screen angle: 0=right, 90=down, 180=left, 270=up.
  1118→      // We map: cursor right (0°) => rotation 0; down (90°) => rotation 1; left (180°) => rotation 2; up (270°) => rotation 3.
  1119→      // But this mapping is relative to the component's "forward" direction. A simpler
  1120→      // approach: compute how many 90° steps the cursor has moved from the start angle,
  1121→      // and add to startRotation.
  1122→      let deltaDeg = curAngle - rd.startAngle;
  1123→      // normalize to -180..180
  1124→      while (deltaDeg > 180) deltaDeg -= 360;
  1125→      while (deltaDeg < -180) deltaDeg += 360;
  1126→      // Snap to nearest 90° step
  1127→      const steps = Math.round(deltaDeg / 90);
  1128→      let targetRotation = (rd.startRotation + steps) % 4;
  1129→      if (targetRotation < 0) targetRotation += 4;
  1130→      const comp = components.find((c) => c.id === rd.componentId);
  1131→      if (comp && comp.rotation !== targetRotation) {
  1132→        // Use rotateComponent repeatedly until we reach the target. Since rotateComponent
  1133→        // only increments by 1, we call it the right number of times.
  1134→        let diff = (targetRotation - comp.rotation + 4) % 4;
  1135→        for (let i = 0; i < diff; i++) {
  1136→          rotateComponent(comp.id);
  1137→        }
  1138→      }
  1139→      return;
  1140→    }
  1141→
  1142→    // wire segment dragging — free 2D drag in any direction
  1143→    if (wireDragRef.current) {
  1144→      const wd = wireDragRef.current;
  1145→      const wire = wires.find((w) => w.id === wd.wireId);
  1146→      if (!wire) return;
  1147→      const fromComp = components.find((c) => c.id === wire.from.componentId);
  1148→      const toComp = components.find((c) => c.id === wire.to.componentId);
  1149→      if (!fromComp || !toComp) return;
  1150→      const fromPlugin = getPlugin(fromComp.type);
  1151→      const toPlugin = getPlugin(toComp.type);
  1152→      if (!fromPlugin || !toPlugin) return;
  1153→      const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
  1154→      const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
  1155→      if (!fromT || !toT) return;
  1156→      const fromPos = getTerminalPos(fromComp, fromT);
  1157→      const toPos = getTerminalPos(toComp, toT);
  1158→      // Always start from the ORIGINAL waypoints captured at drag start.
  1159→      // This gives smooth, predictable behavior: the segment follows the cursor
  1160→      // absolutely, and the path is rebuilt fresh each frame from the original.
  1161→      const newWaypoints = rerouteWireForDrag(
  1162→        { ...fromPos },
  1163→        { ...toPos },
  1164→        wd.segIndex,
  1165→        g,
  1166→        wd.originalWaypoints,
  1167→      );
  1168→      setWireWaypoints(wd.wireId, newWaypoints);
  1169→      return;
  1170→    }
  1171→
  1172→    if (dragRef.current) {
  1173→      if (running) return;
  1174→      const newPos = { x: g.x - dragRef.current.offset.x, y: g.y - dragRef.current.offset.y };
  1175→      moveComponent(dragRef.current.componentId, newPos);
  1176→      return;
  1177→    }
  1178→
  1179→    if (wireDraft) {
  1180→      updateWireCursor(g);
  1181→    }
  1182→
  1183→    // hover detection: priority: terminal > rotate handle > wire handle > wire > component
  1184→    const term = findTerminalAt(g.x, g.y);
  1185→    if (term) {
  1186→      setHover({ componentId: term.componentId, terminal: term, wireId: null, wireHandle: null, rotateHandle: null });
  1187→      return;
  1188→    }
  1189→    // rotate handle
  1190→    if (!running && selection.type === 'component') {
  1191→      const comp = components.find((c) => c.id === selection.id);
  1192→      if (comp) {
  1193→        const handlePos = getRotateHandlePos(comp);
  1194→        if (handlePos) {
  1195→          const dx = sx - handlePos.x;
  1196→          const dy = sy - handlePos.y;
  1197→          if (dx * dx + dy * dy < 144) { // 12px radius hit zone
  1198→            setHover({ componentId: comp.id, terminal: null, wireId: null, wireHandle: null, rotateHandle: comp.id });
  1199→            return;
  1200→          }
  1201→        }
  1202→      }
  1203→    }
  1204→    // wire handle
  1205→    if (!running) {
  1206→      const wh = findWireHandle(sx, sy);
  1207→      if (wh) {
  1208→        setHover({ componentId: null, terminal: null, wireId: wh.wireId, wireHandle: wh, rotateHandle: null });
  1209→        return;
  1210→      }
  1211→    }
  1212→    // wire
  1213→    const wireId = findWireAt(sx, sy);
  1214→    if (wireId) {
  1215→      setHover({ componentId: null, terminal: null, wireId, wireHandle: null, rotateHandle: null });
  1216→      return;
  1217→    }
  1218→    // component
  1219→    const comp = findComponentAt(g.x, g.y);
  1220→    setHover({ componentId: comp?.id ?? null, terminal: null, wireId: null, wireHandle: null, rotateHandle: null });
  1221→  };
  1222→
  1223→  const onMouseUp = (e: React.MouseEvent) => {
  1224→    if (e.button === 1 || e.button === 2) {
  1225→      panRef.current = null;
  1226→      return;
  1227→    }
  1228→    if (rotateDragRef.current) {
  1229→      useEditor.getState().pushHistory();
  1230→      rotateDragRef.current = null;
  1231→      setRotateDrag(null);
  1232→      return;
  1233→    }
  1234→    if (wireDragRef.current) {
  1235→      useEditor.getState().pushHistory();
  1236→      wireDragRef.current = null;
  1237→      return;
  1238→    }
  1239→    if (dragRef.current) {
  1240→      if (!running) useEditor.getState().pushHistory();
  1241→      dragRef.current = null;
  1242→    }
  1243→  };
  1244→
  1245→  const onWheel = (e: React.WheelEvent) => {
  1246→    const rect = canvasRef.current!.getBoundingClientRect();
  1247→    const sx = e.clientX - rect.left;
  1248→    const sy = e.clientY - rect.top;
  1249→    const factor = e.deltaY < 0 ? 1.1 : 1 / 1.1;
  1250→    const newZoom = Math.max(0.4, Math.min(4, zoom * factor));
  1251→    // Keep the grid point under the cursor stationary during zoom.
  1252→    // Grid point under cursor before zoom: (sx - pan.x) / (CELL_SIZE * zoom)
  1253→    // After zoom, we want: sx = gridX * CELL_SIZE * newZoom + newPan.x
  1254→    // => newPan.x = sx - gridX * CELL_SIZE * newZoom
  1255→    const gridX = (sx - pan.x) / (CELL_SIZE * zoom);
  1256→    const gridY = (sy - pan.y) / (CELL_SIZE * zoom);
  1257→    const newPanX = sx - gridX * CELL_SIZE * newZoom;
  1258→    const newPanY = sy - gridY * CELL_SIZE * newZoom;
  1259→    setZoom(newZoom);
  1260→    setPan({ x: newPanX, y: newPanY });
  1261→  };
  1262→
  1263→  const onDrop = (e: React.DragEvent) => {
  1264→    if (running) return;
  1265→    e.preventDefault();
  1266→    const type = e.dataTransfer.getData('application/x-circuit-type');
  1267→    if (!type) return;
  1268→    const rect = canvasRef.current!.getBoundingClientRect();
  1269→    const sx = e.clientX - rect.left;
  1270→    const sy = e.clientY - rect.top;
  1271→    const g = screenToGrid(sx, sy);
  1272→    addComponent(type, g);
  1273→  };
  1274→
  1275→  const onDragOver = (e: React.DragEvent) => {
  1276→    if (running) return;
  1277→    e.preventDefault();
  1278→    e.dataTransfer.dropEffect = 'copy';
  1279→  };
  1280→
  1281→  const onDoubleClick = (e: React.MouseEvent) => {
  1282→    if (running) return;
  1283→    const rect = canvasRef.current!.getBoundingClientRect();
  1284→    const sx = e.clientX - rect.left;
  1285→    const sy = e.clientY - rect.top;
  1286→    const g = screenToGrid(sx, sy);
  1287→    const comp = findComponentAt(g.x, g.y);
  1288→    if (comp) rotateComponent(comp.id);
  1289→  };
  1290→
  1291→  // cursor style based on hover state
  1292→  const getCursorStyle = (): string => {
  1293→    if (rotateDrag) return 'grabbing';
  1294→    if (running) {
  1295→      if (hover.componentId) {
  1296→        const comp = components.find((c) => c.id === hover.componentId);
  1297→        if (comp && TOGGLEABLE_TYPES.has(comp.type)) return 'pointer';
  1298→      }
  1299→      return 'default';
  1300→    }
  1301→    if (hover.terminal) return 'crosshair';
  1302→    if (hover.rotateHandle) return 'grab';
  1303→    if (hover.wireHandle) return 'move';
  1304→    if (hover.wireId) return 'pointer';
  1305→    if (hover.componentId) return 'move';
  1306→    return 'crosshair';
  1307→  };
  1308→
  1309→  // keyboard
  1310→  useEffect(() => {
  1311→    const onKey = (e: KeyboardEvent) => {
  1312→      const target = e.target as HTMLElement;
  1313→      if (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.tagName === 'SELECT') return;
  1314→      if (e.key === 'Delete' || e.key === 'Backspace') {
  1315→        if (running) return;
  1316→        const s = useEditor.getState().selection;
  1317→        if (s.type === 'component') deleteComponent(s.id!);
  1318→        else if (s.type === 'wire') useEditor.getState().deleteWire(s.id!);
  1319→      } else if (e.key === 'r' || e.key === 'R') {
  1320→        if (running) return;
  1321→        const s = useEditor.getState().selection;
  1322→        if (s.type === 'component') rotateComponent(s.id!);
  1323→      } else if (e.key === 'Escape') {
  1324→        cancelWire();
  1325→        setSelection({ type: null, id: null });
  1326→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'z' && !e.shiftKey) {
  1327→        if (running) return;
  1328→        e.preventDefault();
  1329→        useEditor.getState().undo();
  1330→      } else if ((e.ctrlKey || e.metaKey) && (e.key === 'y' || (e.key === 'z' && e.shiftKey))) {
  1331→        if (running) return;
  1332→        e.preventDefault();
  1333→        useEditor.getState().redo();
  1334→      } else if (e.key === ' ') {
  1335→        e.preventDefault();
  1336→        const s = useEditor.getState();
  1337→        s.setRunning(!s.running);
  1338→      }
  1339→    };
  1340→    window.addEventListener('keydown', onKey);
  1341→    return () => window.removeEventListener('keydown', onKey);
  1342→  }, [deleteComponent, rotateComponent, cancelWire, setSelection, running]);
  1343→
  1344→  return (
  1345→    <div
  1346→      ref={containerRef}
  1347→      className="relative h-full w-full overflow-hidden bg-slate-950"
  1348→      onContextMenu={(e) => e.preventDefault()}
  1349→    >
  1350→      <canvas
  1351→        ref={canvasRef}
  1352→        className="absolute inset-0"
  1353→        style={{ cursor: getCursorStyle() }}
  1354→        onMouseDown={onMouseDown}
  1355→        onMouseMove={onMouseMove}
  1356→        onMouseUp={onMouseUp}
  1357→        onMouseLeave={() => { dragRef.current = null; panRef.current = null; wireDragRef.current = null; rotateDragRef.current = null; setRotateDrag(null); }}
  1358→        onWheel={onWheel}
  1359→        onDrop={onDrop}
  1360→        onDragOver={onDragOver}
  1361→        onDoubleClick={onDoubleClick}
  1362→      />
  1363→      {/* status overlay */}
  1364→      <div className="pointer-events-none absolute bottom-2 left-2 rounded-md bg-slate-900/80 px-2 py-1 text-xs font-mono text-slate-400">
  1365→        ({cursor.x.toFixed(1)}, {cursor.y.toFixed(1)})  zoom: {zoom.toFixed(2)}x  {running ? '▶ running' : '⏸ paused'}
  1366→        {running && <span className="ml-2 text-amber-300">· click switches to toggle</span>}
  1367→      </div>
  1368→      <div className="pointer-events-none absolute bottom-2 right-2 rounded-md bg-slate-900/80 px-2 py-1 text-xs font-mono text-slate-400">
  1369→        {running ? 'Click switches to toggle • Space to pause' : 'Drag from left • Double-click/R to rotate • Del delete • Space play/pause'}
  1370→      </div>
  1371→      {/* zoom controls */}
  1372→      <div className="absolute right-2 top-2 flex flex-col gap-1">
  1373→        <button
  1374→          onClick={() => setZoom((z) => Math.min(4, z * 1.2))}
  1375→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700"
  1376→        >+</button>
  1377→        <button
  1378→          onClick={() => setZoom((z) => Math.max(0.4, z / 1.2))}
  1379→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700"
  1380→        >−</button>
  1381→        <button
  1382→          onClick={() => { setZoom(1); setPan({ x: 0, y: 0 }); }}
  1383→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700 text-xs"
  1384→        >⌂</button>
  1385→      </div>
  1386→    </div>
  1387→  );
  1388→}
  1389→
  1390→/** Distance from point (px, py) to segment (ax,ay)-(bx,by) */
  1391→function pointToSegmentDist(px: number, py: number, ax: number, ay: number, bx: number, by: number): number {
  1392→  const dx = bx - ax;
  1393→  const dy = by - ay;
  1394→  const lenSq = dx * dx + dy * dy;
