    13→/** Cache key prefix for default (registered) models. */
    14→const DEFAULT_KEY_PREFIX = 'default:';
    15→/** Sentinel stored in the geometry cache to mark a model-URL load that failed. */
    16→const FAILED_SENTINEL: unique symbol = Symbol('failed');
    17→
    18→/**
    19→ * Look up (and lazily build) the default geometry for a component type.
    20→ * Returns `null` if no default model is registered, or `{ failed: true }`
    21→ * if parsing threw.
    22→ */
    23→function getDefaultGeometry(
    24→  type: string,
    25→  cache: Map<string, THREE.BufferGeometry | typeof FAILED_SENTINEL>,
    26→): THREE.BufferGeometry | null {
    27→  const key = `${DEFAULT_KEY_PREFIX}${type}`;
    28→  const cached = cache.get(key);
    29→  if (cached === FAILED_SENTINEL) return null;
    30→  if (cached) return cached;
    31→  const entry = DEFAULT_MODELS.get(type);
    32→  if (!entry) return null;
    33→  try {
    34→    const model: LoadedModel = parseSTL(entry.stlAscii);
    35→    const geo = modelToGeometry(model);
    36→    cache.set(key, geo);
    37→    return geo;
    38→  } catch (err) {
    39→    console.warn(`[PCB3DViewer] Failed to parse default model for ${type}:`, err);
    40→    cache.set(key, FAILED_SENTINEL);
    41→    return null;
    42→  }
    43→}

    45→/** Fetch and parse a model from a URL, caching the resulting geometry. */
    46→async function fetchModelGeometry(
    47→  url: string,
    48→  cache: Map<string, THREE.BufferGeometry | typeof FAILED_SENTINEL>,
    49→): Promise<void> {
    50→  const cached = cache.get(url);
    51→  if (cached) return;
    52→  try {
    53→    const res = await fetch(url);
    54→    if (!res.ok) throw new Error(`HTTP ${res.status} ${res.statusText}`);
    55→    const buf = await res.arrayBuffer();
    56→    // Parse by inferring the format from the URL's file extension.
    57→    const lower = url.toLowerCase();
    58→    let model: LoadedModel;
    59→    if (lower.endsWith('.stl')) {
    60→      model = parseSTL(buf);
    61→    } else if (lower.endsWith('.wrl') || lower.endsWith('.vrml') || lower.endsWith('.x3dv')) {
    62→      model = parseVRML(new TextDecoder().decode(buf));
    63→    } else if (lower.endsWith('.obj')) {
    64→      model = parseOBJ(new TextDecoder().decode(buf));
    65→    } else if (lower.endsWith('.step') || lower.endsWith('.stp')) {
    66→      model = parseModel(url, buf);
    67→    } else {
    68→      throw new Error(`Unknown model extension for ${url}`);
    69→    }
    70→    cache.set(url, modelToGeometry(model));
    71→  } catch (err) {
    72→    console.warn(`[PCB3DViewer] Failed to load model ${url}:`, err);
    73→    cache.set(url, FAILED_SENTINEL);
    74→  }
    75→}

    77→/** Compute the parametric body height used by the box fallback. */
    78→function parametricBodyHeight(componentType: string): number {
    79→  switch (componentType) {
    80→    case 'dcVoltage':
    81→    case 'acVoltage':
    82→      return 5.0;
    83→    case 'timer555':
    84→    case 'opamp':
    85→      return 3.5;
    86→    case 'arduino':
    87→    case 'arduinoReal':
    88→      return 1.5;
    89→    case 'led':
    90→      return 0.6;
    91→    default:
    92→      return 1.0;
    93→  }
    94→}

    96→/** Position a mesh so its model sits flush on the board surface. */
    97→function positionMeshOnBoard(
    98→  mesh: THREE.Mesh,
    99→  geo: THREE.BufferGeometry,
   100→  fp: Footprint,
   101→  boardThickness: number,
   102→  padHeight: number,
   103→): void {
   104→  geo.computeBoundingBox();
   105→  const bb = geo.boundingBox;
   106→  if (!bb) return;
   107→  const minY = bb.min.y;
   108→  const maxY = bb.max.y;
   109→  if (fp.side === 'top') {
   110→    // Bottom of model aligns with top of pads.
   111→    mesh.position.y = padHeight - minY;
   112→  } else {
   113→    // Top of model aligns with bottom of pads (under the board).
   114→    mesh.position.y = -boardThickness - padHeight - maxY;
   115→  }
   116→  mesh.position.x = fp.position.x;
   117→  mesh.position.z = fp.position.y;
   118→  mesh.rotation.y = (fp.rotation * Math.PI) / 180;
   119→}

   491→      resistor: 0x2a2a3e, capacitor: 0x1a1a1a, led: 0xff3333, diode: 0x1a1a1a,
   492→      timer555: 0x1a1a2e, opamp: 0x1a1a2e, npn: 0x2a2a2a, pnp: 0x2a2a2a,
   493→      nmos: 0x2a2a2a, pmos: 0x2a2a2a, bsim3nmos: 0x2a2a2a, bsim3pmos: 0x2a2a2a,
   494→      bsim4nmos: 0x2a2a2a, bsim4pmos: 0x2a2a2a,
   495→      arduino: 0x005533, arduinoReal: 0x005533, raspberryPi: 0x7c2d12,
   496→      switch: 0x333333, pushButton: 0x333333,
   497→      crystal: 0x888888, inductor: 0x333333, speaker: 0x222222,
   498→      transformer: 0x444444, vco: 0x1a1a2e, and: 0x1a1a2e, or: 0x1a1a2e,
   499→      nand: 0x1a1a2e, nor: 0x1a1a2e, xor: 0x1a1a2e, not: 0x1a1a2e,
   500→    };
   501→
   502→    /** Track which URLs we've already started fetching this render pass. */
   503→    const inFlight = new Set<string>();
   504→
   505→    for (const fp of footprints) {
   506→      const color = componentColors[fp.componentType] ?? 0x111111;
   507→      const mat = new THREE.MeshStandardMaterial({
   508→        color, roughness: 0.6, metalness: 0.2,
   509→      });
   510→
   511→      // 1) External modelUrl takes precedence — look it up in the cache.
   512→      //    If not cached yet, kick off an async fetch (handled below).
   513→      let geo: THREE.BufferGeometry | null = null;
   514→      const modelUrl = fp.modelUrl;
   515→      if (modelUrl) {
   516→        const cached = modelCacheRef.current.get(modelUrl);
   517→        if (cached && cached !== FAILED_SENTINEL) {
   518→          geo = cached;
   519→        } else if (!cached && !inFlight.has(modelUrl)) {
   520→          inFlight.add(modelUrl);
   521→          // Fire-and-forget; when it resolves we bump modelVersion to trigger
   522→          // a re-render, then the cached geometry is picked up synchronously.
   523→          fetchModelGeometry(modelUrl, modelCacheRef.current).finally(bumpModelVersion);
   524→        }
   525→      }
   526→
   527→      // 2) Fall back to the default registered model for this type.
   528→      if (!geo) {
   529→        geo = getDefaultGeometry(fp.componentType, modelCacheRef.current);
   530→      }
   531→
   532→      // 3) Final fallback: parametric box (matches the pre-model rendering).
   533→      if (geo) {
   534→        const mesh = new THREE.Mesh(geo, mat);
   535→        positionMeshOnBoard(mesh, geo, fp, BOARD_THICKNESS, PAD_HEIGHT);
   536→        group.add(mesh);
   537→      } else {
   538→        const bodyH = parametricBodyHeight(fp.componentType);
   539→        const boxGeo = new THREE.BoxGeometry(fp.bodySize.width, bodyH, fp.bodySize.height);
   540→        const mesh = new THREE.Mesh(boxGeo, mat);
   541→        const y = fp.side === 'top'
   542→          ? PAD_HEIGHT + bodyH / 2
   543→          : -BOARD_THICKNESS - PAD_HEIGHT - bodyH / 2;
   544→        mesh.position.set(fp.position.x, y, fp.position.y);
   545→        mesh.rotation.y = (fp.rotation * Math.PI) / 180;
   546→        // Tag for explosion/assembly animation
   547→        (mesh as any).__baseY = y;
   548→        group.add(mesh);
   549→      }
   550→
   551→      // Pads — rendered as metallic discs/cylinders on the board surface
   552→      for (const pad of fp.pads) {
   553→        const r = Math.max(pad.size.width, pad.size.height) / 2;
   554→        const padH = PAD_HEIGHT;
   555→        // Use more segments for smoother circles
   556→        const padGeo = pad.shape === 'circle'
   557→          ? new THREE.CylinderGeometry(r, r, padH, 24)
   558→          : new THREE.BoxGeometry(pad.size.width, padH, pad.size.height);
   559→        // Metallic silver-gold pad material
   560→        const padMat = new THREE.MeshStandardMaterial({
   561→          color: 0xd4a020, roughness: 0.2, metalness: 0.95,
   562→        });
   563→        const padMesh = new THREE.Mesh(padGeo, padMat);
   564→        const padY = fp.side === 'top' ? padH / 2 : -BOARD_THICKNESS - padH / 2;
   565→        padMesh.position.set(pad.position.x, padY, pad.position.y);
   566→        if (pad.shape === 'circle') padMesh.rotation.x = Math.PI / 2;
   567→        (padMesh as any).__baseY = padY;
   568→        group.add(padMesh);
   569→
   570→        // Drill hole for THT pads
   571→        if (pad.drill && pad.drill > 0) {
   572→          const drillR = pad.drill / 2;
   573→          const drillGeo = new THREE.CylinderGeometry(drillR, drillR, BOARD_THICKNESS + 0.05, 16);
   574→          const drillMat = new THREE.MeshStandardMaterial({ color: 0x0a0a0a, roughness: 1.0, metalness: 0.0 });
   575→          const drillMesh = new THREE.Mesh(drillGeo, drillMat);
   576→          drillMesh.position.set(pad.position.x, 0, pad.position.y);
   577→          drillMesh.rotation.x = Math.PI / 2;
   578→          (drillMesh as any).__baseY = 0;
   579→          group.add(drillMesh);
   580→        }
   581→      }
   582→    }
   583→
   584→    sceneRef.current.add(group);
   585→    pcbGroupRef.current = group;
   586→
   587→    // Auto-frame camera
   588→    if (cameraRef.current && controlsRef.current) {
   589→      const centerX = board.width / 2;
   590→      const centerZ = board.height / 2;
   591→      const maxDim = Math.max(board.width, board.height);
   592→      const dist = maxDim * 1.5;
   593→      cameraRef.current.position.set(centerX + dist * 0.5, dist * 0.8, centerZ + dist * 0.5);
   594→      controlsRef.current.target.set(centerX, 0, centerZ);
   595→      controlsRef.current.update();
   596→    }
   597→  }, [footprints, traces, vias, board, padNets, modelVersion, bumpModelVersion]);
   598→
   599→  return (
   600→    <div className="relative h-full w-full overflow-hidden bg-[#0a1628]"
   601→      onContextMenu={(e) => e.preventDefault()}>
   602→      <div ref={containerRef} className="absolute inset-0" />
   603→      {loading && (
   604→        <div className="absolute inset-0 flex items-center justify-center">
   605→          <div className="text-sm text-slate-400">Loading 3D viewer...</div>
   606→        </div>
   607→      )}
   608→      {error && (
   609→        <div className="absolute inset-0 flex items-center justify-center">
   610→          <div className="text-sm text-rose-400">{error}</div>
   611→        </div>
   612→      )}
   613→      {footprints.length === 0 && !loading && !error && (
   614→        <div className="absolute inset-0 flex flex-col items-center justify-center gap-3">
   615→          <div className="text-lg font-semibold text-slate-300">3D PCB Preview</div>
   616→          <div className="text-sm text-slate-500">
   617→            Import a schematic in PCB Layout mode first, then switch to 3D View.
   618→          </div>
   619→        </div>
   620→      )}
   621→      {!loading && !error && footprints.length > 0 && (
   622→        <>
   623→          {/* 3D Viewer control panel — top-right */}
   624→          <div className="absolute right-2 top-2 flex flex-col gap-1 rounded-md bg-slate-900/90 p-2 text-xs">
   625→            <button
   626→              onClick={() => setCrossSection(!crossSection)}
   627→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   628→                crossSection ? 'bg-cyan-500/30 text-cyan-300 border border-cyan-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   629→              }`}
   630→            >
   631→              🔪 Cross-section
   632→            </button>
   633→            {crossSection && (
   634→              <input
   635→                type="range" min={-2} max={5} step={0.1}
   636→                value={crossSectionY}
   637→                onChange={(e) => setCrossSectionY(parseFloat(e.target.value))}
   638→                className="w-full"
   639→              />
   640→            )}
   641→            <button
   642→              onClick={() => setHighQuality(!highQuality)}
   643→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   644→                highQuality ? 'bg-amber-500/30 text-amber-300 border border-amber-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   645→              }`}
   646→            >
   647→              ✨ High-quality render
   648→            </button>
   649→            <button
   650→              onClick={() => setShowCurrentFlow(!showCurrentFlow)}
   651→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   652→                showCurrentFlow ? 'bg-emerald-500/30 text-emerald-300 border border-emerald-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   653→              }`}
   654→            >
   655→              ⚡ Current flow {running ? '' : '(run sim)'}
   656→            </button>
   657→            <button
   658→              onClick={() => setShowVoltageProbes(!showVoltageProbes)}
   659→              className={`flex items-center gap-1 rounded px-2 py-1 transition-colors ${
   660→                showVoltageProbes ? 'bg-blue-500/30 text-blue-300 border border-blue-600' : 'text-slate-300 hover:bg-slate-800 border border-transparent'
   661→              }`}
   662→            >
   663→              📊 Voltage probes
   664→            </button>
   665→            <div className="flex items-center gap-1 px-2 py-1 text-slate-300">
   666→              💥 Explode
   667→              <input
   668→                type="range" min={0} max={1} step={0.05}
   669→                value={explosionFactor}
   670→                onChange={(e) => setExplosionFactor(parseFloat(e.target.value))}
   671→                className="flex-1"
   672→              />
   673→            </div>
   674→            <div className="flex items-center gap-1 px-2 py-1 text-slate-300">
   675→              🔧 Assembly
   676→              <input
   677→                type="range" min={0} max={1} step={0.05}
   678→                value={assemblyProgress}
   679→                onChange={(e) => setAssemblyProgress(parseFloat(e.target.value))}
   680→                className="flex-1"
   681→              />
   682→            </div>
   683→            <button
   684→              onClick={() => {
   685→                setAssemblyProgress(0);
   686→                let p = 0;
   687→                const interval = setInterval(() => {
   688→                  p += 0.05;
   689→                  if (p >= 1) { p = 1; clearInterval(interval); }
   690→                  setAssemblyProgress(p);
   691→                }, 30);
   692→              }}
   693→              className="flex items-center gap-1 rounded px-2 py-1 text-purple-300 hover:bg-slate-800 border border-transparent"
   694→            >
   695→              ▶ Play assembly
   696→            </button>
   697→          </div>
   698→
   699→          {/* Voltage probe labels — floating HTML at projected 3D positions */}
   700→          {showVoltageProbes && simContext && footprints.map((fp) => {
   701→            // Project the footprint's 3D position to screen coords
   702→            if (!cameraRef.current || !rendererRef.current) return null;
   703→            const pos = new THREE.Vector3(fp.position.x, 2, fp.position.y);
   704→            pos.project(cameraRef.current);
   705→            const x = (pos.x + 1) / 2 * (rendererRef.current.domElement.width / window.devicePixelRatio);
   706→            const y = (1 - (pos.y + 1) / 2) * (rendererRef.current.domElement.height / window.devicePixelRatio);
   707→            // Look up voltage from sim context (componentId → first pad's net → node voltage)
   708→            const comp = useEditor.getState().components.find((c) => c.id === fp.componentId);
   709→            let voltage = '—';
   710→            if (comp && simContext) {
   711→              const net = fp.pads[0]?.net;
   712→              if (net) voltage = net;
   713→            }
   714→            return (
   715→              <div
   716→                key={fp.id}
   717→                className="pointer-events-none absolute z-10 rounded bg-blue-900/90 px-1.5 py-0.5 text-[10px] font-mono text-blue-200 border border-blue-700"
   718→                style={{ left: `${x}px`, top: `${y}px`, transform: 'translate(-50%, -150%)' }}
   719→              >
   720→                {fp.refdes}: {voltage}
   721→              </div>
   722→            );
   723→          })}
   724→
   725→          {/* Original bottom overlay */}
   726→          <div className="pointer-events-none absolute bottom-2 left-2 rounded-md bg-[#0a1628]/80 px-3 py-1.5 text-xs font-mono text-slate-400">
   727→            Left-drag: rotate · Right-drag: pan · Scroll: zoom
   728→            {showCurrentFlow && running && <span className="ml-2 text-emerald-300">· ⚡ current flowing</span>}
   729→          </div>
   730→        </>
   731→      )}
   732→    </div>
   733→  );
   734→}
