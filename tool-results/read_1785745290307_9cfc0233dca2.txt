     2→     2→// Each circuit is laid out left-to-right with proper spacing so:
     3→     3→//   - Wires go in clean L-shapes (horizontal then vertical, or vice versa)
     4→     4→//   - No wire crosses through a component body
     5→     5→//   - Ground is at the bottom, sources on the left, loads in the middle/right

    12→    12→function defaultsFor(type: string): Record<string, number | string | boolean> {
    13→    13→  const plugin = getPlugin(type);
    14→    14→  if (!plugin) return {};
    15→    15→  const defaults: Record<string, number | string | boolean> = {};
    16→    16→  for (const p of plugin.parameters) defaults[p.key] = p.default;
    17→    17→  return defaults;
    18→    18→}

    20→    20→function comp(
    21→    21→  type: string,
    22→    22→  id: string,
    23→    23→  pos: [number, number],
    24→    24→  rotation: 0 | 1 | 2 | 3 = 0,
    25→    25→  params?: Record<string, number | string | boolean>,
    26→    26→) {
    27→    27→  return {
    28→    28→    id,
    29→    29→    type,
    30→    30→    position: { x: pos[0], y: pos[1] },
    31→    31→    rotation,
    32→    32→    parameters: { ...defaultsFor(type), ...params },
    33→    33→  };
    34→    34→}

    36→    36→function wire(id: string, fromC: string, fromT: string, toC: string, toT: string, waypoints?: [number, number][]) {
    37→    37→  const w: any = { id, from: { componentId: fromC, terminalId: fromT }, to: { componentId: toC, terminalId: toT } };
    38→    38→  if (waypoints && waypoints.length > 0) {
    39→    39→    w.waypoints = waypoints.map(([x, y]) => ({ x, y }));
    40→    40→  }
    41→    41→  return w;
    42→    42→}

   432→   432→// ----- Example 9: Digital Clock (HH:MM:SS) -----
   433→   433→// A 6-digit digital clock using CD4026 decade counter ICs.
   434→   434→//
   435→   435→// Architecture:
   436→   436→//   1Hz pulse source (crystal) → sec-ones CD4026 → CO → sec-tens → CO →
   437→   437→//   min-ones → CO → min-tens → CO → hr-ones → CO → hr-tens
   438→   438→//
   439→   439→// Each CD4026 counts on the rising edge of its CLK input and drives a 7-segment
   440→   440→// display directly via its a-g outputs. The carry-out (CO) goes HIGH for the
   441→   441→// first half of the count cycle, producing a rising edge on the next CD4026's
   442→   442→// CLK when this counter wraps to 0.
   443→   443→//
   444→   444→// maxCount settings:
   445→   445→//   - ones digits: maxCount=10 (counts 0-9)
   446→   446→//   - tens digits: maxCount=6 (counts 0-5, for seconds/minutes tens)
   447→   447→//   - hours tens: maxCount=3 (counts 0-2, for 24h format)
   448→   448→//
   449→   449→// Note: For a true 24-hour reset (00:00:00 at 24:00:00), add an AND gate that
   450→   450→// detects hr-tens=2 AND hr-ones=4 and drives RST on both hour counters.
   451→   451→// This example omits that for simplicity — it counts 00:00:00 to 29:59:59.
   452→   452→//
   453→   453→// Layout: 6 digit columns, each with a CD4026 (top) and 7-seg (bottom).
   454→   454→//   x=2: hr-tens, x=10: hr-ones, x=20: min-tens, x=28: min-ones,
   455→   455→//   x=38: sec-tens, x=46: sec-ones
   456→   456→//   Crystal at far right, power and ground at bottom.
   457→   457→function digitSegWires(prefix: string, cx: number, segId: string, segCx: number) {
   458→   458→  // Generate 7 segment wires from CD4026 (at cx, y=2) to 7-seg (at segCx, y=8)
   459→   459→  // CD4026 segment outputs at (cx+0..6, 6)
   460→   460→  // 7-seg segment inputs at various positions
   461→   461→  const segPositions: Record<string, [number, number]> = {
   462→   462→    a: [segCx + 0, 8],   // top-left
   463→   463→    b: [segCx + 2, 8],   // top-mid
   464→   464→    c: [segCx + 4, 8],   // top-right
   465→   465→    d: [segCx + 4, 14],  // bot-right
   466→   466→    e: [segCx + 2, 14],  // bot-mid
   467→   467→    f: [segCx + 0, 14],  // bot-left
   468→   468→    g: [segCx + 0, 11],  // mid-left
   469→   469→  };
   470→   470→  const cd4026SegY = 6; // bottom of CD4026
   471→   471→  const segIds = ['a', 'b', 'c', 'd', 'e', 'f', 'g'];
   472→   472→  return segIds.map((s, i) => {
   473→   473→    const [sx, sy] = segPositions[s];
   474→   474→    // Route from (cx+i, 6) to (sx, sy)
   475→   475→    // Simple L-path: go down from CD4026 output to a routing channel, then to the segment
   476→   476→    const waypoints: [number, number][] = [];
   477→   477→    if (sy <= 8) {
   478→   478→      // Top-row segments (a, b, c): route directly down
   479→   479→      waypoints.push([cx + i, 7], [sx, 7]);
   480→   480→    } else {
   481→   481→      // Bottom/side segments (d, e, f, g): route around the 7-seg body
   482→   482→      // Go down to y=7, then horizontal to sx, then down to sy
   483→   483→      waypoints.push([cx + i, 7], [sx, 7]);
   484→   484→    }
   485→   485→    return wire(`${prefix}_${s}`, `ic_${prefix}`, s, `seg_${prefix}`, s, waypoints);
   486→   486→  });
   487→   487→}
