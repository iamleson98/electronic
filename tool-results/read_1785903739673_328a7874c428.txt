    16→// ─────────────────────────────────────────────────────────────────────────────
    17→// Expression parser for behavioral sources
    18→//   Supports: V(node1, node2), I(source), +, -, *, /, ^, exp, log, sin, cos,
    19→//   abs, sqrt, min, max, table, limit, if, time (variable)
    20→// ─────────────────────────────────────────────────────────────────────────────
    21→
    22→type Token = { type: 'num' | 'op' | 'func' | 'paren' | 'var'; value: string };
    23→
    24→function tokenize(expr: string): Token[] {
    25→  const tokens: Token[] = [];
    26→  let i = 0;
    27→  while (i < expr.length) {
    28→    const c = expr[i];
    29→    if (/\s/.test(c)) { i++; continue; }
    30→    if (/[0-9.]/.test(c)) {
    31→      let num = '';
    32→      while (i < expr.length && /[0-9.eE+-]/.test(expr[i])) {
    33→        if ((expr[i] === '+' || expr[i] === '-') && !/[eE]/.test(expr[i - 1] ?? '')) break;
    34→        num += expr[i++];
    35→      }
    36→      tokens.push({ type: 'num', value: num });
    37→      continue;
    38→    }
    39→    if (/[a-zA-Z_]/.test(c)) {
    40→      let id = '';
    41→      while (i < expr.length && /[a-zA-Z0-9_]/.test(expr[i])) id += expr[i++];
    42→      tokens.push(/(exp|log|sin|cos|tan|abs|sqrt|min|max|table|limit|if|V|I|time)/.test(id)
    43→        ? { type: 'func', value: id } : { type: 'var', value: id });
    44→      continue;
    45→    }
    46→    if ('+-*/^()'.includes(c)) {
    47→      tokens.push({ type: c === '(' || c === ')' ? 'paren' : 'op', value: c });
    48→      i++; continue;
    49→    }
    50→    if (c === ',' || c === ';') { i++; continue; }
    51→    i++;
    52→  }
    53→  return tokens;
    54→}
    55→
    56→class ExprEvaluator {
    57→  private tokens: Token[];
    58→  private pos = 0;
    59→  /** function to resolve V(node1, node2) and I(source) */
    60→  private resolver: (kind: 'V' | 'I', args: string[]) => number;
    61→  private time: number;
    62→
    63→  constructor(expr: string, resolver: (kind: 'V' | 'I', args: string[]) => number, time: number) {
    64→    this.tokens = tokenize(expr);
    65→    this.resolver = resolver;
    66→    this.time = time;
    67→  }
    68→
    69→  peek(): Token | null { return this.tokens[this.pos] ?? null; }
    70→  consume(): Token { return this.tokens[this.pos++]; }
    71→
    72→  parse(): number {
    73→    return this.parseExpr();
    74→  }
    75→
    76→  private parseExpr(): number {
    77→    let left = this.parseTerm();
    78→    while (this.peek()?.type === 'op' && (this.peek()!.value === '+' || this.peek()!.value === '-')) {
    79→      const op = this.consume().value;
    80→      const right = this.parseTerm();
    81→      left = op === '+' ? left + right : left - right;
    82→    }
    83→    return left;
    84→  }
    85→
    86→  private parseTerm(): number {
    87→    let left = this.parseFactor();
    88→    while (this.peek()?.type === 'op' && (this.peek()!.value === '*' || this.peek()!.value === '/')) {
    89→      const op = this.consume().value;
    90→      const right = this.parseFactor();
    91→      left = op === '*' ? left * right : left / right;
    92→    }
    93→    return left;
    94→  }
    95→
    96→  private parseFactor(): number {
    97→    let left = this.parseUnary();
    98→    if (this.peek()?.type === 'op' && this.peek()!.value === '^') {
    99→      this.consume();
   100→      const right = this.parseFactor();
   101→      left = Math.pow(left, right);
   102→    }
   103→    return left;
   104→  }
   105→
   106→  private parseUnary(): number {
   107→    if (this.peek()?.type === 'op' && this.peek()!.value === '-') {
   108→      this.consume();
   109→      return -this.parseUnary();
   110→    }
   111→    if (this.peek()?.type === 'op' && this.peek()!.value === '+') {
   112→      this.consume();
   113→      return this.parseUnary();
   114→    }
   115→    return this.parsePrimary();
   116→  }
   117→
   118→  private parsePrimary(): number {
   119→    const tok = this.peek();
   120→    if (!tok) return 0;
   121→    if (tok.type === 'num') { this.consume(); return parseFloat(tok.value); }
   122→    if (tok.type === 'paren' && tok.value === '(') {
   123→      this.consume();
   124→      const v = this.parseExpr();
   125→      if (this.peek()?.value === ')') this.consume();
   126→      return v;
   127→    }
   128→    if (tok.type === 'func') {
   129→      this.consume();
   130→      if (tok.value === 'time') return this.time;
   131→      // function call: expect (
   132→      if (this.peek()?.value === '(') {
   133→        this.consume();
   134→        const args: number[] = [];
   135→        const argStrs: string[] = [];
   136→        // for V/I, we need raw string args; for math functions, parse as expressions
   137→        if (tok.value === 'V' || tok.value === 'I') {
   138→          // collect raw tokens until matching )
   139→          let depth = 1;
   140→          let raw = '';
   141→          while (this.pos < this.tokens.length && depth > 0) {
   142→            const t = this.consume();
   143→            if (t.value === '(') depth++;
   144→            else if (t.value === ')') { depth--; if (depth === 0) break; }
   145→            raw += t.value;
   146→          }
   147→          // split on comma at top level
   148→          const parts = raw.split(',').map((s) => s.trim());
   149→          argStrs.push(...parts);
   150→          return this.resolver(tok.value as 'V' | 'I', argStrs);
   151→        }
   152→        // math functions
   153→        while (this.peek() && this.peek()!.value !== ')') {
   154→          args.push(this.parseExpr());
   155→          if (this.peek()?.value === ',') this.consume();
   156→        }
   157→        if (this.peek()?.value === ')') this.consume();
   158→        switch (tok.value) {
   159→          case 'exp': return Math.exp(args[0]);
   160→          case 'log': return Math.log(args[0]);
   161→          case 'sin': return Math.sin(args[0]);
   162→          case 'cos': return Math.cos(args[0]);
   163→          case 'tan': return Math.tan(args[0]);
   164→          case 'abs': return Math.abs(args[0]);
   165→          case 'sqrt': return Math.sqrt(args[0]);
   166→          case 'min': return Math.min(args[0], args[1]);
   167→          case 'max': return Math.max(args[0], args[1]);
   168→          case 'limit': return Math.max(args[1], Math.min(args[0], args[2]));
   169→          case 'if': return args[0] !== 0 ? args[1] : args[2];
   170→          default: return 0;
   171→        }
   172→      }
   173→    }
   174→    if (tok.type === 'var') { this.consume(); return 0; }  // unknown variable
   175→    this.consume();
   176→    return 0;
   177→  }
   178→}

   180→/**
   181→ * Evaluate a behavioral expression against the current sim context.
   182→ * `lookupNode` is a function that converts a node name to its voltage.
   183→ */
   184→export function evalExpression(
   185→  expr: string,
   186→  nodeVoltage: Float64Array,
   187→  nodeNameToId: Map<string, number>,
   188→  time: number,
   189→  branchCurrents: Map<string, number> = new Map(),
   190→): number {
   191→  const resolver = (kind: 'V' | 'I', args: string[]): number => {
   192→    if (kind === 'V') {
   193→      const n1 = args[0];
   194→      const n2 = args[1] ?? '0';
   195→      const v1 = n1 === '0' || n1.toLowerCase() === 'gnd' ? 0 : (nodeNameToId.get(n1) !== undefined ? nodeVoltage[nodeNameToId.get(n1)!] : 0);
   196→      const v2 = n2 === '0' || n2.toLowerCase() === 'gnd' ? 0 : (nodeNameToId.get(n2) !== undefined ? nodeVoltage[nodeNameToId.get(n2)!] : 0);
   197→      return v1 - v2;
   198→    } else {
   199→      // I(source) — look up branch current
   200→      return branchCurrents.get(args[0]) ?? 0;
   201→    }
   202→  };
   203→  try {
   204→    const e = new ExprEvaluator(expr, resolver, time);
   205→    return e.parse();
   206→  } catch {
   207→    return 0;
   208→  }
   209→}

   877→  stamp(params, terminals, sys, sim) {
   878→    const inp = terminals.find((t) => t.terminalId === 'in+')!.nodeId;
   879→    const inn = terminals.find((t) => t.terminalId === 'in-')!.nodeId;
   880→    const out = terminals.find((t) => t.terminalId === 'out')!.nodeId;
   881→    const gain = params.gain as number;
   882→    const rout = Math.max(1, params.rout as number);
   883→    const voff = (params.voff as number) / 1000; // mV to V
   884→    // Open-loop gain with finite output resistance
   885→    // V(out) = gain * (V(in+) - V(in-) - voff), limited by VCC/VEE
   886→    // For small-signal: just VCVS with gain, plus Rout in series
   887→    sys.stampVCVS(out, 0, inp, inn, gain);
   888→    // offset (current source to in- terminal)
   889→    sys.stampCurrentSource(0, inn, voff * gain / 1e6);  // tiny offset current
   890→    // output resistance
   891→    sys.stampConductance(out, 0, 1 / rout);
   892→    void sim;
