    52→/** Get the orthogonal path points for a wire.
    53→ *  fromPos and toPos are in SCREEN coords. Waypoints are in GRID coords and
    54→ *  are converted to screen coords using the provided converter. */
    55→function getWirePath(
    56→  wire: Wire,
    57→  fromPos: Vec2,
    58→  toPos: Vec2,
    59→  gridToScreenFn: (gx: number, gy: number) => Vec2,
    60→  use45: boolean = false,
    61→): Vec2[] {
    62→  const points: Vec2[] = [fromPos];
    63→  if (wire.waypoints && wire.waypoints.length > 0) {
    64→    for (const wp of wire.waypoints) {
    65→      points.push(gridToScreenFn(wp.x, wp.y));
    66→    }
    67→  } else if (use45) {
    68→    // 45-degree routing: L-shape with 45° diagonal in the middle
    69→    const dx = toPos.x - fromPos.x;
    70→    const dy = toPos.y - fromPos.y;
    71→    const absDx = Math.abs(dx);
    72→    const absDy = Math.abs(dy);
    73→    if (absDx < 1 || absDy < 1) {
    74→      // Nearly straight — just go direct
    75→      points.push({ x: toPos.x, y: fromPos.y });
    76→    } else if (absDx > absDy) {
    77→      // More horizontal: go horizontal, then 45° diagonal, then horizontal
    78→      const diagLen = absDy;
    79→      const sign = Math.sign(dy);
    80→      const horizSign = Math.sign(dx);
    81→      const midX1 = fromPos.x + horizSign * (absDx - diagLen) / 2;
    82→      const midX2 = midX1 + horizSign * diagLen;
    83→      points.push({ x: midX1, y: fromPos.y });
    84→      points.push({ x: midX2, y: fromPos.y + sign * diagLen });
    85→      points.push({ x: midX2, y: toPos.y });
    86→    } else {
    87→      // More vertical: go vertical, then 45° diagonal, then vertical
    88→      const diagLen = absDx;
    89→      const sign = Math.sign(dx);
    90→      const vertSign = Math.sign(dy);
    91→      const midY1 = fromPos.y + vertSign * (absDy - diagLen) / 2;
    92→      const midY2 = midY1 + vertSign * diagLen;
    93→      points.push({ x: fromPos.x, y: midY1 });
    94→      points.push({ x: fromPos.x + sign * diagLen, y: midY2 });
    95→      points.push({ x: toPos.x, y: midY2 });
    96→    }
    97→  } else {
    98→    // default orthogonal routing: go to midpoint X, then to target
    99→    const midX = (fromPos.x + toPos.x) / 2;
   100→    points.push({ x: midX, y: fromPos.y });
   101→    points.push({ x: midX, y: toPos.y });
   102→  }
   103→  points.push(toPos);
   104→  return points;
   105→}

   107→/** Compute the midpoint of a segment for drag handle detection */
   108→function segmentMidpoint(a: Vec2, b: Vec2): Vec2 {
   109→  return { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
   110→}

   112→/** Angle (in degrees, 0-360) from a center point to a target point */
   113→function angleFromCenter(cx: number, cy: number, x: number, y: number): number {
   114→  const angle = Math.atan2(y - cy, x - cx) * 180 / Math.PI;
   115→  return (angle + 360) % 360;
   116→}

   118→/**
   119→ * Re-route a wire so a dragged segment follows the cursor freely in 2D.
   120→ *
   121→ * The wire is ALWAYS kept orthogonal (only horizontal/vertical segments).
   122→ * Terminal positions (fromPos, toPos) are NEVER moved — they're fixed to
   123→ * component terminals. Only waypoints change.
   124→ *
   125→ * Path layout: [fromPos, wp1, wp2, ..., toPos]
   126→ *   segIndex 0 = fromPos→wp1, segIndex 1 = wp1→wp2, ..., last = wpN→toPos
   127→ *
   128→ * When the user drags a segment handle, we move that segment to the cursor
   129→ * position. The dragged segment stays horizontal or vertical (whichever it
   130→ * was), and its perpendicular axis snaps to the cursor. The parallel axis
   131→ * also follows the cursor when possible (for interior segments with waypoints
   132→ * on both ends). Adjacent segments naturally stretch/shrink to stay connected.
   133→ */
   134→function rerouteWireForDrag(
   135→  fromPos: Vec2,
   136→  toPos: Vec2,
   137→  segIndex: number,
   138→  cursorGrid: Vec2,
   139→  existingWaypoints: Vec2[],
   140→): Vec2[] {
   141→  // Build waypoint copy (never touch fromPos/toPos)
   142→  const wps: Vec2[] = existingWaypoints.length > 0
   143→    ? existingWaypoints.map((w) => ({ ...w }))
   144→    : [
   145→        { x: (fromPos.x + toPos.x) / 2, y: fromPos.y },
   146→        { x: (fromPos.x + toPos.x) / 2, y: toPos.y },
   147→      ];
   148→
   149→  // Helper: get path point by index (0=fromPos, 1..n=waypoints, n+1=toPos)
   150→  const getPt = (i: number): Vec2 => {
   151→    if (i === 0) return fromPos;
   152→    if (i === wps.length + 1) return toPos;
   153→    return wps[i - 1];
   154→  };
   155→  // Helper: set a waypoint by path index (only waypoints are settable)
   156→  const setPt = (i: number, val: Vec2) => {
   157→    if (i > 0 && i <= wps.length) wps[i - 1] = val;
   158→  };
   159→
   160→  const a = getPt(segIndex);
   161→  const b = getPt(segIndex + 1);
   162→  const isHorizontal = Math.abs(b.y - a.y) < Math.abs(b.x - a.x);
   163→
   164→  if (isHorizontal) {
   165→    // Horizontal segment: perpendicular axis = Y. Cursor Y becomes the new Y.
   166→    // Parallel axis = X: shift segment along X toward cursor (only waypoints move).
   167→    const newY = cursorGrid.y;
   168→    const midX = (a.x + b.x) / 2;
   169→    const deltaX = cursorGrid.x - midX;
   170→    const aIdx = segIndex;
   171→    const bIdx = segIndex + 1;
   172→    const aIsFixed = (aIdx === 0);           // fromPos can't move
   173→    const bIsFixed = (bIdx === wps.length + 1); // toPos can't move
   174→    // Set Y on both endpoints (waypoints only; fixed terminals keep their Y)
   175→    const newA = { ...getPt(aIdx) };
   176→    const newB = { ...getPt(bIdx) };
   177→    if (!aIsFixed) newA.y = newY;
   178→    if (!bIsFixed) newB.y = newY;
   179→    // Shift X: both endpoints if both are waypoints; only one if the other is fixed
   180→    if (!aIsFixed && !bIsFixed) {
   181→      newA.x += deltaX;
   182→      newB.x += deltaX;
   183→    } else if (aIsFixed && !bIsFixed) {
   184→      newB.x += deltaX;
   185→    } else if (!aIsFixed && bIsFixed) {
   186→      newA.x += deltaX;
   187→    }
   188→    setPt(aIdx, newA);
   189→    setPt(bIdx, newB);
   190→  } else {
   191→    // Vertical segment: perpendicular axis = X. Cursor X becomes the new X.
   192→    // Parallel axis = Y: shift segment along Y toward cursor.
   193→    const newX = cursorGrid.x;
   194→    const midY = (a.y + b.y) / 2;
   195→    const deltaY = cursorGrid.y - midY;
   196→    const aIdx = segIndex;
   197→    const bIdx = segIndex + 1;
   198→    const aIsFixed = (aIdx === 0);
   199→    const bIsFixed = (bIdx === wps.length + 1);
   200→    const newA = { ...getPt(aIdx) };
   201→    const newB = { ...getPt(bIdx) };
   202→    if (!aIsFixed) newA.x = newX;
   203→    if (!bIsFixed) newB.x = newX;
   204→    if (!aIsFixed && !bIsFixed) {
   205→      newA.y += deltaY;
   206→      newB.y += deltaY;
   207→    } else if (aIsFixed && !bIsFixed) {
   208→      newB.y += deltaY;
   209→    } else if (!aIsFixed && bIsFixed) {
   210→      newA.y += deltaY;
   211→    }
   212→    setPt(aIdx, newA);
   213→    setPt(bIdx, newB);
   214→  }
   215→
   216→  // After moving waypoints, the path may have diagonal segments adjacent to
   217→  // fixed terminals. Re-orthogonalize by inserting elbow waypoints.
   218→  return orthogonalizePath(fromPos, toPos, wps);
   219→}
   220→
   221→/**
   222→ * Ensure a wire path is fully orthogonal (no diagonal segments).
   223→ * If moving a waypoint created a diagonal segment adjacent to a fixed terminal,
   224→ * insert an extra elbow waypoint to break it into two orthogonal segments.
   225→ *
   226→ * This is called after rerouteWireForDrag to clean up any diagonals.
   227→ */
   228→function orthogonalizePath(fromPos: Vec2, toPos: Vec2, wps: Vec2[]): Vec2[] {
   229→  const result: Vec2[] = [];
   230→  const full: Vec2[] = [fromPos, ...wps, toPos];
   231→  for (let i = 0; i < full.length - 1; i++) {
   232→    const a = full[i];
   233→    const b = full[i + 1];
   234→    result.push({ ...a });
   235→    // If segment is diagonal, insert an elbow (go horizontal first, then vertical)
   236→    if (Math.abs(a.x - b.x) > 0.01 && Math.abs(a.y - b.y) > 0.01) {
   237→      // Insert elbow at (b.x, a.y) — horizontal first, then vertical
   238→      result.push({ x: b.x, y: a.y });
   239→    }
   240→  }
   241→  result.push({ ...toPos });
   242→  // Convert back to waypoints (exclude fromPos and toPos)
   243→  return result.slice(1, -1);
   244→}

   418→        const b = path[i + 1];

   426→  // find a draggable wire segment midpoint handle

   428→    for (const wire of wires) {
   429→      const fromComp = components.find((c) => c.id === wire.from.componentId);
   430→      const toComp = components.find((c) => c.id === wire.to.componentId);

   433→      const toPlugin = getPlugin(toComp.type);
   434→      if (!fromPlugin || !toPlugin) continue;
   435→      const fromT = fromPlugin.terminals.find((t) => t.id === wire.from.terminalId);
   436→      const toT = toPlugin.terminals.find((t) => t.id === wire.to.terminalId);
   437→      if (!fromT || !toT) continue;
   438→      const fromPos = gridToScreen(getTerminalPos(fromComp, fromT).x, getTerminalPos(fromComp, fromT).y);
   439→      const toPos = gridToScreen(getTerminalPos(toComp, toT).x, getTerminalPos(toComp, toT).y);
   440→      const path = getWirePath(wire, fromPos, toPos, gridToScreen, use45Routing);

   442→        const a = path[i];
   443→        const b = path[i + 1];

   449→        }
   450→      }

   464→      y: centerScreen.y - (bb.height / 2 * CELL_SIZE * zoom) - 18,

   623→          const dotSpacing = 24;
   624→          let totalLen = 0;
   625→          const segLens: number[] = [];

   644→              distAlong = ((distAlong % totalLen) + totalLen) % totalLen;

   647→                if (acc + segLens[i] >= distAlong) {
   648→                  const t = (distAlong - acc) / segLens[i];

   650→                  const b = path[i + 1];

   652→                  const y = a.y + (b.y - a.y) * t;

   654→                  ctx.arc(x, y, 3, 0, Math.PI * 2);
   655→                  ctx.fill();

   657→                }
   658→                acc += segLens[i];

   699→      ctx.translate(origin.x, origin.y);

   841→                for (let i = 0; i < segLens.length; i++) {

   845→                    const b = flowScreenPath[i + 1];

   853→                  acc += segLens[i];

   855→              }
   856→              ctx.shadowBlur = 0;
   857→            }
   858→          }
   859→        }
   860→      }
   861→
   862→      // draw rotation handle on selected component (only when not running)
   863→      if (isSelected && !running) {
   864→        const bb = plugin.boundingBox;
   865→        const centerGrid = { x: comp.position.x + bb.width / 2, y: comp.position.y + bb.height / 2 };
   866→        const centerScreen = gridToScreen(centerGrid.x, centerGrid.y);

   872→          ctx.lineWidth = 1;

   900→          ctx.fill();

   902→          // While dragging rotation, show a visual indicator of the target snap angle

   904→            const cursorScreen = gridToScreen(cursor.x, cursor.y);

   907→            // Up = 270°, Right = 0°, Down = 90°, Left = 180° (screen y is down)

   911→            // Convert cursor angle to a rotation increment (0=right, 1=down, 2=left, 3=up)
   912→            // We want: cursor up (angle ~270) => rotation 3 (or whichever makes component point up)
   913→            // Use the angle to determine nearest of 4 directions
   914→            const normalized = (curAngle + 45) % 360;
   915→            if (normalized < 90) snapRot = 0;
   916→            else if (normalized < 180) snapRot = 1;
   917→            else if (normalized < 270) snapRot = 2;

  1399→        if (running) return;
  1400→        const s = useEditor.getState();
  1401→        if (s.multiSelection.components.size > 0 || s.multiSelection.wires.size > 0) {
  1402→          s.deleteSelected();
  1403→        } else if (s.selection.type === 'component') {
  1404→          deleteComponent(s.selection.id!);
  1405→        } else if (s.selection.type === 'wire') {
  1406→          s.deleteWire(s.selection.id!);
  1407→        }
  1408→      } else if (e.key === 'r' || e.key === 'R') {
  1409→        if (running) return;
  1410→        const s = useEditor.getState().selection;
  1411→        if (s.type === 'component') rotateComponent(s.id!);
  1412→      } else if (e.key === 'Escape') {
  1413→        cancelWire();
  1414→        setSelection({ type: null, id: null });
  1415→        useEditor.getState().clearMultiSelection();
  1416→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'z' && !e.shiftKey) {
  1417→        if (running) return;
  1418→        e.preventDefault();
  1419→        useEditor.getState().undo();
  1420→      } else if ((e.ctrlKey || e.metaKey) && (e.key === 'y' || (e.key === 'z' && e.shiftKey))) {
  1421→        if (running) return;
  1422→        e.preventDefault();
  1423→        useEditor.getState().redo();
  1424→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'c') {
  1425→        if (running) return;
  1426→        e.preventDefault();
  1427→        useEditor.getState().copySelection();
  1428→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'v') {
  1429→        if (running) return;
  1430→        e.preventDefault();
  1431→        useEditor.getState().paste();
  1432→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'd') {
  1433→        if (running) return;
  1434→        e.preventDefault();
  1435→        useEditor.getState().duplicate();
  1436→      } else if ((e.ctrlKey || e.metaKey) && e.key === 'a') {
  1437→        if (running) return;
  1438→        e.preventDefault();
  1439→        const s = useEditor.getState();
  1440→        s.setMultiSelection({
  1441→          components: new Set(s.components.map((c) => c.id)),
  1442→          wires: new Set(s.wires.map((w) => w.id)),
  1443→        });
  1444→      } else if (e.key === ' ') {
  1445→        e.preventDefault();
  1446→        const s = useEditor.getState();
  1447→        s.setRunning(!s.running);
  1448→      } else if (e.key === '\\' || e.key === '|') {
  1449→        e.preventDefault();
  1450→        setUse45Routing((v) => !v);
  1451→      }
  1452→    };
  1453→    window.addEventListener('keydown', onKey);
  1454→    return () => window.removeEventListener('keydown', onKey);
  1455→  }, [deleteComponent, rotateComponent, cancelWire, setSelection, running]);
  1456→
  1457→  return (
  1458→    <div
  1459→      ref={containerRef}
  1460→      className="relative h-full w-full overflow-hidden bg-slate-950"
  1461→      onContextMenu={(e) => e.preventDefault()}
  1462→    >
  1463→      <canvas
  1464→        ref={canvasRef}
  1465→        className="absolute inset-0"
  1466→        style={{ cursor: getCursorStyle() }}
  1467→        onMouseDown={onMouseDown}
  1468→        onMouseMove={onMouseMove}
  1469→        onMouseUp={onMouseUp}
  1470→        onMouseLeave={() => { dragRef.current = null; panRef.current = null; wireDragRef.current = null; rotateDragRef.current = null; setRotateDrag(null); }}
  1471→        onWheel={onWheel}
  1472→        onDrop={onDrop}
  1473→        onDragOver={onDragOver}
  1474→        onDoubleClick={onDoubleClick}
  1475→      />
  1476→      {/* status overlay */}
  1477→      <div className="pointer-events-none absolute bottom-2 left-2 rounded-md bg-slate-900/80 px-2 py-1 text-xs font-mono text-slate-400">
  1478→        ({cursor.x.toFixed(1)}, {cursor.y.toFixed(1)})  zoom: {zoom.toFixed(2)}x  {running ? '▶ running' : '⏸ paused'}
  1479→        {running && <span className="ml-2 text-amber-300">· click switches to toggle</span>}
  1480→      </div>
  1481→      <div className="pointer-events-none absolute bottom-2 right-2 rounded-md bg-slate-900/80 px-2 py-1 text-xs font-mono text-slate-400">
  1482→        {running ? 'Click switches to toggle • Space to pause' : 'Drag from left • Double-click/R to rotate • Del delete • Space play/pause'}
  1483→      </div>
  1484→      {/* zoom controls */}
  1485→      <div className="absolute right-2 top-2 flex flex-col gap-1">
  1486→        <button
  1487→          onClick={() => setZoom((z) => Math.min(4, z * 1.2))}
  1488→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700"
  1489→        >+</button>
  1490→        <button
  1491→          onClick={() => setZoom((z) => Math.max(0.4, z / 1.2))}
  1492→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700"
  1493→        >−</button>
  1494→        <button
  1495→          onClick={() => { setZoom(1); setPan({ x: 0, y: 0 }); }}
  1496→          className="h-7 w-7 rounded bg-slate-800 text-slate-200 hover:bg-slate-700 text-xs"
  1497→        >⌂</button>
  1498→      </div>
  1499→    </div>
  1500→  );
  1501→}

  1503→/** Distance from point (px, py) to segment (ax,ay)-(bx,by) */
  1504→function pointToSegmentDist(px: number, py: number, ax: number, ay: number, bx: number, by: number): number {
  1505→  const dx = bx - ax;
  1506→  const dy = by - ay;
  1507→  const lenSq = dx * dx + dy * dy;
  1508→  if (lenSq === 0) return Math.hypot(px - ax, py - ay);
  1509→  let t = ((px - ax) * dx + (py - ay) * dy) / lenSq;
  1510→  t = Math.max(0, Math.min(1, t));
  1511→  const cx = ax + t * dx;
  1512→  const cy = ay + t * dy;
  1513→  return Math.hypot(px - cx, py - cy);
  1514→}
