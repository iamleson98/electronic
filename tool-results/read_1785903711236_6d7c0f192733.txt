   167→  stamp(params, terminals, sys, sim) {
   168→    const c = terminals.find((t) => t.terminalId === 'c')!.nodeId;
   169→    const b = terminals.find((t) => t.terminalId === 'b')!.nodeId;
   170→    const e = terminals.find((t) => t.terminalId === 'e')!.nodeId;
   171→    const hfe = params.hfe as number;
   172→    const vbeOn = params.vbe as number;
   173→    const vceSat = params.satV as number;
   174→
   175→    const vbe = sim.nodeVoltage[b] - sim.nodeVoltage[e];
   176→    const vce = sim.nodeVoltage[c] - sim.nodeVoltage[e];
   177→    const st = sim.state.__global ?? (sim.state.__global = {});
   178→    const key = `npn_${c}_${b}_${e}`;
   179→    const prevOn = st[key] ?? false;
   180→    // Previous step's actual base current (through the b-e voltage source).
   181→    // This is the right signal for "is the external circuit still driving
   182→    // the base?" — when the button is released, this drops to ~0 even though
   183→    // the npn's own voltage source holds V_B at vbeOn.
   184→    const prevIb = (st[key + '_ib'] as number) ?? 0;
   185→    // Turn-on: Vbe above threshold (external circuit drives the base).
   186→    // Stay-on: Vbe above threshold (with hysteresis) AND external base
   187→    // current is still flowing (above 1 nA noise floor).
   188→    // Turn-off: external base current has dropped to ~0 (e.g., button released).
   189→    const on = prevOn
   190→      ? (vbe > vbeOn - 0.1 && prevIb > 1e-9)
   191→      : (vbe > vbeOn);
   192→    st[key] = on;
   193→    if (!on) {
   194→      // Off: tiny leak so the matrix stays non-singular and the base is
   195→      // pulled to a defined voltage by the external circuit.
   196→      sys.stampConductance(c, e, 1e-9);
   197→      sys.stampConductance(b, e, 1e-9);
   198→      st[key + '_ib'] = 0;
   199→      st[key + '_branch'] = -1;
   200→      return;
   201→    }
   202→    // On: model the B-E junction as a voltage source (Vbe = vbeOn).
   203→    // The current through this voltage source IS the external base current
   204→    // — when the button is released, this drops to 0 even though V_B stays
   205→    // at vbeOn. The CCCS then produces I_C = hfe * I_B = 0, so no collector
   206→    // current flows. The npn is "on" in state but conducts nothing — which
   207→    // is exactly the desired behavior for an open base.
   208→    const ibBranch = sys.stampVoltageSource(b, e, vbeOn);
   209→    sys.stampCCCS(c, e, ibBranch, hfe);
   210→    st[key + '_branch'] = ibBranch;
   211→    // Saturation clamp: only when there's actual collector current to
   212→    // saturate. Without this guard, the 100S clamp would provide a current
   213→    // path C→E even when I_B (and thus I_C) is zero — which is the
   214→    // "current flows when switch is open" bug.
   215→    const prevIc = hfe * prevIb;
   216→    if (prevIc > 1e-9 && vce < vceSat) {
   217→      // Clamp Vce ≈ vceSat using a parallel conductance (not a second voltage
   218→      // source — that would create a singular matrix with the CCCS).
   219→      sys.stampConductance(c, e, 100);
   220→    }
   221→  },
   222→  step(params, terminals, sim) {
   223→    const c = terminals.find((t) => t.terminalId === 'c')!.nodeId;
   224→    const b = terminals.find((t) => t.terminalId === 'b')!.nodeId;
   225→    const e = terminals.find((t) => t.terminalId === 'e')!.nodeId;
   226→    const st = sim.state.__global ?? (sim.state.__global = {});
   227→    const key = `npn_${c}_${b}_${e}`;
   228→    const branchIdx = st[key + '_branch'] as number;
   229→    if (branchIdx == null || branchIdx < 0) {
   230→      st[key + '_ib'] = 0;
   231→      return;
   232→    }
   233→    // sim.branchCurrent[i] corresponds to extra var at index numNonGroundNodes + i.
   234→    // The branch index from stampVoltageSource is the matrix index of the
   235→    // extra variable. Convert to sim.branchCurrent index.
   236→    const numNonGround = sim.nodeVoltage.length - 1;
   237→    const relIdx = branchIdx - numNonGround;
   238→    if (relIdx < 0 || relIdx >= sim.branchCurrent.length) {
   239→      st[key + '_ib'] = 0;
   240→      return;
   241→    }
   242→    // Branch current is positive when conventional current flows from b to e
   243→    // externally (i.e., INTO the base). That's exactly the external base
   244→    // drive we need to detect turn-off.
   245→    st[key + '_ib'] = sim.branchCurrent[relIdx];
   246→  },
   247→  measure(params, terminals, sim) {
   248→    const c = terminals.find((t) => t.terminalId === 'c')!.nodeId;
   249→    const b = terminals.find((t) => t.terminalId === 'b')!.nodeId;
   250→    const e = terminals.find((t) => t.terminalId === 'e')!.nodeId;
   251→    return [
   252→      { label: 'Vbe', value: (sim.nodeVoltage[b] - sim.nodeVoltage[e]).toFixed(3), unit: 'V' },
   253→      { label: 'Vce', value: (sim.nodeVoltage[c] - sim.nodeVoltage[e]).toFixed(3), unit: 'V' },
   254→    ];
   255→  },

   406→  // 555 timer: state-aware stamp and step using sim.state keyed by terminal topology.
   407→  // The state (flip-flop, outHigh) persists across steps via sim.state[key].
   408→  //
   409→  // Two modes:
   410→  //   1. Normal mode (astable=false): uses external RC network, reads THR/TRIG
   411→  //      voltages. Includes fast-forward to advance sim.time toward threshold
   412→  //      crossings.
   413→  //   2. Astable mode (astable=true): computes output directly from sim.time
   414→  //      using R1/R2/C parameters. Bypasses the slow RC simulation so the
   415→  //      555 oscillates at visible speed. The external RC components are
   416→  //      still wired (for visual authenticity) but the 555 uses its internal
   417→  //      timing model.
   418→  stamp(params, terminals, sys, sim) {
   419→    const vcc = params.vcc as number;
   420→    const vccNode = terminals.find((t) => t.terminalId === 'vcc')!.nodeId;
   421→    const gndNode = terminals.find((t) => t.terminalId === 'gnd')!.nodeId;
   422→    if (vccNode !== gndNode) sys.stampConductance(vccNode, gndNode, 1e-9);
   423→
   424→    // CTRL pin: if left unconnected, add weak pull to 2/3 VCC
   425→    const ctrlNode = terminals.find((t) => t.terminalId === 'ctrl')!.nodeId;
   426→    if (ctrlNode !== gndNode && ctrlNode !== vccNode) {
   427→      const ctrlG = 1 / 5e6;
   428→      sys.stampConductance(ctrlNode, gndNode, ctrlG);
   429→      sys.stampCurrentSource(gndNode, ctrlNode, (2 / 3) * vcc * ctrlG);
   430→    }
   431→
   432→    const key = stateKey555(terminals);
   433→    const st = sim.state[key] ?? (sim.state[key] = { ff: false, outHigh: false, prevV: 0, prevT: -1 });
   434→    const out = terminals.find((t) => t.terminalId === 'out')!.nodeId;
   435→    const dis = terminals.find((t) => t.terminalId === 'dis')!.nodeId;
   436→
   437→    // ── Astable mode: compute output directly from sim.time ────────────
   438→    if (params.astable) {
   439→      const r1 = (params.r1 as number) || 47000;
   440→      const r2 = (params.r2 as number) || 47000;
   441→      const c = (params.c as number) || 1e-5;
   442→      // 555 astable formula:
   443→      //   t_high = 0.693 * (R1 + R2) * C
   444→      //   t_low  = 0.693 * R2 * C
   445→      //   period = t_high + t_low = 0.693 * (R1 + 2*R2) * C
   446→      const tHigh = 0.693 * (r1 + r2) * c;
   447→      const tLow = 0.693 * r2 * c;
   448→      const period = tHigh + tLow;
   449→      const phase = (sim.time % period);
   450→      st.outHigh = phase < tHigh;
   451→      st.ff = st.outHigh;
   452→
   453→      if (out !== gndNode) sys.stampVoltageSource(out, gndNode, st.outHigh ? vcc : 0);
   454→      // DIS pin: conducts to GND when output is LOW (ff=false)
   455→      if (!st.ff) {
   456→        if (dis !== gndNode) sys.stampConductance(dis, gndNode, 1 / 50);
   457→      } else {
   458→        if (dis !== gndNode) sys.stampConductance(dis, gndNode, 1e-9);
   459→      }
   460→      return;
   461→    }
   462→
   463→    // ── Normal mode: use external RC, with fast-forward ────────────────
   464→    const thrNode = terminals.find((t) => t.terminalId === 'thr')!.nodeId;
   465→    const vThr = sim.nodeVoltage[thrNode] ?? 0;
   466→    const ctrlV = ctrlNode !== gndNode && ctrlNode !== vccNode ? sim.nodeVoltage[ctrlNode] ?? 0 : 0;
   467→    const vThresh = ctrlV > 0.1 ? ctrlV : (2 / 3) * vcc;
   468→    const vTrigThresh = ctrlV > 0.1 ? ctrlV / 2 : (1 / 3) * vcc;
   469→    const prevV = st.prevV ?? 0;
   470→    const prevT = st.prevT ?? -1;
   471→    const elapsed = sim.time - prevT;
   472→    if (elapsed > 0 && Math.abs(vThr - prevV) > 1e-6) {
   473→      const dVdt = (vThr - prevV) / elapsed;
   474→      if (Math.abs(dVdt) > 0.01) {
   475→        let timeToCross = Infinity;
   476→        if (st.ff && vThr < vThresh) {
   477→          timeToCross = (vThresh - vThr) / dVdt;
   478→        } else if (!st.ff && vThr > vTrigThresh) {
   479→          timeToCross = (vTrigThresh - vThr) / dVdt;
   480→        }
   481→        if (timeToCross > 0 && timeToCross < 2) {
   482→          const advance = Math.min(0.016, timeToCross * 0.9);
   483→          if (advance > sim.dt) {
   484→            sim.time = sim.time + advance;
   485→          }
   486→        }
   487→      }
   488→    }
   489→    st.prevV = vThr;
   490→    st.prevT = sim.time;
   491→
   492→    if (out !== gndNode) sys.stampVoltageSource(out, gndNode, st.outHigh ? vcc : 0);
   493→    if (!st.ff) {
   494→      if (dis !== gndNode) sys.stampConductance(dis, gndNode, 1 / 50);
   495→    } else {
   496→      if (dis !== gndNode) sys.stampConductance(dis, gndNode, 1e-9);
   497→    }
   498→  },
   499→  step(params, terminals, sim, instance) {
   500→    const vcc = params.vcc as number;
   501→    const thr = terminals.find((t) => t.terminalId === 'thr')!.nodeId;
   502→    const trig = terminals.find((t) => t.terminalId === 'trig')!.nodeId;
   503→    const rst = terminals.find((t) => t.terminalId === 'rst')!.nodeId;
   504→    const ctrl = terminals.find((t) => t.terminalId === 'ctrl')!.nodeId;
   505→    const key = stateKey555(terminals);
   506→    const st = sim.state[key] ?? (sim.state[key] = { ff: false, outHigh: false });
   507→    const vThr = sim.nodeVoltage[thr];
   508→    const vTrig = sim.nodeVoltage[trig];
   509→    const vRst = sim.nodeVoltage[rst];
   510→    const vCtrl = sim.nodeVoltage[ctrl];
   511→    const vThresh = vCtrl > 0.1 ? vCtrl : (2 / 3) * vcc;
   512→    const vTrigThresh = vCtrl > 0.1 ? vCtrl / 2 : (1 / 3) * vcc;
   513→    if (vRst < 0.4) {
   514→      st.ff = false;
   515→    } else {
   516→      if (vThr > vThresh) st.ff = false;
   517→      if (vTrig < vTrigThresh) st.ff = true;
   518→    }
   519→    st.outHigh = st.ff;
   520→  },
   521→  measure(params, terminals, sim) {
   522→    const out = terminals.find((t) => t.terminalId === 'out')!.nodeId;
   523→    const key = stateKey555(terminals);
   524→    const st = sim.state[key];
   525→    return [
   526→      { label: 'Vout', value: sim.nodeVoltage[out].toFixed(3), unit: 'V' },
   527→      { label: 'State', value: st?.outHigh ? 'HIGH' : 'LOW', unit: '' },
   528→    ];
   529→  },
   530→};
   531→
   532→// State key for 555 timer (stable across steps)
   533→function stateKey555(terminals: { terminalId: string; nodeId: number }[]): string {
   534→  return 't555_' + terminals.map(t => `${t.terminalId}=${t.nodeId}`).join('_');
   535→}

   537→// ----- Logic gates (AND, OR, NOT, NAND, NOR, XOR) -----
   538→function makeLogicGate(type: string, name: string, symbol: string, op: (a: boolean, b?: boolean) => boolean): ComponentPlugin {
   539→  return {
   540→    type,
   541→    name,
   542→    category: 'logic',
   543→    description: `${name} logic gate. Drives output to VCC (logic 1) or 0 (logic 0).`,
   544→    symbol,
   545→    boundingBox: { width: 4, height: 3 },
   546→    terminals: type === 'not'
   547→      ? [
   548→          { id: 'a', label: 'A', position: { x: 0, y: 1.5 } },
   549→          { id: 'y', label: 'Y', position: { x: 4, y: 1.5 } },
   550→          { id: 'vcc', label: 'VCC', position: { x: 2, y: 0 } },
   551→          { id: 'gnd', label: 'GND', position: { x: 2, y: 3 } },
   552→        ]
   553→      : [
   554→          { id: 'a', label: 'A', position: { x: 0, y: 1 } },
   555→          { id: 'b', label: 'B', position: { x: 0, y: 2 } },
   556→          { id: 'y', label: 'Y', position: { x: 4, y: 1.5 } },
   557→          { id: 'vcc', label: 'VCC', position: { x: 2, y: 0 } },
   558→          { id: 'gnd', label: 'GND', position: { x: 2, y: 3 } },
   559→        ],
   560→    parameters: [
   561→      { key: 'vcc', label: 'Logic High (VCC)', type: 'number', default: 5, unit: 'V', min: 1, max: 18, step: 0.1 },
   562→      { key: 'threshold', label: 'Switch Threshold', type: 'number', default: 2.5, unit: 'V', min: 0.1, max: 18, step: 0.1 },
   563→    ],
   564→    render(ctx, params, cellSize) {
   565→      // leads
   566→      if (type === 'not') {
   567→        ctx.beginPath();
   568→        ctx.moveTo(0, 1.5 * cellSize);
   569→        ctx.lineTo(cellSize, 1.5 * cellSize);
   570→        ctx.moveTo(3 * cellSize, 1.5 * cellSize);
   571→        ctx.lineTo(4 * cellSize, 1.5 * cellSize);
   572→        ctx.stroke();
   573→        ctx.beginPath();
   574→        ctx.moveTo(cellSize, cellSize * 0.6);
   575→        ctx.lineTo(cellSize, cellSize * 2.4);
   576→        ctx.lineTo(2.7 * cellSize, 1.5 * cellSize);
   577→        ctx.closePath();
   578→        ctx.fillStyle = '#fce7f3';
   579→        ctx.fill();
   580→        ctx.stroke();
   581→        ctx.beginPath();
   582→        ctx.arc(2.85 * cellSize, 1.5 * cellSize, 4, 0, Math.PI * 2);
   583→        ctx.stroke();
   584→      } else {
   585→        ctx.beginPath();
   586→        ctx.moveTo(0, cellSize);
   587→        ctx.lineTo(cellSize, cellSize);
   588→        ctx.moveTo(0, 2 * cellSize);
   589→        ctx.lineTo(cellSize, 2 * cellSize);
   590→        ctx.moveTo(3 * cellSize, 1.5 * cellSize);
   591→        ctx.lineTo(4 * cellSize, 1.5 * cellSize);
   592→        ctx.stroke();
   593→        // D-shape body
   594→        ctx.beginPath();
   595→        ctx.moveTo(cellSize, cellSize * 0.5);
   596→        ctx.lineTo(cellSize, cellSize * 2.5);
   597→        ctx.lineTo(2 * cellSize, cellSize * 2.5);
   598→        ctx.arc(2 * cellSize, 1.5 * cellSize, cellSize, Math.PI / 2, -Math.PI / 2, true);
   599→        ctx.closePath();
   600→        ctx.fillStyle = '#fce7f3';
   601→        ctx.fill();
   602→        ctx.stroke();
   603→        if (type.startsWith('n')) {
   604→          ctx.beginPath();
   605→          ctx.arc(3.1 * cellSize, 1.5 * cellSize, 4, 0, Math.PI * 2);
   606→          ctx.stroke();
   607→        }
   608→      }
   609→      drawLabel(ctx, symbol, 2 * cellSize, 1.5 * cellSize);
   610→    },
   611→    stamp(params, terminals, sys, sim) {
   612→      const vccV = params.vcc as number;
   613→      const thresh = params.threshold as number;
   614→      const a = terminals.find((t) => t.terminalId === 'a')!.nodeId;
   615→      const b = terminals.find((t) => t.terminalId === 'b')?.nodeId;
   616→      const y = terminals.find((t) => t.terminalId === 'y')!.nodeId;
   617→      const gnd = terminals.find((t) => t.terminalId === 'gnd')!.nodeId;
   618→      const key = `gate_${y}`;
   619→      const st = sim.state[key] ?? (sim.state[key] = { out: false });
   620→      const aHigh = sim.nodeVoltage[a] > thresh;
   621→      const bHigh = b !== undefined ? sim.nodeVoltage[b] > thresh : undefined;
   622→      st.out = op(aHigh, bHigh);
   623→      // Drive output as a voltage source (only if Y is wired to a non-ground node)
   624→      if (y !== gnd) sys.stampVoltageSource(y, gnd, st.out ? vccV : 0);
   625→    },
   626→    step(params, terminals, sim) {
   627→      // nothing; stamp does the work each iteration
   628→    },
   629→    measure(params, terminals, sim) {
   630→      const a = terminals.find((t) => t.terminalId === 'a')!.nodeId;
   631→      const b = terminals.find((t) => t.terminalId === 'b')?.nodeId;
   632→      const y = terminals.find((t) => t.terminalId === 'y')!.nodeId;
   633→      return [
   634→        { label: 'A', value: sim.nodeVoltage[a].toFixed(2), unit: 'V' },
   635→        ...(b !== undefined ? [{ label: 'B', value: sim.nodeVoltage[b].toFixed(2), unit: 'V' as const }] : []),
   636→        { label: 'Y', value: sim.nodeVoltage[y].toFixed(2), unit: 'V' },
   637→      ];
   638→    },
   639→  };
   640→}

   643→registerPlugin(makeLogicGate('or', 'OR', '≥1', (a, b) => !!(a || b)));
   644→registerPlugin(makeLogicGate('nand', 'NAND', '&', (a, b) => !(a && b)));
   645→registerPlugin(makeLogicGate('nor', 'NOR', '≥1', (a, b) => !(a || b)));
   646→registerPlugin(makeLogicGate('xor', 'XOR', '=1', (a, b) => !!(a !== b)));
   647→registerPlugin(makeLogicGate('not', 'NOT', '1', (a) => !a));
