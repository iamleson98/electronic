    52→  const setRunning = useEditor((s) => s.setRunning);
    53→  const setSpeed = useEditor((s) => s.setSpeed);

    55→  const step = useEditor((s) => s.step);
    56→  const reset = useEditor((s) => s.reset);
    57→
    58→  // Toast sim errors when they occur

    60→    if (simError) {

    62→    }

    64→  const clear = useEditor((s) => s.clear);
    65→  const undo = useEditor((s) => s.undo);
    66→  const redo = useEditor((s) => s.redo);
    67→  const serialize = useEditor((s) => s.serialize);
    68→  const loadDocument = useEditor((s) => s.loadDocument);
    69→  // Hierarchical sheet navigation
    70→  const activeSheet = useEditor((s) => s.activeSheet);
    71→  const setActiveSheet = useEditor((s) => s.setActiveSheet);
    72→  const sheets = useEditor((s) => s.sheets);
    73→  const childSheets = useEditor((s) => s.childSheets);
    74→  const showGrid = useEditor((s) => s.showGrid);
    75→  const setShowGrid = useEditor((s) => s.setShowGrid);
    76→  const snapToGrid = useEditor((s) => s.snapToGrid);
    77→  const setSnapToGrid = useEditor((s) => s.setSnapToGrid);
    78→  const past = useEditor((s) => s.past.length);
    79→  const future = useEditor((s) => s.future.length);
    80→  // KiCad-parity new state
    81→  const units = useEditor((s) => s.units);
    82→  const setUnits = useEditor((s) => s.setUnits);
    83→  const showPinNumbers = useEditor((s) => s.showPinNumbers);
    84→  const setShowPinNumbers = useEditor((s) => s.setShowPinNumbers);
    85→  const showPinNames = useEditor((s) => s.showPinNames);
    86→  const setShowPinNames = useEditor((s) => s.setShowPinNames);
    87→  const showPinElecTypes = useEditor((s) => s.showPinElecTypes);
    88→  const setShowPinElecTypes = useEditor((s) => s.setShowPinElecTypes);
    89→  const showRefdes = useEditor((s) => s.showRefdes);
    90→  const setShowRefdes = useEditor((s) => s.setShowRefdes);
    91→  const showValues = useEditor((s) => s.showValues);
    92→  const setShowValues = useEditor((s) => s.setShowValues);
    93→  const activeTool = useEditor((s) => s.activeTool);
    94→  const setActiveTool = useEditor((s) => s.setActiveTool);
    95→
    96→  const fileInputRef = useRef<HTMLInputElement | null>(null);
    97→  const schImportInputRef = useRef<HTMLInputElement | null>(null);
    98→  const [showMyCircuits, setShowMyCircuits] = useState(false);
    99→  const [showSpiceImport, setShowSpiceImport] = useState(false);
   100→  const [showSubCircuit, setShowSubCircuit] = useState(false);
   101→  const [showSymbolEditor, setShowSymbolEditor] = useState(false);
   102→  const [showFindReplace, setShowFindReplace] = useState(false);
   103→  const [showViolations, setShowViolations] = useState(false);
   104→  const [showNetInspector, setShowNetInspector] = useState(false);
   105→  const [showPageSetup, setShowPageSetup] = useState(false);
   106→  const [showSavedViews, setShowSavedViews] = useState(false);
   107→  const [showSheets, setShowSheets] = useState(false);
   108→  const [showNetClasses, setShowNetClasses] = useState(false);
   109→  const [showSettings, setShowSettings] = useState(false);
   110→  // Simulation analysis dialogs
   111→  const [showAnalysis, setShowAnalysis] = useState(false);
   112→  const [showOptions, setShowOptions] = useState(false);
   113→  const [showMeasurement, setShowMeasurement] = useState(false);
   114→  const [showBatchSweep, setShowBatchSweep] = useState(false);
   115→  const [showStimuliEditor, setShowStimuliEditor] = useState(false);
   116→  // Track camera so SaveViewDialog gets current pan/zoom
   117→  const [camera, setCamera] = useState({ x: 0, y: 0, zoom: 1 });
   118→
   119→  // Listen for "open find/replace" event from canvas hotkey
   120→  useEffect(() => {
   121→    const handler = () => setShowFindReplace(true);
   122→    window.addEventListener('circuitlab:open-find-replace', handler);
   123→    // poll camera from canvas via custom event
   124→    const camHandler = (e: Event) => {
   125→      const detail = (e as CustomEvent).detail;
   126→      if (detail) setCamera(detail);
   127→    };
   128→    window.addEventListener('circuitlab:camera', camHandler);
   129→    // ask canvas for current camera
   130→    const askCamera = () => window.dispatchEvent(new CustomEvent('circuitlab:request-camera'));
   131→    askCamera();
   132→    const interval = setInterval(askCamera, 2000);
   133→    return () => {
   134→      window.removeEventListener('circuitlab:open-find-replace', handler);
   135→      window.removeEventListener('circuitlab:camera', camHandler);
   136→      clearInterval(interval);
   137→    };
   138→  }, []);
   139→
   140→  const handleSave = useCallback(() => {
   141→    const doc = serialize();
   142→    const blob = new Blob([JSON.stringify(doc, null, 2)], { type: 'application/json' });
   143→    const url = URL.createObjectURL(blob);
   144→    const a = document.createElement('a');
   145→    a.href = url;
   146→    a.download = `circuit_${Date.now()}.json`;
   147→    a.click();
   148→    URL.revokeObjectURL(url);
   149→  }, [serialize]);
   150→
   151→  const handleLoad = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
   152→    const file = e.target.files?.[0];
   153→    if (!file) return;
   154→    const reader = new FileReader();
   155→    reader.onload = () => {
   156→      try {
   157→        const doc = JSON.parse(reader.result as string);
   158→        if (doc && doc.version === 1 && Array.isArray(doc.components) && Array.isArray(doc.wires)) {
   159→          loadDocument(doc);
   160→          toast.success(`Loaded circuit: ${file.name}`);
   161→        } else {
   162→          toast.error('Invalid circuit file: missing version, components, or wires array');
   163→        }
   164→      } catch (err) {
   165→        toast.error('Failed to parse file: ' + (err as Error).message);
   166→      }
   167→    };
   168→    reader.readAsText(file);
   169→    e.target.value = '';
   170→  }, [loadDocument]);
   171→
   172→  // KiCad / Eagle schematic import
   173→  const handleSchImport = useCallback(async (e: React.ChangeEvent<HTMLInputElement>) => {
   174→    const file = e.target.files?.[0];
   175→    if (!file) return;
   176→    const reader = new FileReader();
   177→    reader.onload = () => {
   178→      try {
   179→        const text = reader.result as string;
   180→        const result = parseSchematicFile(text, file.name);
   181→        loadDocument(result.document);
   182→        if (result.unknownSymbols.length > 0) {
   183→          toast.warning(`Imported with ${result.unknownSymbols.length} unknown symbols skipped`, {
   184→            description: result.unknownSymbols.slice(0, 5).join('\n'),
   185→          });
   186→        } else {
   187→          toast.success(`Imported ${file.name}`);
   188→        }
   189→        if (result.warnings.length > 0) {
   190→          console.log('Import warnings:', result.warnings);
   191→        }
   192→      } catch (err) {
   193→        toast.error('Import failed: ' + (err as Error).message);
   194→      }
   195→    };
   196→    reader.readAsText(file);
   197→    e.target.value = '';
   198→  }, [loadDocument]);
   199→
   200→  const handlePlotSVG = useCallback(() => {
   201→    const doc = serialize();
   202→    const svg = exportSchematicSVG(doc);
   203→    downloadText(svg, `schematic_${Date.now()}.svg`, 'image/svg+xml');
   204→    toast.success('Exported SVG');
   205→  }, [serialize]);
   206→
   207→  const handlePlotPNG = useCallback(async () => {
   208→    try {
   209→      const doc = serialize();
   210→      const blob = await exportSchematicPNG(doc, 2);
   211→      downloadBlob(blob, `schematic_${Date.now()}.png`);
   212→      toast.success('Exported PNG');
   213→    } catch (err) {
   214→      toast.error('PNG export failed: ' + (err as Error).message);
   215→    }
   216→  }, [serialize]);
   217→
   218→  const handlePlotPDF = useCallback(async () => {
   219→    try {
   220→      const doc = serialize();
   221→      const blob = await exportSchematicPDF(doc);
   222→      downloadBlob(blob, `schematic_${Date.now()}.pdf`);
   223→      toast.success('Exported PDF');
   224→    } catch (err) {
   225→      toast.error('PDF export failed: ' + (err as Error).message);
   226→    }
   227→  }, [serialize]);
   228→
   229→  const handleExportSPICE = useCallback(() => {
   230→    const doc = serialize();
   231→    const net = exportSPICENetlist(doc, doc.metadata?.title ?? 'Circuit');
   232→    downloadText(net, `circuit_${Date.now()}.cir`, 'text/plain');
   233→    toast.success('Exported SPICE netlist');
   234→  }, [serialize]);
   235→
   236→  const handleExportKiCadNet = useCallback(() => {
   237→    const doc = serialize();
   238→    const net = exportKiCadNetlist(doc, doc.metadata?.title ?? 'Circuit');
   239→    downloadText(net, `circuit_${Date.now()}.net`, 'application/xml');
   240→    toast.success('Exported KiCad netlist');
   241→  }, [serialize]);
   242→
   243→  const handleExportBOM = useCallback((format: 'csv' | 'html' | 'xml') => {
   244→    const doc = serialize();
   245→    const ext = format;
   246→    if (format === 'csv') {
   247→      downloadText(exportBOMCSV(doc), `bom_${Date.now()}.csv`, 'text/csv');
   248→    } else if (format === 'html') {
   249→      downloadText(exportBOMHTML(doc), `bom_${Date.now()}.html`, 'text/html');
   250→    } else {
   251→      downloadText(exportBOMXML(doc), `bom_${Date.now()}.xml`, 'application/xml');
   252→    }
   253→    toast.success(`Exported BOM (${format.toUpperCase()})`);
   254→  }, [serialize]);
   255→
   256→  return (

   261→          <div className="flex h-7 w-7 items-center justify-center rounded bg-gradient-to-br from-cyan-400 to-emerald-500 text-slate-900">
   262→            <Zap size={16} strokeWidth={2.5} />
   263→          </div>

   265→        </div>

   269→          <div className="mr-2 flex items-center gap-1 rounded bg-slate-800/60 px-2 py-1 text-xs font-mono">

   272→              onClick={() => setActiveSheet('')}
   273→              title="Back to root sheet (Esc)"
   274→            >
   275→              Root
   276→            </button>
   277→            <ChevronRight size={12} className="text-slate-500" />
   278→            <span className="text-emerald-300">
   279→              {sheets.find((s) => s.fileName === activeSheet)?.sheetName ??
   280→                childSheets[activeSheet]?.sheets?.find?.((s: any) => s.fileName === activeSheet)?.sheetName ??
   281→                activeSheet.replace(/\.kicad_sch$/, '')}
   282→            </span>
   283→          </div>
   284→        )}
   285→
   286→        {/* Run / Pause / Step */}
   287→        <Tooltip>
   288→          <TooltipTrigger asChild>
   289→            <Button
   290→              size="sm"
   291→              variant={running ? 'destructive' : 'default'}
   292→              className={running ? '' : 'bg-emerald-500 text-slate-900 hover:bg-emerald-400'}

   298→          </TooltipTrigger>

   309→        <Tooltip>
   310→          <TooltipTrigger asChild>
   311→            <Button size="sm" variant="ghost" onClick={() => reset()} disabled={running}>

   314→          </TooltipTrigger>

   318→        <div className="mx-1 h-5 w-px bg-slate-700" />

   320→        {/* Speed control */}

   322→          <Gauge size={14} className="text-slate-400" />
   323→          <span className="hidden text-xs text-slate-400 sm:inline">Speed</span>
   324→          <Slider
   325→            value={[Math.log2(speed)]}
   326→            min={-2}
   327→            max={6}
   328→            step={0.5}
   329→            onValueChange={(v) => setSpeed(Math.pow(2, v[0]))}
   330→            className="w-28"
   331→          />
   332→          <span className="w-12 text-right font-mono text-xs text-slate-300">{speed.toFixed(1)}x</span>
   333→        </div>
   334→
   335→        {/* Timestep control */}
   336→        <div className="flex items-center gap-2 px-1">
   337→          <span className="hidden text-xs text-slate-400 lg:inline">dt</span>
   338→          <Slider
   339→            value={[Math.log10(dt)]}
   340→            min={-7}
   341→            max={-2}
   342→            step={0.5}
   343→            onValueChange={(v) => setDt(Math.pow(10, v[0]))}
   344→            className="w-24"
   345→          />
   346→          <span className="w-16 text-right font-mono text-xs text-slate-300">
   347→            {dt >= 1e-3 ? `${(dt * 1e3).toFixed(1)}ms` : `${(dt * 1e6).toFixed(1)}µs`}
   348→          </span>
   349→        </div>
   350→
   351→        <div className="mx-1 h-5 w-px bg-slate-700" />
   352→
   353→        {/* Undo / Redo */}
   354→        <Tooltip>
   355→          <TooltipTrigger asChild>
   356→            <Button size="sm" variant="ghost" onClick={() => undo()} disabled={past === 0 || running}>
   357→              <Undo2 size={14} />
   358→            </Button>
   359→          </TooltipTrigger>
   360→          <TooltipContent>{running ? 'Pause to undo' : 'Undo (Ctrl+Z)'}</TooltipContent>
   361→        </Tooltip>
   362→        <Tooltip>
   363→          <TooltipTrigger asChild>
   364→            <Button size="sm" variant="ghost" onClick={() => redo()} disabled={future === 0 || running}>
   365→              <Redo2 size={14} />
   366→            </Button>
   367→          </TooltipTrigger>
   368→          <TooltipContent>{running ? 'Pause to redo' : 'Redo (Ctrl+Y)'}</TooltipContent>
   369→        </Tooltip>
   370→
   371→        <div className="mx-1 h-5 w-px bg-slate-700" />
   372→
   373→        {/* Examples */}
   374→        <DropdownMenu>
   375→          <DropdownMenuTrigger asChild>
   376→            <Button size="sm" variant="ghost" disabled={running}>
   377→              <FileText size={14} />
   378→              <span className="ml-1 hidden md:inline">Examples</span>
   379→              <ChevronDown size={12} className="ml-1" />
   380→            </Button>
   381→          </DropdownMenuTrigger>
   382→          <DropdownMenuContent align="start" className="w-72 bg-slate-900 border-slate-700 max-h-[70vh] overflow-y-auto">
   383→            <DropdownMenuLabel className="text-slate-300">Load Example Circuit</DropdownMenuLabel>
   384→            <DropdownMenuSeparator className="bg-slate-700" />
   385→            {exampleCategories.map((cat) => (
   386→              <DropdownMenuGroup key={cat.label}>
   387→                <DropdownMenuLabel className="text-xs text-cyan-400 font-semibold uppercase tracking-wide px-2 pt-3 pb-1">
   388→                  {cat.label}

   396→                    <span className="text-sm font-medium">{ex.name}</span>
   397→                    <span className="text-xs text-slate-400">{ex.description}</span>
   398→                  </DropdownMenuItem>
   399→                ))}
   400→              </DropdownMenuGroup>
   401→            ))}
   402→          </DropdownMenuContent>
   403→        </DropdownMenu>
   404→
   405→        {/* Database (My Circuits) */}
   406→        <Tooltip>
   407→          <TooltipTrigger asChild>
   408→            <Button size="sm" variant="ghost" onClick={() => setShowMyCircuits(true)} disabled={running}>
   409→              <Database size={14} />
   410→              <span className="ml-1 hidden lg:inline">My Circuits</span>
   411→            </Button>
   412→          </TooltipTrigger>
   413→          <TooltipContent>{running ? 'Pause simulation to manage saved circuits' : 'Open the saved circuits library (database)'}</TooltipContent>
   414→        </Tooltip>
   415→
   416→        {/* SPICE Import */}
   417→        <Tooltip>
   418→          <TooltipTrigger asChild>
   419→            <Button size="sm" variant="ghost" onClick={() => setShowSpiceImport(true)} disabled={running}>
   420→              <FileCode size={14} />
   421→              <span className="ml-1 hidden lg:inline">SPICE</span>
   422→            </Button>
   423→          </TooltipTrigger>
   424→          <TooltipContent>{running ? 'Pause simulation to import SPICE' : 'Import a SPICE netlist (.cir / .net)'}</TooltipContent>
   425→        </Tooltip>
   426→
   427→        {/* Sub-Circuit builder */}
   428→        <Tooltip>
   429→          <TooltipTrigger asChild>
   430→            <Button size="sm" variant="ghost" onClick={() => setShowSubCircuit(true)} disabled={running}>
   431→              <Boxes size={14} />
   432→              <span className="ml-1 hidden lg:inline">Sub-Circuit</span>
   433→            </Button>
   434→          </TooltipTrigger>
   435→          <TooltipContent>{running ? 'Pause simulation to create sub-circuits' : 'Convert current circuit into a reusable component'}</TooltipContent>
   436→        </Tooltip>
   437→
   438→        {/* Symbol Editor — WYSIWYG symbol designer */}
   439→        <Tooltip>
   440→          <TooltipTrigger asChild>
   441→            <Button size="sm" variant="ghost" onClick={() => setShowSymbolEditor(true)} disabled={running}>
   442→              <Pencil size={14} />
   443→              <span className="ml-1 hidden lg:inline">Symbol Editor</span>
   444→            </Button>
   445→          </TooltipTrigger>
   446→          <TooltipContent>{running ? 'Pause simulation to design symbols' : 'Open the WYSIWYG symbol editor to design new component symbols'}</TooltipContent>
   447→        </Tooltip>
   448→
   449→        {/* ERC (Electrical Rule Check) */}
   450→        <Tooltip>
   451→          <TooltipTrigger asChild>
   452→            <Button size="sm" variant="ghost" onClick={() => {
   453→              const result = useEditor.getState().runERC();
   454→              if (result.passed) {
   455→                toast.success(`ERC passed — ${result.stats.warnings} warning(s)`);
   456→              } else {
   457→                toast.warning(`ERC: ${result.stats.errors} error(s), ${result.stats.warnings} warning(s)`);
   458→              }
   459→            }} disabled={running}>
   460→              <ShieldCheck size={14} />
   461→              <span className="ml-1 hidden lg:inline">ERC</span>
   462→            </Button>
   463→          </TooltipTrigger>
   464→          <TooltipContent>Electrical Rule Check — find unconnected pins, power shorts, conflicting drivers</TooltipContent>
   465→        </Tooltip>
   466→
   467→        {/* Find */}
   468→        <Tooltip>
   469→          <TooltipTrigger asChild>
   470→            <Button size="sm" variant="ghost" onClick={() => setShowFindReplace(true)} disabled={running}>
   471→              <Search size={14} />
   472→              <span className="ml-1 hidden lg:inline">Find</span>
   473→            </Button>
   474→          </TooltipTrigger>
   475→          <TooltipContent>Find / Replace (Ctrl+F)</TooltipContent>
   476→        </Tooltip>
   477→
   478→        {/* Simulate menu — KiCad/ngspice parity */}
   479→        <DropdownMenu>
   480→          <DropdownMenuTrigger asChild>
   481→            <Button size="sm" variant="ghost">
   482→              <Activity size={14} />
   483→              <span className="ml-1 hidden lg:inline">Simulate</span>
