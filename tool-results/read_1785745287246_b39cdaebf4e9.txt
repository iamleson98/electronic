     1→// Pre-built example circuits — clean, professional layouts with no overlapping wires.
     2→// Each circuit is laid out left-to-right with proper spacing so:
     3→//   - Wires go in clean L-shapes (horizontal then vertical, or vice versa)
     4→//   - No wire crosses through a component body
     5→//   - Ground is at the bottom, sources on the left, loads in the middle/right
     6→//   - Components are spaced at least 4 grid units apart
     7→
     8→import type { CircuitDocument } from './types';
     9→import { getPlugin } from './registry';
    10→import './components';
    11→
    12→function defaultsFor(type: string): Record<string, number | string | boolean> {
    13→  const plugin = getPlugin(type);
    14→  if (!plugin) return {};
    15→  const defaults: Record<string, number | string | boolean> = {};
    16→  for (const p of plugin.parameters) defaults[p.key] = p.default;
    17→  return defaults;
    18→}
    19→
    20→function comp(
    21→  type: string,
    22→  id: string,
    23→  pos: [number, number],
    24→  rotation: 0 | 1 | 2 | 3 = 0,
    25→  params?: Record<string, number | string | boolean>,
    26→) {
    27→  return {
    28→    id,
    29→    type,
    30→    position: { x: pos[0], y: pos[1] },
    31→    rotation,
    32→    parameters: { ...defaultsFor(type), ...params },
    33→  };
    34→}
    35→
    36→function wire(id: string, fromC: string, fromT: string, toC: string, toT: string, waypoints?: [number, number][]) {
    37→  const w: any = { id, from: { componentId: fromC, terminalId: fromT }, to: { componentId: toC, terminalId: toT } };
    38→  if (waypoints && waypoints.length > 0) {
    39→    w.waypoints = waypoints.map(([x, y]) => ({ x, y }));
    40→  }
    41→  return w;
    42→}

   432→// ----- Example 9: Digital Clock (HH:MM:SS) -----
   433→// A 6-digit digital clock using CD4026 decade counter ICs.
   434→//
   435→// Architecture:
   436→//   1Hz pulse source (crystal) → sec-ones CD4026 → CO → sec-tens → CO →
   437→//   min-ones → CO → min-tens → CO → hr-ones → CO → hr-tens
   438→//
   439→// Each CD4026 counts on the rising edge of its CLK input and drives a 7-segment
   440→// display directly via its a-g outputs. The carry-out (CO) goes HIGH for the
   441→// first half of the count cycle, producing a rising edge on the next CD4026's
   442→// CLK when this counter wraps to 0.
   443→//
   444→// maxCount settings:
   445→//   - ones digits: maxCount=10 (counts 0-9)
   446→//   - tens digits: maxCount=6 (counts 0-5, for seconds/minutes tens)
   447→//   - hours tens: maxCount=3 (counts 0-2, for 24h format)
   448→//
   449→// Note: For a true 24-hour reset (00:00:00 at 24:00:00), add an AND gate that
   450→// detects hr-tens=2 AND hr-ones=4 and drives RST on both hour counters.
   451→// This example omits that for simplicity — it counts 00:00:00 to 29:59:59.
   452→//
   453→// Layout: 6 digit columns, each with a CD4026 (top) and 7-seg (bottom).
   454→//   x=2: hr-tens, x=10: hr-ones, x=20: min-tens, x=28: min-ones,
   455→//   x=38: sec-tens, x=46: sec-ones
   456→//   Crystal at far right, power and ground at bottom.
   457→function digitSegWires(prefix: string, cx: number, segId: string, segCx: number) {
   458→  // Generate 7 segment wires from CD4026 (at cx, y=2) to 7-seg (at segCx, y=8)
   459→  // CD4026 segment outputs at (cx+0..6, 6)
   460→  // 7-seg segment inputs at various positions
   461→  const segPositions: Record<string, [number, number]> = {
   462→    a: [segCx + 0, 8],   // top-left
   463→    b: [segCx + 2, 8],   // top-mid
   464→    c: [segCx + 4, 8],   // top-right
   465→    d: [segCx + 4, 14],  // bot-right
   466→    e: [segCx + 2, 14],  // bot-mid
   467→    f: [segCx + 0, 14],  // bot-left
   468→    g: [segCx + 0, 11],  // mid-left
   469→  };
   470→  const cd4026SegY = 6; // bottom of CD4026
   471→  const segIds = ['a', 'b', 'c', 'd', 'e', 'f', 'g'];
   472→  return segIds.map((s, i) => {
   473→    const [sx, sy] = segPositions[s];
   474→    // Route from (cx+i, 6) to (sx, sy)
   475→    // Simple L-path: go down from CD4026 output to a routing channel, then to the segment
   476→    const waypoints: [number, number][] = [];
   477→    if (sy <= 8) {
   478→      // Top-row segments (a, b, c): route directly down
   479→      waypoints.push([cx + i, 7], [sx, 7]);
   480→    } else {
   481→      // Bottom/side segments (d, e, f, g): route around the 7-seg body
   482→      // Go down to y=7, then horizontal to sx, then down to sy
   483→      waypoints.push([cx + i, 7], [sx, 7]);
   484→    }
   485→    return wire(`${prefix}_${s}`, `ic_${prefix}`, s, `seg_${prefix}`, s, waypoints);
   486→  });
   487→}

   790→// is set by 7 consecutive pin assignments (one per segment).

   796→  // pins = [a, b, c, d, e, f, g] pin names

   939→// ----- Example 10: Simple Seconds Counter (0-99) -----
   940→// A minimalist 2-digit counter using only 2 CD4026 chips (vs 6 in the full clock).
   941→// Architecture:
   942→//   1Hz crystal → CD4026 ones (0-9) → CO → CD4026 tens (0-9) → 7-seg displays
   943→//
   944→// The ones counter increments on each clock pulse. When it wraps from 9→0,
   945→// its CO output produces a rising edge that clocks the tens counter. Both
   946→// counters use maxCount=10, giving a 00-99 range (counts 00→01→...→99→00).
   947→//
   948→// Total: 7 components, ~25 wires (vs 15 components, 74 wires for the 6-digit clock).
   949→export const exampleSimpleClock: CircuitDocument = {
   950→  version: 1,
   951→  components: [
   952→    // 1Hz crystal oscillator — drives the ones counter
   953→    comp('pulseSource', 'xtal', [32, 2], 0, { high: 5, low: 0, frequency: 1, duty: 50 }),
   954→    // 5V power supply
   955→    comp('dcVoltage', 'vcc1', [32, 8], 0, { voltage: 5 }),
   956→    comp('ground', 'gnd1', [16, 18], 0, {}),
   957→    // CD4026 #1 — ones digit (counts 0-9)
   958→    comp('cd4026', 'ic_so', [10, 2], 0, { maxCount: 10, vcc: 5 }),
   959→    // CD4026 #2 — tens digit (counts 0-9, carries from ones)
   960→    comp('cd4026', 'ic_st', [22, 2], 0, { maxCount: 10, vcc: 5 }),
   961→    // 7-segment displays (green) — offset by 1 from CD4026 to center segments
   962→    comp('sevenSegment', 'seg_so', [11, 8], 0, { color: 'green', threshold: 2.0 }),
   963→    comp('sevenSegment', 'seg_st', [23, 8], 0, { color: 'green', threshold: 2.0 }),
   964→  ],
   965→  wires: [
   966→    // Crystal → ones counter CLK
   967→    wire('clk_xtal_so', 'xtal', 'p', 'ic_so', 'clk', [[33, 1], [10, 1]]),
   968→    wire('xtal_gnd', 'xtal', 'n', 'gnd1', 'g', [[33, 18]]),
   969→    // Carry chain: ones CO → tens CLK
   970→    wire('co_so_st', 'ic_so', 'co', 'ic_st', 'clk', [[16, 1], [22, 1]]),
   971→    // VCC for both CD4026s
   972→    wire('vcc_so', 'vcc1', 'p', 'ic_so', 'vcc', [[33, 3], [11, 3]]),
   973→    wire('vcc_st', 'vcc1', 'p', 'ic_st', 'vcc', [[33, 3], [23, 3]]),
   974→    wire('vcc_gnd', 'vcc1', 'n', 'gnd1', 'g', [[33, 18]]),
   975→    // GND for both CD4026s
   976→    wire('gnd_so', 'ic_so', 'gnd', 'gnd1', 'g', [[15, 18]]),
   977→    wire('gnd_st', 'ic_st', 'gnd', 'gnd1', 'g', [[27, 18]]),
   978→    // RST for both CD4026s (tied to ground — no reset)
   979→    wire('rst_so', 'ic_so', 'rst', 'gnd1', 'g', [[10, 18]]),
   980→    wire('rst_st', 'ic_st', 'rst', 'gnd1', 'g', [[22, 18]]),
   981→    // 7-seg COM → ground (com terminal at x=segCx+4, y=11)
   982→    wire('com_so', 'seg_so', 'com', 'gnd1', 'g', [[15, 11], [15, 18]]),
   983→    wire('com_st', 'seg_st', 'com', 'gnd1', 'g', [[27, 11], [27, 18]]),
   984→    // Segment wires (7 per digit × 2 digits = 14 wires)
   985→    ...digitSegWires('so', 10, 'seg_so', 11),
   986→    ...digitSegWires('st', 22, 'seg_st', 23),
   987→  ],
