   276→   276→/** Get the reference designator prefix for a component type */
   277→   277→function refdesPrefix(type: string): string {
   278→   278→  switch (type) {
   279→   279→    case 'resistor': return 'R';
   280→   280→    case 'capacitor': return 'C';
   281→   281→    case 'inductor': return 'L';
   282→   282→    case 'led': return 'LED';
   283→   283→    case 'diode': return 'D';
   284→   284→    case 'zener': return 'DZ';
   285→   285→    case 'schottky': return 'D';
   286→   286→    case 'dcVoltage':
   287→   287→    case 'acVoltage':
   288→   288→    case 'pulseSource': return 'V';
   289→   289→    case 'currentSource': return 'I';
   290→   290→    case 'timer555':
   291→   291→    case 'opamp':
   292→   292→    case 'opampRails':
   293→   293→    case 'voltageRegulator':
   294→   294→    case 'vco': return 'U';
   295→   295→    case 'npn':
   296→   296→    case 'pnp':
   297→   297→    case 'nmos':
   298→   298→    case 'pmos': return 'Q';
   299→   299→    case 'switch':
   300→   300→    case 'pushButton': return 'SW';
   301→   301→    case 'fuse': return 'F';
   302→   302→    case 'crystal': return 'Y';
   303→   303→    case 'transformer': return 'T';
   304→   304→    case 'speaker':
   305→   305→    case 'buzzer': return 'LS';
   306→   306→    case 'dcMotor': return 'M';
   307→   307→    case 'photoresistor': return 'LDR';
   308→   308→    case 'sevenSegment': return 'DSP';
   309→   309→    case 'ground':
   310→   310→    case 'powerGND': return 'GND';
   311→   311→    case 'junction': return 'J';
   312→   312→    case 'oscilloscope':
   313→   313→    case 'voltmeter':
   314→   314→    case 'ammeter': return 'TP';
   315→   315→    case 'arduino':
   316→   316→    case 'arduinoReal':
   317→   317→    case 'raspberryPi': return 'U';
   318→   318→    case 'netLabel':
   319→   319→    case 'busLabel': return 'NL';
   320→   320→    case 'bus': return 'BUS';
   321→   321→    default:
   322→   322→      if (type.startsWith('7400_') || type.startsWith('7402_') || type.startsWith('7404_') ||
   323→   323→          type.startsWith('7408_') || type.startsWith('7432_') || type.startsWith('7486_') ||
   324→   324→          type.startsWith('7474_')) return 'U';
   325→   325→      if (type.startsWith('custom_')) return 'U';
   326→   326→      return 'U';
   327→   327→  }
   328→   328→}

   330→   330→/** Generate the next refdes for a given type, based on existing components */
   331→   331→function nextRefdes(type: string, components: CircuitComponent[]): string {
   332→   332→  const prefix = refdesPrefix(type);
   333→   333→  let maxNum = 0;
   334→   334→  for (const c of components) {
   335→   335→    if (refdesPrefix(c.type) === prefix) {
   336→   336→      const ref = c.refdes ?? c.id;
   337→   337→      const match = ref.match(/(\d+)$/);
   338→   338→      if (match) {
   339→   339→        const n = parseInt(match[1]);
   340→   340→        if (n > maxNum) maxNum = n;
   341→   341→      }
   342→   342→    }
   343→   343→  }
   344→   344→  return `${prefix}${maxNum + 1}`;
   345→   345→}

   347→   347→function snapshot(s: {
   348→   348→  components: CircuitComponent[];
   349→   349→  wires: Wire[];
   350→   350→  drawings?: DrawingPrimitive[];
   351→   351→  noConnects?: NoConnectMarker[];
   352→   352→  groups?: Group[];
   353→   353→  sheets?: HierarchicalSheet[];
   354→   354→  netClasses?: NetClass[];
   355→   355→}) {
   356→   356→  return {
   357→   357→    components: s.components.map((c) => ({ ...c, parameters: { ...c.parameters }, simState: undefined, fields: c.fields ? c.fields.map((f) => ({ ...f })) : undefined })),
   358→   358→    wires: s.wires.map((w) => ({ ...w })),
   359→   359→    drawings: s.drawings ? s.drawings.map((d) => ({ ...d })) : [],
   360→   360→    noConnects: s.noConnects ? s.noConnects.map((n) => ({ ...n })) : [],
   361→   361→    groups: s.groups ? s.groups.map((g) => ({ ...g, componentIds: [...g.componentIds], wireIds: [...g.wireIds], drawingIds: [...g.drawingIds] })) : [],
   362→   362→    sheets: s.sheets ? s.sheets.map((sh) => ({ ...sh, pins: sh.pins.map((p) => ({ ...p })) })) : [],
   363→   363→    netClasses: s.netClasses ? s.netClasses.map((nc) => ({ ...nc, nets: [...nc.nets] })) : [],
   364→   364→  };
   365→   365→}

  1296→  1296→  setSpeed: (speed) => set({ speed }),

  1299→  1299→  step: () => {
  1300→  1300→    const s = get();
  1301→  1301→    if (s.components.length === 0 && s.sheets.length === 0) return;
  1302→  1302→
  1303→  1303→    try {
  1304→  1304→
  1305→  1305→    // ── Cross-sheet simulation ────────────────────────────────────────────
  1306→  1306→    // If we're inside a sub-sheet OR the root has sub-sheets, we need to
  1307→  1307→    // flatten the hierarchy into a single (components, wires) pair before
  1308→  1308→    // simulating. We do this on every step so sub-sheet edits are picked up
  1309→  1309→    // immediately. The flattened components have prefixed IDs (e.g. "amp.R1")
  1310→  1310→    // so they don't collide with the active sheet's IDs.

  1312→  1312→    let simWires = s.wires;
  1313→  1313→    let usingHierarchy = false;
  1314→  1314→    if (s.activeSheet || s.sheets.length > 0) {
  1315→  1315→      usingHierarchy = true;

  1317→  1317→      let childSheets = s.childSheets;
  1318→  1318→      if (s.activeSheet) {
  1319→  1319→        childSheets = {
  1320→  1320→          ...childSheets,
  1321→  1321→          [s.activeSheet]: snapshotSheet(s.components, s.wires, s.sheets),
  1322→  1322→        };
  1323→  1323→      }
  1324→  1324→      // Build root document from either the active root state or the saved __root__
  1325→  1325→      const rootDoc = s.activeSheet

  1327→  1327→        : { version: 1 as const, components: s.components, wires: s.wires, sheets: s.sheets };

  1329→  1329→      simComponents = flat.components;
  1330→  1330→      simWires = flat.wires;
  1331→  1331→    }

  1336→  1336→      if (p) plugins.set(c.type, p);
  1337→  1337→    }

  1339→  1339→    const persistentState = s.simContext?.state ?? {};
  1340→  1340→    const prev = s.simContext
  1341→  1341→      ? {
  1342→  1342→          nodeVoltage: s.simContext.nodeVoltage,
  1343→  1343→          branchCurrent: s.simContext.branchCurrent,
  1344→  1344→          time: s.simContext.time,
  1345→  1345→          state: persistentState,
  1346→  1346→        }
  1347→  1347→      : { nodeVoltage: new Float64Array(0), branchCurrent: new Float64Array(0), time: 0, state: persistentState };

  1349→  1349→    // For speed >= 1: run floor(speed) sub-steps per frame (e.g., 4x = 4 steps)
  1350→  1350→    // For speed < 1: run 1 step every Nth frame, where N = ceil(1/speed)
  1351→  1351→    //   This is tracked via simStepCounter in the store state
  1352→  1352→    const speed = s.speed;
  1353→  1353→    let subSteps: number;
  1354→  1354→    if (speed >= 1) {
  1355→  1355→      subSteps = Math.max(1, Math.floor(speed));
  1356→  1356→    } else {
  1357→  1357→      // Sub-real-time: skip steps to slow down the simulation
  1358→  1358→      const skipInterval = Math.ceil(1 / speed); // e.g., 0.5x → skip every 2nd frame
  1359→  1359→      const stepCount = (s as any).__stepCount ?? 0;
  1360→  1360→      (s as any).__stepCount = stepCount + 1;
  1361→  1361→      if (stepCount % skipInterval !== 0) {
  1362→  1362→        return; // skip this frame to slow down
  1363→  1363→      }
  1364→  1364→      subSteps = 1;
  1365→  1365→    }
  1366→  1366→    const dt = s.dt;
  1367→  1367→    // Build simOptions from the editor's settings (temperature, .IC, .NODESET)
  1368→  1368→    const simOpts = {
  1369→  1369→      initialConditions: s.simOptions?.initialConditions,
  1370→  1370→      nodeSets: s.simOptions?.nodeSets,
  1371→  1371→    };
  1372→  1372→    let result: { sim: SimContext; branchCurrentSize: number; nodeMap: any } | null = null;
  1373→  1373→    for (let i = 0; i < subSteps; i++) {
  1374→  1374→      result = simulateStep(simComponents, simWires, plugins, prev, dt, simOpts);
  1375→  1375→      if (!result) break;
  1376→  1376→      prev.nodeVoltage = result.sim.nodeVoltage;
  1377→  1377→      prev.branchCurrent = result.sim.branchCurrent;
  1378→  1378→      prev.time = result.sim.time;
  1379→  1379→      prev.state = result.sim.state;
  1380→  1380→
  1381→  1381→      // Digital fast-forward: advance sim.time toward the next clock edge.
  1382→  1382→      // Handles two types of clock sources:
  1383→  1383→      //   1. pulseSource (crystal) — when no capacitors/inductors present
  1384→  1384→      //   2. timer555 in astable mode — computes output from sim.time + R/C params
  1385→  1385→      // Without fast-forward, a 1Hz clock would take ~10000 steps per cycle.
  1386→  1386→      const hasCapacitor = simComponents.some(c => c.type === 'capacitor' || c.type === 'inductor');
  1387→  1387→      const hasPulseSource = simComponents.some(c => c.type === 'pulseSource');
  1388→  1388→      const hasArduino = simComponents.some(c => c.type === 'arduinoReal' || c.type === 'arduino');
  1389→  1389→      const hasAstable555 = simComponents.some(c =>
  1390→  1390→        c.type === 'timer555' && (c.parameters.astable as boolean) === true);
  1391→  1391→
  1392→  1392→      // Compute the minimum clock frequency across all digital clock sources
  1393→  1393→      let minFreq = Infinity;
  1394→  1394→
  1395→  1395→      // Pulse sources
  1396→  1396→      if (hasPulseSource && !hasCapacitor && !hasArduino) {
  1397→  1397→        for (const c of simComponents) {
  1398→  1398→          if (c.type === 'pulseSource') {
  1399→  1399→            const f = c.parameters.frequency as number;
  1400→  1400→            if (f > 0 && f < minFreq) minFreq = f;
  1401→  1401→          }
  1402→  1402→        }
  1403→  1403→      }
  1404→  1404→
  1405→  1405→      // Astable 555 timers: frequency = 1 / (0.693 * (R1 + 2*R2) * C)
  1406→  1406→      if (hasAstable555) {
  1407→  1407→        for (const c of simComponents) {
  1408→  1408→          if (c.type === 'timer555' && (c.parameters.astable as boolean) === true) {
  1409→  1409→            const r1 = (c.parameters.r1 as number) || 47000;
  1410→  1410→            const r2 = (c.parameters.r2 as number) || 47000;
  1411→  1411→            const cap = (c.parameters.c as number) || 1e-5;
  1412→  1412→            const period = 0.693 * (r1 + 2 * r2) * cap;
  1413→  1413→            const f = 1 / period;
  1414→  1414→            if (f > 0 && f < minFreq) minFreq = f;
  1415→  1415→          }
  1416→  1416→        }
  1417→  1417→      }
  1418→  1418→
  1419→  1419→      if (minFreq !== Infinity && minFreq > 0) {
  1420→  1420→        const currentTime = result.sim.time;
  1421→  1421→        const phase = (currentTime * minFreq) % 1;
  1422→  1422→        const timeToRisingEdge = (1.0 - phase) / minFreq;
  1423→  1423→        const advance = Math.min(0.016, timeToRisingEdge + 0.001);
  1424→  1424→        if (advance > dt) {
  1425→  1425→          const newTime = currentTime + advance;
  1426→  1426→          prev.time = newTime;
  1427→  1427→          result.sim.time = newTime;
  1428→  1428→        }
  1429→  1429→      }
  1430→  1430→    }
  1431→  1431→    if (!result) {
  1432→  1432→      // Solver failed — likely singular matrix (no ground, or conflicting voltage sources)
  1433→  1433→      const hasGround = simComponents.some(c => c.type === 'ground');
  1434→  1434→      const errorMsg = !hasGround
  1435→  1435→        ? 'Simulation failed: No ground reference found. Add a Ground component to your circuit.'
  1436→  1436→        : 'Simulation failed: Singular matrix. Check for conflicting voltage sources or short circuits.';
  1437→  1437→      set({ running: false, paused: true, simError: errorMsg });
  1438→  1438→      return;
  1439→  1439→    }
  1440→  1440→    // update traces for oscilloscope components — when using hierarchy, the
  1441→  1441→    // active sheet's oscilloscope IDs match the flattened IDs (no prefix when
  1442→  1442→    // on root). When inside a sub-sheet, oscilloscope IDs need prefix lookup.
  1443→  1443→    const traces = [...s.traces];
  1444→  1444→    for (const comp of simComponents) {
  1445→  1445→      if (comp.type !== 'oscilloscope') continue;
  1446→  1446→      const plugin = plugins.get('oscilloscope');
  1447→  1447→      if (!plugin) continue;

  1449→  1449→      const measurements = plugin.measure(comp.parameters, terminals, result.sim);
  1450→  1450→      const v = parseFloat(measurements[0]?.value ?? '0');
  1451→  1451→      // .PRINT directive: if printNodes is set, log matching node voltages to console
  1452→  1452→      if (s.simOptions.printNodes && s.simOptions.printNodes.length > 0) {

  1458→  1458→            console.log(`[.PRINT t=${result.sim.time.toFixed(6)}] V(${termKey}) = ${nodeV.toFixed(6)} V`);
  1459→  1459→          }
  1460→  1460→        }
  1461→  1461→      }

  1466→  1466→      if (traceIdx < 0) {

  1469→  1469→          color: (comp.parameters.color as string) || '#22d3ee',
  1470→  1470→          label: (comp.parameters.label as string) || 'CH',
  1471→  1471→          samples: [],
  1472→  1472→        });

  1474→  1474→      }
  1475→  1475→      const trace = traces[traceIdx];
  1476→  1476→      trace.samples.push({ time: result.sim.time, voltage: v });

  1479→  1479→      }

  1481→  1481→    }

  1492→  1492→        }
