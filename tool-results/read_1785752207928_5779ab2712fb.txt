   274→/** Get the reference designator prefix for a component type */
   275→function refdesPrefix(type: string): string {
   276→  switch (type) {
   277→    case 'resistor': return 'R';
   278→    case 'capacitor': return 'C';
   279→    case 'inductor': return 'L';
   280→    case 'led': return 'LED';
   281→    case 'diode': return 'D';
   282→    case 'zener': return 'DZ';
   283→    case 'schottky': return 'D';
   284→    case 'dcVoltage':
   285→    case 'acVoltage':
   286→    case 'pulseSource': return 'V';
   287→    case 'currentSource': return 'I';
   288→    case 'timer555':
   289→    case 'opamp':
   290→    case 'opampRails':
   291→    case 'voltageRegulator':
   292→    case 'vco': return 'U';
   293→    case 'npn':
   294→    case 'pnp':
   295→    case 'nmos':
   296→    case 'pmos': return 'Q';
   297→    case 'switch':
   298→    case 'pushButton': return 'SW';
   299→    case 'fuse': return 'F';
   300→    case 'crystal': return 'Y';
   301→    case 'transformer': return 'T';
   302→    case 'speaker':
   303→    case 'buzzer': return 'LS';
   304→    case 'dcMotor': return 'M';
   305→    case 'photoresistor': return 'LDR';
   306→    case 'sevenSegment': return 'DSP';
   307→    case 'ground':
   308→    case 'powerGND': return 'GND';
   309→    case 'junction': return 'J';
   310→    case 'oscilloscope':
   311→    case 'voltmeter':
   312→    case 'ammeter': return 'TP';
   313→    case 'arduino':
   314→    case 'arduinoReal':
   315→    case 'raspberryPi': return 'U';
   316→    case 'netLabel':
   317→    case 'busLabel': return 'NL';
   318→    case 'bus': return 'BUS';
   319→    default:
   320→      if (type.startsWith('7400_') || type.startsWith('7402_') || type.startsWith('7404_') ||
   321→          type.startsWith('7408_') || type.startsWith('7432_') || type.startsWith('7486_') ||
   322→          type.startsWith('7474_')) return 'U';
   323→      if (type.startsWith('custom_')) return 'U';
   324→      return 'U';
   325→  }
   326→}

   328→/** Generate the next refdes for a given type, based on existing components */
   329→function nextRefdes(type: string, components: CircuitComponent[]): string {
   330→  const prefix = refdesPrefix(type);
   331→  let maxNum = 0;
   332→  for (const c of components) {
   333→    if (refdesPrefix(c.type) === prefix) {
   334→      const ref = c.refdes ?? c.id;
   335→      const match = ref.match(/(\d+)$/);
   336→      if (match) {
   337→        const n = parseInt(match[1]);
   338→        if (n > maxNum) maxNum = n;
   339→      }
   340→    }
   341→  }
   342→  return `${prefix}${maxNum + 1}`;
   343→}

   345→function snapshot(s: {
   346→  components: CircuitComponent[];
   347→  wires: Wire[];
   348→  drawings?: DrawingPrimitive[];
   349→  noConnects?: NoConnectMarker[];
   350→  groups?: Group[];
   351→  sheets?: HierarchicalSheet[];
   352→  netClasses?: NetClass[];
   353→}) {
   354→  return {
   355→    components: s.components.map((c) => ({ ...c, parameters: { ...c.parameters }, simState: undefined, fields: c.fields ? c.fields.map((f) => ({ ...f })) : undefined })),
   356→    wires: s.wires.map((w) => ({ ...w })),
   357→    drawings: s.drawings ? s.drawings.map((d) => ({ ...d })) : [],
   358→    noConnects: s.noConnects ? s.noConnects.map((n) => ({ ...n })) : [],
   359→    groups: s.groups ? s.groups.map((g) => ({ ...g, componentIds: [...g.componentIds], wireIds: [...g.wireIds], drawingIds: [...g.drawingIds] })) : [],
   360→    sheets: s.sheets ? s.sheets.map((sh) => ({ ...sh, pins: sh.pins.map((p) => ({ ...p })) })) : [],
   361→    netClasses: s.netClasses ? s.netClasses.map((nc) => ({ ...nc, nets: [...nc.nets] })) : [],
   362→  };
   363→}

   367→// helper: get plugin defaults
   368→function defaultsFor(type: string): Record<string, number | string | boolean> {
   369→  const plugin = getPlugin(type);
   370→  if (!plugin) return {};
   371→  const defaults: Record<string, number | string | boolean> = {};
   372→  for (const p of plugin.parameters) defaults[p.key] = p.default;
   373→  return defaults;

   391→  future: [],

   398→  netClasses: [],

   405→  activeSheet: '',

   407→  // display
   408→  units: 'grid',
   409→  gridSize: 1,
   410→  showPinNumbers: false,
   411→  showPinNames: false,
   412→  showPinElecTypes: false,

   416→  theme: 'dark',
   417→  customHotkeys: {},

   420→  addComponent: (type, position) => {
   421→    get().pushHistory();
   422→    const id = genId('comp');
   423→    const s = get();
   424→    const refdes = nextRefdes(type, s.components);
   425→    const comp: CircuitComponent = {
   426→      id,
   427→      type,
   428→      position: { ...position },
   429→      rotation: 0,
   430→      parameters: defaultsFor(type),
   431→      refdes,
   432→    };
   433→    set((s) => ({ components: [...s.components, comp], selection: { type: 'component', id } }));
   434→    return id;
   435→  },
   436→
   437→  moveComponent: (id, position) => {
   438→    set((s) => ({
   439→      components: s.components.map((c) =>
   440→        c.id === id ? { ...c, position: { ...position } } : c,
   441→      ),
   442→    }));
   443→  },
   444→
   445→  rotateComponent: (id) => {
   446→    get().pushHistory();
   447→    set((s) => ({
   448→      components: s.components.map((c) =>
   449→        c.id === id
   450→          ? { ...c, rotation: (((c.rotation + 1) % 4) as 0 | 1 | 2 | 3) }
   451→          : c,
   452→      ),
   453→    }));
   454→  },
   455→
   456→  mirrorComponent: (id, axis) => {
   457→    get().pushHistory();
   458→    set((s) => ({
   459→      components: s.components.map((c) =>
   460→        c.id === id
   461→          ? { ...c, [axis === 'x' ? 'mirrorX' : 'mirrorY']: !c[axis === 'x' ? 'mirrorX' : 'mirrorY'] }
   462→          : c,
   463→      ),
   464→    }));
   465→  },
   466→
   467→  rotateComponentFree: (id, degrees) => {
   468→    // Snap to 15° increments (24 distinct angles). Stored in `rotationDeg`.
   469→    // The canvas's render loop should use rotationDeg if present, otherwise fall back to rotation*90.
   470→    get().pushHistory();
   471→    const snapped = Math.round(degrees / 15) * 15;
   472→    set((s) => ({
   473→      components: s.components.map((c) =>
   474→        c.id === id ? { ...c, rotationDeg: snapped } : c,
   475→      ),
   476→    }));
   477→  },
   478→
   479→  toggleDeMorgan: (id) => {
   480→    const comp = get().components.find((c) => c.id === id);
   481→    if (!comp) return;
   482→    const plugin = getPlugin(comp.type);
   483→    if (!plugin?.hasAlternateBody) return;
   484→    get().pushHistory();
   485→    set((s) => ({
   486→      components: s.components.map((c) =>
   487→        c.id === id ? { ...c, convert: c.convert === 2 ? 1 : 2 } : c,
   488→      ),
   489→    }));
   490→  },
   491→
   492→  swapPins: (id, pinIdA, pinIdB) => {
   493→    const comp = get().components.find((c) => c.id === id);
   494→    if (!comp) return false;
   495→    const plugin = getPlugin(comp.type);
   496→    if (!plugin?.pinSwapGroups) return false;
   497→    // Check if both pins are in the same swap group
   498→    const inSameGroup = plugin.pinSwapGroups.some((group) =>
   499→      group.includes(pinIdA) && group.includes(pinIdB),
   500→    );
   501→    if (!inSameGroup) return false;
   502→    // Swap: we re-emit all wires that referenced pinIdA to pinIdB and vice versa.
   503→    // This requires walking the wires array and swapping terminal IDs.
   504→    get().pushHistory();
   505→    set((s) => ({
   506→      wires: s.wires.map((w) => {
   507→        if (w.from.componentId === id && w.from.terminalId === pinIdA) {
   508→          return { ...w, from: { ...w.from, terminalId: pinIdB } };
   509→        }
   510→        if (w.from.componentId === id && w.from.terminalId === pinIdB) {
   511→          return { ...w, from: { ...w.from, terminalId: pinIdA } };
   512→        }
   513→        if (w.to.componentId === id && w.to.terminalId === pinIdA) {
   514→          return { ...w, to: { ...w.to, terminalId: pinIdB } };
   515→        }
   516→        if (w.to.componentId === id && w.to.terminalId === pinIdB) {
   517→          return { ...w, to: { ...w.to, terminalId: pinIdA } };
   518→        }

   945→      // from a sub-sheet, the root snapshot is in `childSheets['__root__']`

   990→        wires: [],

  1283→  step: () => {
  1284→    const s = get();
  1285→    if (s.components.length === 0 && s.sheets.length === 0) return;
  1286→
  1287→    // ── Cross-sheet simulation ────────────────────────────────────────────
  1288→    // If we're inside a sub-sheet OR the root has sub-sheets, we need to
  1289→    // flatten the hierarchy into a single (components, wires) pair before
  1290→    // simulating. We do this on every step so sub-sheet edits are picked up
  1291→    // immediately. The flattened components have prefixed IDs (e.g. "amp.R1")
  1292→    // so they don't collide with the active sheet's IDs.
  1293→    let simComponents = s.components;
  1294→    let simWires = s.wires;
  1295→    let usingHierarchy = false;
  1296→    if (s.activeSheet || s.sheets.length > 0) {
  1297→      usingHierarchy = true;
  1298→      // Save the current sheet back into childSheets so flattenHierarchy sees it
  1299→      let childSheets = s.childSheets;
  1300→      if (s.activeSheet) {
  1301→        childSheets = {
  1302→          ...childSheets,
  1303→          [s.activeSheet]: snapshotSheet(s.components, s.wires, s.sheets),
  1304→        };
  1305→      }
  1306→      // Build root document from either the active root state or the saved __root__
  1307→      const rootDoc = s.activeSheet
  1308→        ? (childSheets as any).__root__ ?? { version: 1 as const, components: [], wires: [], sheets: [] }
  1309→        : { version: 1 as const, components: s.components, wires: s.wires, sheets: s.sheets };
  1310→      const flat = flattenHierarchy(rootDoc as any, childSheets as any);
  1311→      simComponents = flat.components;
  1312→      simWires = flat.wires;
  1313→    }
  1314→
  1315→    const plugins = new Map<string, any>();
  1316→    for (const c of simComponents) {
  1317→      const p = getPlugin(c.type);
  1318→      if (p) plugins.set(c.type, p);
  1319→    }
  1320→    // Keep a persistent state map across steps so capacitors/inductors/555 remember their state
  1321→    const persistentState = s.simContext?.state ?? {};
  1322→    const prev = s.simContext
  1323→      ? {
  1324→          nodeVoltage: s.simContext.nodeVoltage,
  1325→          branchCurrent: s.simContext.branchCurrent,
  1326→          time: s.simContext.time,
  1327→          state: persistentState,
  1328→        }
  1329→      : { nodeVoltage: new Float64Array(0), branchCurrent: new Float64Array(0), time: 0, state: persistentState };
  1330→    // run sub-steps based on speed
  1331→    const subSteps = Math.max(1, Math.floor(s.speed));
  1332→    const dt = s.dt;
  1333→    // Build simOptions from the editor's settings (temperature, .IC, .NODESET)
  1334→    const simOpts = {
  1335→      initialConditions: s.simOptions?.initialConditions,
  1336→      nodeSets: s.simOptions?.nodeSets,
  1337→    };
  1338→    let result: { sim: SimContext; branchCurrentSize: number; nodeMap: any } | null = null;
  1339→    for (let i = 0; i < subSteps; i++) {
  1340→      result = simulateStep(simComponents, simWires, plugins, prev, dt, simOpts);
  1341→      if (!result) break;
  1342→      prev.nodeVoltage = result.sim.nodeVoltage;
  1343→      prev.branchCurrent = result.sim.branchCurrent;
  1344→      prev.time = result.sim.time;
  1345→      prev.state = result.sim.state;
  1346→
  1347→      // Digital fast-forward: advance sim.time toward the next clock edge.
  1348→      // Handles two types of clock sources:
  1349→      //   1. pulseSource (crystal) — when no capacitors/inductors present
  1350→      //   2. timer555 in astable mode — computes output from sim.time + R/C params
  1351→      // Without fast-forward, a 1Hz clock would take ~10000 steps per cycle.
  1352→      const hasCapacitor = simComponents.some(c => c.type === 'capacitor' || c.type === 'inductor');
  1353→      const hasPulseSource = simComponents.some(c => c.type === 'pulseSource');
  1354→      const hasArduino = simComponents.some(c => c.type === 'arduinoReal' || c.type === 'arduino');
  1355→      const hasAstable555 = simComponents.some(c =>
  1356→        c.type === 'timer555' && (c.parameters.astable as boolean) === true);
  1357→
  1358→      // Compute the minimum clock frequency across all digital clock sources
  1359→      let minFreq = Infinity;
  1360→
  1361→      // Pulse sources
  1362→      if (hasPulseSource && !hasCapacitor && !hasArduino) {
  1363→        for (const c of simComponents) {
  1364→          if (c.type === 'pulseSource') {
  1365→            const f = c.parameters.frequency as number;
  1366→            if (f > 0 && f < minFreq) minFreq = f;
  1367→          }
  1368→        }
  1369→      }
  1370→
  1371→      // Astable 555 timers: frequency = 1 / (0.693 * (R1 + 2*R2) * C)
  1372→      if (hasAstable555) {
  1373→        for (const c of simComponents) {
  1374→          if (c.type === 'timer555' && (c.parameters.astable as boolean) === true) {
  1375→            const r1 = (c.parameters.r1 as number) || 47000;
  1376→            const r2 = (c.parameters.r2 as number) || 47000;
  1377→            const cap = (c.parameters.c as number) || 1e-5;
  1378→            const period = 0.693 * (r1 + 2 * r2) * cap;
  1379→            const f = 1 / period;
  1380→            if (f > 0 && f < minFreq) minFreq = f;
  1381→          }
  1382→        }
  1383→      }
  1384→
  1385→      if (minFreq !== Infinity && minFreq > 0) {
  1386→        const currentTime = result.sim.time;
  1387→        const phase = (currentTime * minFreq) % 1;
  1388→        const timeToRisingEdge = (1.0 - phase) / minFreq;
  1389→        const advance = Math.min(0.016, timeToRisingEdge + 0.001);
  1390→        if (advance > dt) {
  1391→          const newTime = currentTime + advance;
  1392→          prev.time = newTime;
  1393→          result.sim.time = newTime;
  1394→        }
  1395→      }
  1396→    }
  1397→    if (!result) {
  1398→      set({ running: false, paused: true });
  1399→      return;
  1400→    }
  1401→    // update traces for oscilloscope components — when using hierarchy, the
  1402→    // active sheet's oscilloscope IDs match the flattened IDs (no prefix when
  1403→    // on root). When inside a sub-sheet, oscilloscope IDs need prefix lookup.
  1404→    const traces = [...s.traces];
  1405→    for (const comp of simComponents) {
  1406→      if (comp.type !== 'oscilloscope') continue;
  1407→      const plugin = plugins.get('oscilloscope');
  1408→      if (!plugin) continue;
  1409→      const terminals = getTerminalsForComponent(comp, plugin, result.nodeMap);
  1410→      const measurements = plugin.measure(comp.parameters, terminals, result.sim);
  1411→      const v = parseFloat(measurements[0]?.value ?? '0');
  1412→      // .PRINT directive: if printNodes is set, log matching node voltages to console
  1413→      if (s.simOptions.printNodes && s.simOptions.printNodes.length > 0) {
  1414→        for (const termKey of s.simOptions.printNodes) {
  1415→          const nodeId = result.nodeMap.terminalNode.get(termKey);
  1416→          if (nodeId != null && nodeId > 0) {
  1417→            const nodeV = result.sim.nodeVoltage[nodeId - 1] ?? 0;
  1418→            // eslint-disable-next-line no-console
  1419→            console.log(`[.PRINT t=${result.sim.time.toFixed(6)}] V(${termKey}) = ${nodeV.toFixed(6)} V`);
  1420→          }
  1421→        }
  1422→      }
  1423→      // When using hierarchy, the trace's componentId is the active sheet's local ID
  1424→      // (without prefix). Map back: if comp.id contains a '.', strip the prefix.
  1425→      const traceKey = usingHierarchy && comp.id.includes('.') ? comp.id.split('.').slice(1).join('.') : comp.id;
  1426→      let traceIdx = traces.findIndex((t) => t.componentId === traceKey);
  1427→      if (traceIdx < 0) {
  1428→        traces.push({
  1429→          componentId: traceKey,
  1430→          color: (comp.parameters.color as string) || '#22d3ee',
  1431→          label: (comp.parameters.label as string) || 'CH',
  1432→          samples: [],
  1433→        });
  1434→        traceIdx = traces.length - 1;
  1435→      }
  1436→      const trace = traces[traceIdx];
  1437→      trace.samples.push({ time: result.sim.time, voltage: v });
  1438→      if (trace.samples.length > s.maxTraceSamples) {
  1439→        trace.samples = trace.samples.slice(-s.maxTraceSamples);
  1440→      }
  1441→      traces[traceIdx] = { ...trace };
  1442→    }
  1443→    // Run physics validation (debug — catches simulation bugs).
  1444→    // Only runs when a special debug flag is set to avoid perf overhead in production.
  1445→    let physicsViolations: PhysicsViolation[] = s.physicsViolations;
  1446→    if (typeof window !== 'undefined' && (window as any).__PHYSICS_DEBUG__) {
  1447→      try {
  1448→        const validation = validatePhysics(simComponents, simWires, plugins, result.sim);
  1449→        physicsViolations = validation.violations;
  1450→        if (!validation.passed) {
  1451→          // eslint-disable-next-line no-console
  1452→          console.warn('[Physics] Violations detected:', validation.violations.length);
  1453→        }
  1454→      } catch (e) {
  1455→        // validation errors should never break the simulation
  1456→      }
  1457→    }
  1458→    set({ simContext: result.sim, traces, physicsViolations });
  1459→  },
