    69→  const setSnapToGrid = useEditor((s) => s.setSnapToGrid);
    70→  const past = useEditor((s) => s.past.length);

    72→  // KiCad-parity new state

    75→  const showPinNumbers = useEditor((s) => s.showPinNumbers);

    83→  const showValues = useEditor((s) => s.showValues);
    84→  const setShowValues = useEditor((s) => s.setShowValues);

    87→
    88→  const fileInputRef = useRef<HTMLInputElement | null>(null);

    90→  const [showMyCircuits, setShowMyCircuits] = useState(false);
    91→  const [showSpiceImport, setShowSpiceImport] = useState(false);
    92→  const [showSubCircuit, setShowSubCircuit] = useState(false);
    93→  const [showSymbolEditor, setShowSymbolEditor] = useState(false);

   133→    const doc = serialize();

   135→    const url = URL.createObjectURL(blob);
   136→    const a = document.createElement('a');
   137→    a.href = url;
   138→    a.download = `circuit_${Date.now()}.json`;
   139→    a.click();
   140→    URL.revokeObjectURL(url);
   141→  }, [serialize]);

   144→    const file = e.target.files?.[0];
   145→    if (!file) return;
   146→    const reader = new FileReader();
   147→    reader.onload = () => {
   148→      try {
   149→        const doc = JSON.parse(reader.result as string);
   150→        if (doc.version === 1 && Array.isArray(doc.components) && Array.isArray(doc.wires)) {
   151→          loadDocument(doc);
   152→        } else {
   153→          alert('Invalid circuit file');
   154→        }
   155→      } catch (err) {
   156→        alert('Failed to parse file: ' + (err as Error).message);
   157→      }
   158→    };
   159→    reader.readAsText(file);
   160→    e.target.value = '';
   161→  }, [loadDocument]);
   162→
   163→  // KiCad / Eagle schematic import
   164→  const handleSchImport = useCallback(async (e: React.ChangeEvent<HTMLInputElement>) => {
   165→    const file = e.target.files?.[0];
   166→    if (!file) return;
   167→    const reader = new FileReader();
   168→    reader.onload = () => {
   169→      try {
   170→        const text = reader.result as string;
   171→        const result = parseSchematicFile(text, file.name);
   172→        loadDocument(result.document);
   173→        if (result.unknownSymbols.length > 0) {
   174→          toast.warning(`Imported with ${result.unknownSymbols.length} unknown symbols skipped`, {
   175→            description: result.unknownSymbols.slice(0, 5).join('\n'),
   176→          });
   177→        } else {
   178→          toast.success(`Imported ${file.name}`);
   179→        }
   180→        if (result.warnings.length > 0) {
   181→          console.log('Import warnings:', result.warnings);
   182→        }
   183→      } catch (err) {
   184→        toast.error('Import failed: ' + (err as Error).message);
   185→      }
   186→    };
   187→    reader.readAsText(file);
   188→    e.target.value = '';
   189→  }, [loadDocument]);
   190→

   197→

   200→      const doc = serialize();
   201→      const blob = await exportSchematicPNG(doc, 2);
   202→      downloadBlob(blob, `schematic_${Date.now()}.png`);
   203→      toast.success('Exported PNG');

   205→      toast.error('PNG export failed: ' + (err as Error).message);
   206→    }
   207→  }, [serialize]);
   208→
   209→  const handlePlotPDF = useCallback(async () => {
   210→    try {
   211→      const doc = serialize();
   212→      const blob = await exportSchematicPDF(doc);
   213→      downloadBlob(blob, `schematic_${Date.now()}.pdf`);

   216→      toast.error('PDF export failed: ' + (err as Error).message);
   217→    }
   218→  }, [serialize]);
   219→

   235→    const doc = serialize();
   236→    const ext = format;
   237→    if (format === 'csv') {
   238→      downloadText(exportBOMCSV(doc), `bom_${Date.now()}.csv`, 'text/csv');

   420→          <TooltipTrigger asChild>
   421→            <Button size="sm" variant="ghost" onClick={() => setShowSubCircuit(true)} disabled={running}>
