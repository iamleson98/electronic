     1→     1→// The simulation engine: builds a node index from the wire graph, stamps
     2→     2→// every component into an MNA system, solves it, and propagates results.
     3→     3→//
     4→     4→// Supports DC operating point and transient analysis (backward Euler).
     5→     5→// Non-linear components do a few Newton iterations per step (simple).
     6→     6→
     7→     7→import type {
     8→     8→  CircuitComponent,
     9→     9→  ComponentPlugin,
    10→    10→  MnaSystem,
    11→    11→  SimContext,
    12→    12→  Wire,
    13→    13→} from './types';
    14→    14→import { createMnaSystem, solveMna } from './solver';
    15→    15→import { createSparseMnaSystem, solveSparse, shouldUseSparseSolver, asMnaSystem } from './sparse-klu';
    16→    16→
    17→    17→export interface NodeMap {
    18→    18→  /** key = `${componentId}:${terminalId}` -> node id (0 = ground) */
    19→    19→  terminalNode: Map<string, number>;
    20→    20→  /** total number of nodes (including ground) */
    21→    21→  numNodes: number;
    22→    22→}
    23→    23→
    24→    24→/**
    25→    25→ * Expand a bus vector name like "D[0..7]" into ["D0", "D1", ..., "D7"].
    26→    26→ * Supports [start..end] (ascending or descending).
    27→    27→ * Single-bit names like "D0" return ["D0"].
    28→    28→ * Invalid patterns return [name] unchanged.
    29→    29→ */
    30→    30→export function expandBusVector(name: string): string[] {
    31→    31→  // Match patterns: BASE[START..END] or BASE[START:END]
    32→    32→  const m = name.match(/^([A-Za-z_]\w*)\[(\d+)\.\.(\d+)\]$/);
    33→    33→  if (!m) return [name];
    34→    34→  const [, base, startStr, endStr] = m;
    35→    35→  const start = parseInt(startStr, 10);
    36→    36→  const end = parseInt(endStr, 10);
    37→    37→  if (start <= end) {
    38→    38→    const bits: string[] = [];
    39→    39→    for (let i = start; i <= end; i++) bits.push(`${base}${i}`);
    40→    40→    return bits;
    41→    41→  } else {
    42→    42→    const bits: string[] = [];
    43→    43→    for (let i = start; i >= end; i--) bits.push(`${base}${i}`);
    44→    44→    return bits;
    45→    45→  }
    46→    46→}
    47→    47→
    48→    48→/**
    49→    49→ * Build a node map by union-find over all terminal connections.
    50→    50→ * Any terminal connected to a `ground` plugin's terminal becomes node 0.
    51→    51→ */
    52→    52→export function buildNodeMap(components: CircuitComponent[], wires: Wire[], plugins: Map<string, ComponentPlugin>): NodeMap {
    53→    53→  const terminalNode = new Map<string, number>();
    54→    54→  const parent: number[] = [0]; // node 0 is ground
    55→    55→
    56→    56→  function find(x: number): number {
    57→    57→    while (parent[x] !== x) {
    58→    58→      parent[x] = parent[parent[x]];
    59→    59→      x = parent[x];
    60→    60→    }
    61→    61→    return x;
    62→    62→  }
    63→    63→  function union(a: number, b: number) {
    64→    64→    const ra = find(a);
    65→    65→    const rb = find(b);
    66→    66→    if (ra === rb) return;
    67→    67→    // prefer to keep ground (0) as root
    68→    68→    if (ra === 0) parent[rb] = 0;
    69→    69→    else if (rb === 0) parent[ra] = 0;
    70→    70→    else parent[rb] = ra;
    71→    71→  }
    72→    72→  function newNode(): number {
    73→    73→    const id = parent.length;
    74→    74→    parent.push(id);
    75→    75→    return id;
    76→    76→  }
    77→    77→
    78→    78→  function termKey(cId: string, tId: string) {
    79→    79→    return `${cId}:${tId}`;
    80→    80→  }
    81→    81→  function nodeFor(cId: string, tId: string): number {
    82→    82→    const k = termKey(cId, tId);
    83→    83→    let n = terminalNode.get(k);
    84→    84→    if (n === undefined) {
    85→    85→      n = newNode();
    86→    86→      terminalNode.set(k, n);
    87→    87→    }
    88→    88→    return n;
    89→    89→  }
    90→    90→
    91→    91→  // First: pre-assign ground nodes from any "ground" plugin
    92→    92→  for (const comp of components) {
    93→    93→    const plugin = plugins.get(comp.type);
    94→    94→    if (!plugin) continue;
    95→    95→    if (comp.type === 'ground' || comp.type === 'powerGND') {
    96→    96→      for (const t of plugin.terminals) {
    97→    97→        const k = termKey(comp.id, t.id);
    98→    98→        terminalNode.set(k, 0);
    99→    99→      }
   100→   100→    }
   101→   101→  }
   102→   102→
   103→   103→  // Power symbols & net labels: unify all terminals with the same `net` parameter
   104→   104→  const netToNode = new Map<string, number>();
   105→   105→  for (const comp of components) {
   106→   106→    const plugin = plugins.get(comp.type);
   107→   107→    if (!plugin) continue;
   108→   108→    const isPowerSymbol =
   109→   109→      comp.type === 'powerGND' || comp.type === 'powerAGND' ||
   110→   110→      comp.type === 'powerVCC' || comp.type === 'power5V' || comp.type === 'power3V3' ||
   111→   111→      comp.type === 'power1V8' || comp.type === 'power2V5' ||
   112→   112→      comp.type === 'power12V' || comp.type === 'powerMinus12V' ||
   113→   113→      comp.type === 'powerMinus5V' || comp.type === 'powerAVDD' ||
   114→   114→      comp.type === 'powerVBAT' || comp.type === 'powerFlag' ||
   115→   115→      comp.type === 'netLabel' || comp.type === 'busLabel' ||
   116→   116→      comp.type === 'busVectorLabel' || comp.type === 'hierLabel';
   117→   117→    if (!isPowerSymbol) continue;
   118→   118→    const netName = (comp.parameters.net as string) || '';
   119→   119→    if (!netName) continue;
   120→   120→    if (netName === 'GND' || netName === 'gnd' || netName === '0') {
   121→   121→      for (const t of plugin.terminals) terminalNode.set(termKey(comp.id, t.id), 0);
   122→   122→      continue;
   123→   123→    }
   124→   124→    // Bus vector expansion: a label with text "D[0..7]" creates 8 implicit
   125→   125→    // nodes D0, D1, ..., D7. Each terminal of the busVectorLabel plugin gets
   126→   126→    // unified to its corresponding bit's shared node.
   127→   127→    const expandedBits = expandBusVector(netName);
   128→   128→    if (expandedBits.length > 1) {
   129→   129→      // Multi-bit bus vector — assign each terminal a separate bit's node
   130→   130→      for (let i = 0; i < plugin.terminals.length && i < expandedBits.length; i++) {
   131→   131→        const bitName = expandedBits[i];
   132→   132→        let bitNode = netToNode.get(bitName);
   133→   133→        if (bitNode === undefined) { bitNode = newNode(); netToNode.set(bitName, bitNode); }
   134→   134→        terminalNode.set(termKey(comp.id, plugin.terminals[i].id), bitNode);
   135→   135→      }
   136→   136→      continue;
   137→   137→    }
   138→   138→    let sharedNode = netToNode.get(netName);
   139→   139→    if (sharedNode === undefined) { sharedNode = newNode(); netToNode.set(netName, sharedNode); }
   140→   140→    for (const t of plugin.terminals) terminalNode.set(termKey(comp.id, t.id), sharedNode);
   141→   141→  }
   142→   142→
   143→   143→  // Process wires - union terminals
   144→   144→  for (const wire of wires) {
   145→   145→    const aKey = termKey(wire.from.componentId, wire.from.terminalId);
   146→   146→    const bKey = termKey(wire.to.componentId, wire.to.terminalId);
   147→   147→    const a = nodeFor(wire.from.componentId, wire.from.terminalId);
   148→   148→    const b = nodeFor(wire.to.componentId, wire.to.terminalId);
   149→   149→    // if either is already ground (0), union handles it
   150→   150→    union(a, b);
   151→   151→    const root = find(a);
   152→   152→    terminalNode.set(aKey, root);
   153→   153→    terminalNode.set(bKey, root);
   154→   154→  }
   155→   155→
   156→   156→  // Finalize: compress all paths and remap to 0..N contiguous
   157→   157→  // Collect all distinct roots
   158→   158→  const rootSet = new Set<number>();
   159→   159→  rootSet.add(0);
   160→   160→  for (const n of terminalNode.values()) {
   161→   161→    rootSet.add(find(n));
   162→   162→  }
   163→   163→  // assign contiguous ids, ground stays 0
   164→   164→  const rootToId = new Map<number, number>();
   165→   165→  rootToId.set(0, 0);
   166→   166→  let nextId = 1;
   167→   167→  for (const r of rootSet) {
   168→   168→    if (r === 0) continue;
   169→   169→    rootToId.set(r, nextId++);
   170→   170→  }
   171→   171→  for (const [k, n] of terminalNode) {
   172→   172→    terminalNode.set(k, rootToId.get(find(n))!);
   173→   173→  }
   174→   174→
   175→   175→  return { terminalNode, numNodes: nextId };
   176→   176→}
   177→   177→
   178→   178→export function getTerminalsForComponent(
   179→   179→  comp: CircuitComponent,
   180→   180→  plugin: ComponentPlugin,
   181→   181→  nodeMap: NodeMap,
   182→   182→): { terminalId: string; nodeId: number }[] {
   183→   183→  return plugin.terminals.map((t) => ({
   184→   184→    terminalId: t.id,
   185→   185→    nodeId: nodeMap.terminalNode.get(`${comp.id}:${t.id}`) ?? 0,
   186→   186→  }));
   187→   187→}
   188→   188→
   189→   189→/**
   190→   190→ * Compute the current flowing through each wire, in amperes.
   191→   191→ * Returns a map of wireId -> current (positive = from `from` terminal to `to` terminal).
   192→   192→ *
   193→   193→ * This follows the **conventional current** convention (positive charge flow from + to -),
   194→   194→ * which is what Falstad and most simulators use for the animated dots.
   195→   195→ *
   196→   196→ * Method:
   197→   197→ * - For each wire, we look at the component at the `from` end and compute the current
   198→   198→ *   flowing INTO that terminal from the wire's perspective.
   199→   199→ * - For passive components (R, C, L, diode, LED): current = (V(a)-V(b))/R flows from a to b
   200→   200→ *   THROUGH the component. If the wire is attached to terminal 'a', then current leaving
   201→   201→ *   the wire INTO 'a' = +I (the wire delivers current to 'a'). If attached to 'b',
   202→   202→ *   current leaving the wire INTO 'b' = -I.
   203→   203→ *   But we want current flowing FROM wire.to TO wire.from... actually simpler:
   204→   204→ *   The current through the wire = the current through the component, signed so that
   205→   205→ *   positive = conventional current flowing from wire.from to wire.to.
   206→   206→ *
   207→   207→ * - For voltage sources: current flows OUT of the + terminal (conventional current).
   208→   208→ *   We compute this from the actual MNA branch current if available, or approximate
   209→   209→ *   from the load.
   210→   210→ *
   211→   211→ * The sign convention: positive current = dots animate from `from` to `to`.
   212→   212→ * In the renderer, positive = amber dots, negative = cyan dots (reverse).
   213→   213→ */
   214→   214→export function computeWireCurrents(
   215→   215→  components: CircuitComponent[],
   216→   216→  wires: Wire[],
   217→   217→  plugins: Map<string, ComponentPlugin>,
   218→   218→  sim: SimContext,
   219→   219→): Map<string, number> {
   220→   220→  // First compute per-component currents
   221→   221→  const compCurrents = computeComponentCurrents(components, wires, plugins, sim);
   222→   222→  const result = new Map<string, number>();
   223→   223→  const nodeMap = buildNodeMap(components, wires, plugins);
   224→   224→
   225→   225→  // Build nodeCurrentOut for KCL-based voltage source current
   226→   226→  // This sums up ALL currents leaving each node from ALL component types,
   227→   227→  // so the voltage source current = total current drawn from its + node.
   228→   228→  const nodeCurrentOut = new Map<number, number>();
   229→   229→  for (const comp of components) {
   230→   230→    const plugin = plugins.get(comp.type);
   231→   231→    if (!plugin) continue;
   232→   232→    const terms = getTerminalsForComponent(comp, plugin, nodeMap);
   233→   233→    const i = compCurrents.get(comp.id) ?? 0;
   234→   234→
   235→   235→    if (comp.type === 'resistor' || comp.type === 'capacitor' || comp.type === 'inductor' ||
   236→   236→        comp.type === 'led' || comp.type === 'diode' || comp.type === 'switch' || comp.type === 'pushButton') {
   237→   237→      const t1Id = 'a';
   238→   238→      const t2Id = comp.type === 'led' || comp.type === 'diode' ? 'k' : 'b';
   239→   239→      const n1 = terms.find((t) => t.terminalId === t1Id)?.nodeId ?? 0;
   240→   240→      const n2 = terms.find((t) => t.terminalId === t2Id)?.nodeId ?? 0;
   241→   241→      nodeCurrentOut.set(n1, (nodeCurrentOut.get(n1) ?? 0) + i);
   242→   242→      nodeCurrentOut.set(n2, (nodeCurrentOut.get(n2) ?? 0) - i);
   243→   243→    } else if (comp.type === 'currentSource') {
   244→   244→      const p = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   245→   245→      const n = terms.find((t) => t.terminalId === 'n')?.nodeId ?? 0;
   246→   246→      nodeCurrentOut.set(p, (nodeCurrentOut.get(p) ?? 0) + i);
   247→   247→      nodeCurrentOut.set(n, (nodeCurrentOut.get(n) ?? 0) - i);
   248→   248→    } else if (comp.type === 'speaker' || comp.type === 'lamp' || comp.type === 'dcMotor') {
   249→   249→      // Load components: current flows a→b
   250→   250→      const n1 = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   251→   251→      const n2 = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   252→   252→      nodeCurrentOut.set(n1, (nodeCurrentOut.get(n1) ?? 0) + i);
   253→   253→      nodeCurrentOut.set(n2, (nodeCurrentOut.get(n2) ?? 0) - i);
   254→   254→    } else if (comp.type === 'timer555') {
   255→   255→      // 555: current exits OUT pin, enters VCC pin
   256→   256→      const outNode = terms.find((t) => t.terminalId === 'out')?.nodeId ?? 0;
   257→   257→      const vccNode = terms.find((t) => t.terminalId === 'vcc')?.nodeId ?? 0;
   258→   258→      nodeCurrentOut.set(outNode, (nodeCurrentOut.get(outNode) ?? 0) + i);
   259→   259→      nodeCurrentOut.set(vccNode, (nodeCurrentOut.get(vccNode) ?? 0) - i);
   260→   260→    } else if (comp.type === 'opamp') {
   261→   261→      // Op-amp: current exits OUT pin
   262→   262→      const outNode = terms.find((t) => t.terminalId === 'out')?.nodeId ?? 0;
   263→   263→      nodeCurrentOut.set(outNode, (nodeCurrentOut.get(outNode) ?? 0) + i);
   264→   264→    } else if (comp.type === 'npn' || comp.type === 'nmos') {
   265→   265→      // Transistor: current flows C→E or D→S
   266→   266→      const cOrD = comp.type === 'npn' ? 'c' : 'd';
   267→   267→      const eOrS = comp.type === 'npn' ? 'e' : 's';
   268→   268→      const cNode = terms.find((t) => t.terminalId === cOrD)?.nodeId ?? 0;
   269→   269→      const eNode = terms.find((t) => t.terminalId === eOrS)?.nodeId ?? 0;
   270→   270→      nodeCurrentOut.set(cNode, (nodeCurrentOut.get(cNode) ?? 0) + i);
   271→   271→      nodeCurrentOut.set(eNode, (nodeCurrentOut.get(eNode) ?? 0) - i);
   272→   272→    } else if (comp.type === 'pnp' || comp.type === 'pmos') {
   273→   273→      // PNP/PMOS: current flows E→C or S→D (reversed)
   274→   274→      const eOrS = comp.type === 'pnp' ? 'e' : 's';
   275→   275→      const cOrD = comp.type === 'pnp' ? 'c' : 'd';
   276→   276→      const eNode = terms.find((t) => t.terminalId === eOrS)?.nodeId ?? 0;
   277→   277→      const cNode = terms.find((t) => t.terminalId === cOrD)?.nodeId ?? 0;
   278→   278→      nodeCurrentOut.set(eNode, (nodeCurrentOut.get(eNode) ?? 0) + i);
   279→   279→      nodeCurrentOut.set(cNode, (nodeCurrentOut.get(cNode) ?? 0) - i);
   280→   280→    } else if (comp.type === 'voltmeter' || comp.type === 'ammeter' || comp.type === 'oscilloscope') {
   281→   281→      // Meters: negligible current (10MΩ input)
   282→   282→      const pNode = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   283→   283→      const nNode = terms.find((t) => t.terminalId === 'n')?.nodeId ?? 0;
   284→   284→      nodeCurrentOut.set(pNode, (nodeCurrentOut.get(pNode) ?? 0) + i);
   285→   285→      nodeCurrentOut.set(nNode, (nodeCurrentOut.get(nNode) ?? 0) - i);
   286→   286→    } else if (comp.type === 'sevenSegment') {
   287→   287→      // 7-segment: each segment is a conductance to com (no Vf).
   288→   288→      // Use the same hysteresis model as the stamp.
   289→   289→      const com = terms.find((t) => t.terminalId === 'com')?.nodeId ?? 0;
   290→   290→      const threshold = (comp.parameters.threshold as number) ?? 2.0;
   291→   291→      const rSeg = 220;
   292→   292→      const st = sim.state.__global ?? {};
   293→   293→      const stateKey = `7seg_${terms.map(t => t.nodeId).join('_')}`;
   294→   294→      const segStates = (st[stateKey] ?? {}) as Record<string, boolean>;
   295→   295→      for (const seg of ['a', 'b', 'c', 'd', 'e', 'f', 'g']) {
   296→   296→        const segNode = terms.find((t) => t.terminalId === seg)?.nodeId ?? 0;
   297→   297→        const v = sim.nodeVoltage[segNode] - sim.nodeVoltage[com];
   298→   298→        const prevOn = segStates[seg] ?? false;
   299→   299→        const on = prevOn ? v > threshold * 0.5 : v > threshold;
   300→   300→        if (on) {
   301→   301→          const iSeg = v / rSeg;
   302→   302→          nodeCurrentOut.set(segNode, (nodeCurrentOut.get(segNode) ?? 0) + iSeg);
   303→   303→          nodeCurrentOut.set(com, (nodeCurrentOut.get(com) ?? 0) - iSeg);
   304→   304→        }
   305→   305→      }
   306→   306→    } else if (comp.type === 'cd4026') {
   307→   307→      // CD4026: handled in a second pass below (after all 7-segment displays
   308→   308→      // have contributed their segment currents to nodeCurrentOut).
   309→   309→      // We can't compute the CD4026's VCC current here because the 7-seg
   310→   310→      // displays (which determine the segment currents) may not have been
   311→   311→      // processed yet (they appear later in the component list).
   312→   312→    }
   313→   313→    // Voltage sources, ground, junction, power symbols: skip (they define the current, not draw it)
   314→   314→  }
   315→   315→
   316→   316→  // Second pass: add CD4026 VCC current (sum of all ON segment currents).
   317→   317→  // This must run AFTER all 7-segment displays have contributed to nodeCurrentOut.
   318→   318→  for (const comp of components) {
   319→   319→    if (comp.type !== 'cd4026') continue;
   320→   320→    const plugin = plugins.get(comp.type);
   321→   321→    if (!plugin) continue;
   322→   322→    const terms = getTerminalsForComponent(comp, plugin, nodeMap);
   323→   323→    const vccNode = terms.find((t) => t.terminalId === 'vcc')?.nodeId ?? 0;
   324→   324→    let totalSegI = 0;
   325→   325→    for (const seg of ['a', 'b', 'c', 'd', 'e', 'f', 'g']) {
   326→   326→      const segNode = terms.find((t) => t.terminalId === seg)?.nodeId ?? 0;
   327→   327→      totalSegI += Math.abs(nodeCurrentOut.get(segNode) ?? 0);
   328→   328→    }
   329→   329→    if (vccNode > 0) {
   330→   330→      nodeCurrentOut.set(vccNode, (nodeCurrentOut.get(vccNode) ?? 0) + totalSegI);
   331→   331→    }
   332→   332→  }
   333→   333→
   334→   334→  for (const wire of wires) {
   335→   335→    // For each wire, compute current from BOTH the from and to components.
   336→   336→    //
   337→   337→    // Sign convention:
   338→   338→    //   computeTerminalCurrent returns "current leaving the component terminal toward the wire"
   339→   339→    //   Positive = current flowing OUT of the terminal INTO the wire.
   340→   340→    //
   341→   341→    // For a wire from→to:
   342→   342→    //   fromCurrent > 0 means current flows FROM the from-component INTO the wire → flows from→to ✓
   343→   343→    //   toCurrent > 0 means current flows FROM the to-component INTO the wire → flows to→from
   344→   344→    //   So wire current (from→to positive) = fromCurrent = -toCurrent
   345→   345→    //
   346→   346→    // In a series circuit: fromCurrent = -toCurrent (KCL).
   347→   347→    // Take the one with larger magnitude for reliability.
   348→   348→    const fromComp = components.find((c) => c.id === wire.from.componentId);
   349→   349→    const toComp = components.find((c) => c.id === wire.to.componentId);
   350→   350→    if (!fromComp || !toComp) continue;
   351→   351→    const fromPlugin = plugins.get(fromComp.type);
   352→   352→    const toPlugin = plugins.get(toComp.type);
   353→   353→    if (!fromPlugin || !toPlugin) continue;
   354→   354→
   355→   355→    const fromCurrent = computeTerminalCurrent(fromComp, fromPlugin, wire.from.terminalId, nodeMap, sim, compCurrents, nodeCurrentOut);
   356→   356→    const toCurrent = computeTerminalCurrent(toComp, toPlugin, wire.to.terminalId, nodeMap, sim, compCurrents, nodeCurrentOut);
   357→   357→
   358→   358→    // fromCurrent = current leaving from-component → entering wire at from end → flows from→to
   359→   359→    // toCurrent = current leaving to-component → entering wire at to end → flows to→from
   360→   360→    // Wire current (from→to positive) = fromCurrent = -toCurrent
   361→   361→    //
   362→   362→    // Prefer the SMALLER non-zero magnitude. In a series circuit both ends agree
   363→   363→    // (same magnitude), so either works. But when multiple wires share a node
   364→   364→    // (e.g., 6 CD4026 VCC pins on the same 5V supply), the fromCurrent gives the
   365→   365→    // TOTAL node current (727mA) while toCurrent gives the INDIVIDUAL draw (45mA).
   366→   366→    // The individual draw is correct for THIS wire.
   367→   367→    const fromMag = Math.abs(fromCurrent);
   368→   368→    const toMag = Math.abs(toCurrent);
   369→   369→    let current: number;
   370→   370→    if (toMag > 1e-12 && (fromMag < 1e-12 || toMag < fromMag)) {
   371→   371→      // Prefer toCurrent when it's available and smaller (or fromCurrent is zero)
   372→   372→      current = -toCurrent;
   373→   373→    } else if (fromMag > 1e-12) {
   374→   374→      // Fall back to fromCurrent
   375→   375→      current = fromCurrent;
   376→   376→    } else {
   377→   377→      current = 0;
   378→   378→    }
   379→   379→
   380→   380→    result.set(wire.id, current);
   381→   381→  }
   382→   382→
   383→   383→  return result;
   384→   384→}

   386→   386→/**
   387→   387→ * Compute the current leaving a specific terminal of a component, in amperes.
   388→   388→ * Positive = current flowing OUT of the terminal (into the wire).
   389→   389→ */
   390→   390→function computeTerminalCurrent(
   391→   391→  comp: CircuitComponent,
   392→   392→  plugin: ComponentPlugin,
   393→   393→  terminalId: string,
   394→   394→  nodeMap: any,
   395→   395→  sim: SimContext,
   396→   396→  compCurrents: Map<string, number>,
   397→   397→  nodeCurrentOut: Map<number, number>,
   398→   398→): number {
   399→   399→  const terms = getTerminalsForComponent(comp, plugin, nodeMap);
   400→   400→  const compCurrent = compCurrents.get(comp.id) ?? 0;
   401→   401→
   402→   402→  if (comp.type === 'resistor' || comp.type === 'capacitor' || comp.type === 'inductor') {
   403→   403→    // compCurrent > 0 = current flows a→b through component.
   404→   404→    // At 'a': current ENTERS component FROM wire → "leaving terminal toward wire" = -I
   405→   405→    // At 'b': current EXITS component TO wire → "leaving terminal toward wire" = +I
   406→   406→    // This means: if wire.from is at 'b', fromCurrent = +I → dots flow from→to (correct).
   407→   407→    //             if wire.from is at 'a', fromCurrent = -I → dots flow to→from (correct, since
   408→   408→    //             current actually flows from the source toward 'a').
   409→   409→    const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   410→   410→    const termNode = terms.find((t) => t.terminalId === terminalId)?.nodeId ?? 0;
   411→   411→    return (termNode === a) ? -compCurrent : compCurrent;
   412→   412→  } else if (comp.type === 'led' || comp.type === 'diode') {
   413→   413→    // Same as resistor: compCurrent > 0 = current flows a→k (forward).
   414→   414→    // At 'a': -I (enters component). At 'k': +I (exits component).
   415→   415→    const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   416→   416→    const termNode = terms.find((t) => t.terminalId === terminalId)?.nodeId ?? 0;
   417→   417→    return (termNode === a) ? -compCurrent : compCurrent;
   418→   418→  } else if (comp.type === 'dcVoltage' || comp.type === 'acVoltage' || comp.type === 'pulseSource') {
   419→   419→    // Voltage source: compCurrent > 0 = current flows p→n externally (out of +, into -).
   420→   420→    // At 'p': current EXITS source TO wire → "leaving terminal" = +I
   421→   421→    // At 'n': current ENTERS source FROM wire → "leaving terminal" = -I
   422→   422→    const p = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   423→   423→    const termNode = terms.find((t) => t.terminalId === terminalId)?.nodeId ?? 0;
   424→   424→    return (termNode === p) ? compCurrent : -compCurrent;
   425→   425→  } else if (comp.type === 'currentSource') {
   426→   426→    // Current source: same as voltage source — current flows p→n externally.
   427→   427→    const p = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   428→   428→    const termNode = terms.find((t) => t.terminalId === terminalId)?.nodeId ?? 0;
   429→   429→    return (termNode === p) ? compCurrent : -compCurrent;
   430→   430→  } else if (comp.type === 'switch' || comp.type === 'pushButton') {
   431→   431→    // Same as resistor: a→b, at 'a' = -I, at 'b' = +I
   432→   432→    const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   433→   433→    const termNode = terms.find((t) => t.terminalId === terminalId)?.nodeId ?? 0;
   434→   434→    return (termNode === a) ? -compCurrent : compCurrent;
   435→   435→  } else if (comp.type === 'npn' || comp.type === 'pnp' || comp.type === 'nmos' || comp.type === 'pmos') {
   436→   436→    // 3-terminal transistor/MOSFET
   437→   437→    // Physics (NPN/NMOS): current flows C→E (or D→S) THROUGH the device.
   438→   438→    //   At collector/drain: current ENTERS (leaves terminal = negative)
   439→   439→    //   At emitter/source: current EXITS (leaves terminal = positive)
   440→   440→    // Physics (PNP/PMOS): current flows E→C (or S→D) — reversed.
   441→   441→    //   At emitter/source: current ENTERS (leaves terminal = negative)
   442→   442→    //   At collector/drain: current EXITS (leaves terminal = positive)
   443→   443→    const isNpn = comp.type === 'npn' || comp.type === 'nmos';
   444→   444→    const cOrD = isNpn ? (comp.type === 'npn' ? 'c' : 'd') : (comp.type === 'pnp' ? 'c' : 'd');
   445→   445→    const eOrS = isNpn ? (comp.type === 'npn' ? 'e' : 's') : (comp.type === 'pnp' ? 'e' : 's');
   446→   446→    if (terminalId === eOrS) {
   447→   447→      // Emitter/Source: for NPN/NMOS current EXITS here; for PNP/PMOS current ENTERS here
   448→   448→      return isNpn ? compCurrent : -compCurrent;
   449→   449→    } else if (terminalId === cOrD) {
   450→   450→      // Collector/Drain: for NPN/NMOS current ENTERS here; for PNP/PMOS current EXITS here
   451→   451→      return isNpn ? -compCurrent : compCurrent;
   452→   452→    } else {
   453→   453→      // Base/Gate: small current, return 0 for flow visualization
   454→   454→      return 0;
   455→   455→    }
   456→   456→  } else if (comp.type === 'sevenSegment') {
   457→   457→    // 7-segment: current flows from segment terminal → com terminal.
   458→   458→    // compCurrent > 0 = current flows from each segment to com (through the LED).
   459→   459→    // For each segment terminal: current ENTERS the display → "leaving terminal" = -I_per_seg
   460→   460→    // For the com terminal: current EXITS the display → "leaving terminal" = +totalI
   461→   461→    const comTerm = terms.find((t) => t.terminalId === 'com')?.nodeId ?? 0;
   462→   462→    const termNode = terms.find((t) => t.terminalId === terminalId)?.nodeId ?? 0;
   463→   463→    if (terminalId === 'com') {
   464→   464→      // Current exits through com (sum of all segment currents)
   465→   465→      return compCurrent;
   466→   466→    } else {
   467→   467→      // For a segment terminal: compute this segment's current from nodeCurrentOut
   468→   468→      // (current leaving the segment node through the display's LED).
   469→   469→      // nodeCurrentOut at the segment node includes the display's draw.
   470→   470→      // We want -I_seg (current enters the display at this terminal).
   471→   471→      const segI = Math.abs(nodeCurrentOut.get(termNode) ?? 0);
   472→   472→      void comTerm;
   473→   473→      return -segI;
   474→   474→    }
   475→   475→  } else if (comp.type === 'cd4026') {
   476→   476→    // CD4026: segment outputs (a-g) are voltage sources driving 5V (ON) or 0V (OFF).
   477→   477→    // Current flows FROM the CD4026 segment pin INTO the wire → "leaving terminal" = +I_seg.
   478→   478→    // The current is computed from nodeCurrentOut at the segment node (the 7-seg
   479→   479→    // draws current through its internal resistance).
   480→   480→    // For the CLK, RST, VCC, GND, CO pins: return 0 (not visualized).
   481→   481→    const segIds = ['a', 'b', 'c', 'd', 'e', 'f', 'g'];
   482→   482→    if (segIds.includes(terminalId)) {
   483→   483→      const termNode = terms.find((t) => t.terminalId === terminalId)?.nodeId ?? 0;
   484→   484→      // nodeCurrentOut at the segment node = current leaving through the 7-seg.
   485→   485→      // This equals the current the CD4026 is sourcing from this pin.
   486→   486→      const segI = Math.abs(nodeCurrentOut.get(termNode) ?? 0);
   487→   487→      return segI;
   488→   488→    }
   489→   489→    // VCC pin: current ENTERS the CD4026 (from the power supply).
   490→   490→    // "leaving terminal" = -totalSegI (negative = current enters component).
   491→   491→    // Compute THIS CD4026's segment currents only (not the shared VCC node total).
   492→   492→    if (terminalId === 'vcc') {
   493→   493→      let totalSegI = 0;
   494→   494→      for (const seg of segIds) {
   495→   495→        const segNode = terms.find((t) => t.terminalId === seg)?.nodeId ?? 0;
   496→   496→        totalSegI += Math.abs(nodeCurrentOut.get(segNode) ?? 0);
   497→   497→      }
   498→   498→      return -totalSegI;
   499→   499→    }
   500→   500→    // CO output: current flows out when HIGH (driving the next CD4026's CLK)
   501→   501→    if (terminalId === 'co') {
   502→   502→      const coNode = terms.find((t) => t.terminalId === 'co')?.nodeId ?? 0;
   503→   503→      return Math.abs(nodeCurrentOut.get(coNode) ?? 0);
   504→   504→    }
   505→   505→    return 0;
   506→   506→  }
   507→   507→  return 0;
   508→   508→}
   509→   509→
   510→   510→/**
   511→   511→ * Compute the current flowing THROUGH each component (from its first terminal to
   512→   512→ * its second terminal), in amperes. Returns a map of componentId -> current.
   513→   513→ *
   514→   514→ * For 2-terminal components (R, C, L, diode, LED, switch): current = V(a)-V(b) / R etc.
   515→   515→ * For voltage sources: current = current through external circuit (from nodeCurrentOut).
   516→   516→ * For 3-terminal components (BJT, MOSFET): current = collector/drain current.
   517→   517→ *
   518→   518→ * Positive current = conventional current flowing from the first terminal to the
   519→   519→ * second terminal (for 2-terminal) or from collector/drain to emitter/source (for 3-terminal).
   520→   520→ */
   521→   521→export function computeComponentCurrents(
   522→   522→  components: CircuitComponent[],
   523→   523→  wires: Wire[],
   524→   524→  plugins: Map<string, ComponentPlugin>,
   525→   525→  sim: SimContext,
   526→   526→): Map<string, number> {
   527→   527→  const result = new Map<string, number>();
   528→   528→  const nodeMap = buildNodeMap(components, wires, plugins);
   529→   529→
   530→   530→  // ── Compute per-node current leaving through passive components ──────
   531→   531→  // This is used to find the current supplied by voltage sources.
   532→   532→  const nodeCurrentOut = new Map<number, number>();
   533→   533→  for (const comp of components) {
   534→   534→    const plugin = plugins.get(comp.type);
   535→   535→    if (!plugin) continue;
   536→   536→    const terms = getTerminalsForComponent(comp, plugin, nodeMap);
   537→   537→
   538→   538→    if (comp.type === 'resistor') {
   539→   539→      const r = Math.max(1e-9, comp.parameters.resistance as number);
   540→   540→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   541→   541→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   542→   542→      const i = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   543→   543→      nodeCurrentOut.set(a, (nodeCurrentOut.get(a) ?? 0) + i);
   544→   544→      nodeCurrentOut.set(b, (nodeCurrentOut.get(b) ?? 0) - i);
   545→   545→    } else if (comp.type === 'capacitor') {
   546→   546→      const C = Math.max(1e-15, comp.parameters.capacitance as number);
   547→   547→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   548→   548→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   549→   549→      const st = sim.state.__global ?? {};
   550→   550→      const vPrev = st[`cap_${a}_${b}`] ?? 0;
   551→   551→      const i = (C / Math.max(sim.dt, 1e-12)) * ((sim.nodeVoltage[a] - sim.nodeVoltage[b]) - vPrev);
   552→   552→      nodeCurrentOut.set(a, (nodeCurrentOut.get(a) ?? 0) + i);
   553→   553→      nodeCurrentOut.set(b, (nodeCurrentOut.get(b) ?? 0) - i);
   554→   554→    } else if (comp.type === 'inductor') {
   555→   555→      const L = Math.max(1e-12, comp.parameters.inductance as number);
   556→   556→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   557→   557→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   558→   558→      const st = sim.state.__global ?? {};
   559→   559→      const iPrev = st[`ind_${a}_${b}`] ?? 0;
   560→   560→      const v = sim.nodeVoltage[a] - sim.nodeVoltage[b];
   561→   561→      const dt = Math.max(sim.dt, 1e-12);
   562→   562→      const i = iPrev + (v / L) * dt;
   563→   563→      nodeCurrentOut.set(a, (nodeCurrentOut.get(a) ?? 0) + i);
   564→   564→      nodeCurrentOut.set(b, (nodeCurrentOut.get(b) ?? 0) - i);
   565→   565→    } else if (comp.type === 'led' || comp.type === 'diode') {
   566→   566→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   567→   567→      const k = terms.find((t) => t.terminalId === 'k')?.nodeId ?? 0;
   568→   568→      const v = sim.nodeVoltage[a] - sim.nodeVoltage[k];
   569→   569→      const vf = (comp.parameters.forwardV as number) || 0.7;
   570→   570→      const r = comp.type === 'led'
   571→   571→        ? Math.max(0.01, comp.parameters.seriesR as number)
   572→   572→        : Math.max(0.001, comp.parameters.onR as number);
   573→   573→      const st = sim.state.__global ?? {};
   574→   574→      const on = st[`${comp.type}_${a}_${k}`] ?? false;
   575→   575→      const i = on ? Math.max(0, (v - vf) / r) : 0;
   576→   576→      nodeCurrentOut.set(a, (nodeCurrentOut.get(a) ?? 0) + i);
   577→   577→      nodeCurrentOut.set(k, (nodeCurrentOut.get(k) ?? 0) - i);
   578→   578→    } else if (comp.type === 'switch' || comp.type === 'pushButton') {
   579→   579→      const closed = comp.type === 'switch' ? comp.parameters.closed : comp.parameters.pressed;
   580→   580→      if (closed) {
   581→   581→        const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   582→   582→        const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   583→   583→        const i = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / 0.01;
   584→   584→        nodeCurrentOut.set(a, (nodeCurrentOut.get(a) ?? 0) + i);
   585→   585→        nodeCurrentOut.set(b, (nodeCurrentOut.get(b) ?? 0) - i);
   586→   586→      }
   587→   587→    } else if (comp.type === 'currentSource') {
   588→   588→      const p = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   589→   589→      const n = terms.find((t) => t.terminalId === 'n')?.nodeId ?? 0;
   590→   590→      const i = comp.parameters.current as number;
   591→   591→      nodeCurrentOut.set(p, (nodeCurrentOut.get(p) ?? 0) + i);
   592→   592→      nodeCurrentOut.set(n, (nodeCurrentOut.get(n) ?? 0) - i);
   593→   593→    } else if (comp.type === 'speaker' || comp.type === 'lamp' || comp.type === 'dcMotor') {
   594→   594→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   595→   595→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   596→   596→      const r = Math.max(0.1, (comp.parameters.impedance as number) ?? (comp.parameters.resistance as number) ?? 8);
   597→   597→      const i = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   598→   598→      nodeCurrentOut.set(a, (nodeCurrentOut.get(a) ?? 0) + i);
   599→   599→      nodeCurrentOut.set(b, (nodeCurrentOut.get(b) ?? 0) - i);
   600→   600→    } else if (comp.type === 'photoresistor') {
   601→   601→      const darkR = comp.parameters.darkR as number;
   602→   602→      const lightR = comp.parameters.lightR as number;
   603→   603→      const light = Math.max(0, Math.min(1, comp.parameters.light as number));
   604→   604→      const r = Math.max(1e-6, darkR + (lightR - darkR) * light);
   605→   605→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   606→   606→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   607→   607→      const i = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   608→   608→      nodeCurrentOut.set(a, (nodeCurrentOut.get(a) ?? 0) + i);
   609→   609→      nodeCurrentOut.set(b, (nodeCurrentOut.get(b) ?? 0) - i);
   610→   610→    } else if (comp.type === 'voltmeter' || comp.type === 'ammeter' || comp.type === 'oscilloscope') {
   611→   611→      // Meters: very high impedance (10MΩ)
   612→   612→      const p = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   613→   613→      const n = terms.find((t) => t.terminalId === 'n')?.nodeId ?? 0;
   614→   614→      const i = (sim.nodeVoltage[p] - sim.nodeVoltage[n]) / 1e7;
   615→   615→      nodeCurrentOut.set(p, (nodeCurrentOut.get(p) ?? 0) + i);
   616→   616→      nodeCurrentOut.set(n, (nodeCurrentOut.get(n) ?? 0) - i);
   617→   617→    } else if (comp.type === 'sevenSegment') {
   618→   618→      // 7-segment: each segment draws current from its node to com.
   619→   619→      const com = terms.find((t) => t.terminalId === 'com')?.nodeId ?? 0;
   620→   620→      const threshold = (comp.parameters.threshold as number) ?? 2.0;
   621→   621→      const rSeg = 220;
   622→   622→      const st = sim.state.__global ?? {};
   623→   623→      const stateKey = `7seg_${terms.map(t => t.nodeId).join('_')}`;
   624→   624→      const segStates = (st[stateKey] ?? {}) as Record<string, boolean>;
   625→   625→      for (const seg of ['a', 'b', 'c', 'd', 'e', 'f', 'g']) {
   626→   626→        const segNode = terms.find((t) => t.terminalId === seg)?.nodeId ?? 0;
   627→   627→        const v = sim.nodeVoltage[segNode] - sim.nodeVoltage[com];
   628→   628→        const prevOn = segStates[seg] ?? false;
   629→   629→        const on = prevOn ? v > threshold * 0.5 : v > threshold;
   630→   630→        if (on) {
   631→   631→          const iSeg = v / rSeg;
   632→   632→          nodeCurrentOut.set(segNode, (nodeCurrentOut.get(segNode) ?? 0) + iSeg);
   633→   633→          nodeCurrentOut.set(com, (nodeCurrentOut.get(com) ?? 0) - iSeg);
   634→   634→        }
   635→   635→      }
   636→   636→    }
   637→   637→    // Voltage sources, 555, opamp, transistors, logic gates, oscillators:
   638→   638→    // Their current is determined by the external circuit, not by their own impedance.
   639→   639→    // We'll compute it from nodeCurrentOut below.
   640→   640→  }
   641→   641→
   642→   642→  // Second pass: add CD4026 VCC current (sum of all ON segment currents).
   643→   643→  // Must run after 7-segment displays have contributed to nodeCurrentOut.
   644→   644→  for (const comp of components) {
   645→   645→    if (comp.type !== 'cd4026') continue;
   646→   646→    const plugin = plugins.get(comp.type);
   647→   647→    if (!plugin) continue;
   648→   648→    const terms = getTerminalsForComponent(comp, plugin, nodeMap);
   649→   649→    const vccNode = terms.find((t) => t.terminalId === 'vcc')?.nodeId ?? 0;
   650→   650→    let totalSegI = 0;
   651→   651→    for (const seg of ['a', 'b', 'c', 'd', 'e', 'f', 'g']) {
   652→   652→      const segNode = terms.find((t) => t.terminalId === seg)?.nodeId ?? 0;
   653→   653→      totalSegI += Math.abs(nodeCurrentOut.get(segNode) ?? 0);
   654→   654→    }
   655→   655→    if (vccNode > 0) {
   656→   656→      nodeCurrentOut.set(vccNode, (nodeCurrentOut.get(vccNode) ?? 0) + totalSegI);
   657→   657→    }
   658→   658→  }
   659→   659→
   660→   660→  // Compute per-component current
   661→   661→  for (const comp of components) {
   662→   662→    const plugin = plugins.get(comp.type);
   663→   663→    if (!plugin) continue;
   664→   664→    const terms = getTerminalsForComponent(comp, plugin, nodeMap);
   665→   665→    let current = 0;
   666→   666→
   667→   667→    if (comp.type === 'resistor') {
   668→   668→      const r = Math.max(1e-9, comp.parameters.resistance as number);
   669→   669→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   670→   670→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   671→   671→      current = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   672→   672→    } else if (comp.type === 'capacitor') {
   673→   673→      const C = Math.max(1e-15, comp.parameters.capacitance as number);
   674→   674→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   675→   675→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   676→   676→      const st = sim.state.__global ?? {};
   677→   677→      const vPrev = st[`cap_${a}_${b}`] ?? 0;
   678→   678→      current = (C / Math.max(sim.dt, 1e-12)) * ((sim.nodeVoltage[a] - sim.nodeVoltage[b]) - vPrev);
   679→   679→    } else if (comp.type === 'inductor') {
   680→   680→      const L = Math.max(1e-12, comp.parameters.inductance as number);
   681→   681→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   682→   682→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   683→   683→      const st = sim.state.__global ?? {};
   684→   684→      const iPrev = st[`ind_${a}_${b}`] ?? 0;
   685→   685→      const v = sim.nodeVoltage[a] - sim.nodeVoltage[b];
   686→   686→      const dt = Math.max(sim.dt, 1e-12);
   687→   687→      current = iPrev + (v / L) * dt;
   688→   688→    } else if (comp.type === 'led' || comp.type === 'diode') {
   689→   689→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   690→   690→      const k = terms.find((t) => t.terminalId === 'k')?.nodeId ?? 0;
   691→   691→      const v = sim.nodeVoltage[a] - sim.nodeVoltage[k];
   692→   692→      const vf = (comp.parameters.forwardV as number) || 0.7;
   693→   693→      const r = comp.type === 'led'
   694→   694→        ? Math.max(0.01, comp.parameters.seriesR as number)
   695→   695→        : Math.max(0.001, comp.parameters.onR as number);
   696→   696→      const st = sim.state.__global ?? {};
   697→   697→      const on = st[`${comp.type}_${a}_${k}`] ?? false;
   698→   698→      current = on ? Math.max(0, (v - vf) / r) : 0;
   699→   699→    } else if (comp.type === 'dcVoltage' || comp.type === 'acVoltage' || comp.type === 'pulseSource') {
   700→   700→      const p = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   701→   701→      current = nodeCurrentOut.get(p) ?? 0;
   702→   702→    } else if (comp.type === 'currentSource') {
   703→   703→      current = comp.parameters.current as number;
   704→   704→    } else if (comp.type === 'switch' || comp.type === 'pushButton') {
   705→   705→      const closed = comp.type === 'switch' ? comp.parameters.closed : comp.parameters.pressed;
   706→   706→      if (closed) {
   707→   707→        const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   708→   708→        const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   709→   709→        current = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / 0.01;
   710→   710→      }
   711→   711→    } else if (comp.type === 'npn' || comp.type === 'pnp') {
   712→   712→      const isNpn = comp.type === 'npn';
   713→   713→      const cNode = terms.find((t) => t.terminalId === 'c')?.nodeId ?? 0;
   714→   714→      const eNode = terms.find((t) => t.terminalId === 'e')?.nodeId ?? 0;
   715→   715→      if (isNpn) {
   716→   716→        // NPN: current flows C→E. Collector is connected through load to VCC.
   717→   717→        current = -(nodeCurrentOut.get(cNode) ?? 0);
   718→   718→      } else {
   719→   719→        // PNP: current flows E→C. Collector is connected through load to GND.
   720→   720→        // Use collector node (connected to load) not emitter (connected to VCC directly).
   721→   721→        current = nodeCurrentOut.get(cNode) ?? 0;
   722→   722→      }
   723→   723→    } else if (comp.type === 'nmos' || comp.type === 'pmos') {
   724→   724→      const isNmos = comp.type === 'nmos';
   725→   725→      const dNode = terms.find((t) => t.terminalId === 'd')?.nodeId ?? 0;
   726→   726→      const sNode = terms.find((t) => t.terminalId === 's')?.nodeId ?? 0;
   727→   727→      if (isNmos) {
   728→   728→        // NMOS: current flows D→S. Current leaving drain node through load = device current.
   729→   729→        current = -(nodeCurrentOut.get(dNode) ?? 0);
   730→   730→      } else {
   731→   731→        // PMOS: current flows S→D. Current leaving drain node through load = device current.
   732→   732→        // Use drain node (connected to load) not source (connected to VCC directly).
   733→   733→        current = nodeCurrentOut.get(dNode) ?? 0;
   734→   734→      }
   735→   735→    } else if (comp.type === 'timer555') {
   736→   736→      const outNode = terms.find((t) => t.terminalId === 'out')?.nodeId ?? 0;
   737→   737→      current = nodeCurrentOut.get(outNode) ?? 0;
   738→   738→    } else if (comp.type === 'opamp' || comp.type === 'opampRails') {
   739→   739→      const outNode = terms.find((t) => t.terminalId === 'out')?.nodeId ?? 0;
   740→   740→      current = nodeCurrentOut.get(outNode) ?? 0;
   741→   741→    } else if (comp.type === 'arduino' || comp.type === 'arduinoReal' || comp.type === 'raspberryPi') {
   742→   742→      let totalCurrent = 0;
   743→   743→      for (const t of terms) {
   744→   744→        totalCurrent += Math.abs(nodeCurrentOut.get(t.nodeId) ?? 0);
   745→   745→      }
   746→   746→      current = totalCurrent;
   747→   747→    } else if (comp.type === 'voltmeter' || comp.type === 'ammeter' || comp.type === 'oscilloscope') {
   748→   748→      const p = terms.find((t) => t.terminalId === 'p')?.nodeId ?? 0;
   749→   749→      const n = terms.find((t) => t.terminalId === 'n')?.nodeId ?? 0;
   750→   750→      current = (sim.nodeVoltage[p] - sim.nodeVoltage[n]) / 1e7;
   751→   751→    } else if (comp.type === 'vco' || comp.type === 'crystal') {
   752→   752→      const outNode = terms.find((t) => t.terminalId === 'out')?.nodeId ?? 0;
   753→   753→      current = nodeCurrentOut.get(outNode) ?? 0;
   754→   754→    } else if (comp.type === 'speaker' || comp.type === 'lamp' || comp.type === 'dcMotor') {
   755→   755→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   756→   756→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   757→   757→      const r = Math.max(0.1, (comp.parameters.impedance as number) ?? (comp.parameters.resistance as number) ?? 8);
   758→   758→      current = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   759→   759→    } else if (comp.type === 'photoresistor') {
   760→   760→      const darkR = comp.parameters.darkR as number;
   761→   761→      const lightR = comp.parameters.lightR as number;
   762→   762→      const light = Math.max(0, Math.min(1, comp.parameters.light as number));
   763→   763→      const r = Math.max(1e-6, darkR + (lightR - darkR) * light);
   764→   764→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   765→   765→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   766→   766→      current = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   767→   767→    } else if (comp.type === 'sevenSegment') {
   768→   768→      // 7-segment: current = sum of segment currents (each segment ~10mA when on)
   769→   769→      const st = sim.state.__global ?? {};
   770→   770→      let totalI = 0;
   771→   771→      for (const seg of ['a', 'b', 'c', 'd', 'e', 'f', 'g']) {
   772→   772→        const segNode = terms.find((t) => t.terminalId === seg)?.nodeId ?? 0;
   773→   773→        totalI += Math.abs(nodeCurrentOut.get(segNode) ?? 0);
   774→   774→      }
   775→   775→      current = totalI;
   776→   776→    } else if (comp.type === 'cd4026') {
   777→   777→      // CD4026: total current = sum of all segment output currents.
   778→   778→      // Each ON segment sources current into the external 7-seg display.
   779→   779→      let totalI = 0;
   780→   780→      for (const seg of ['a', 'b', 'c', 'd', 'e', 'f', 'g']) {
   781→   781→        const segNode = terms.find((t) => t.terminalId === seg)?.nodeId ?? 0;
   782→   782→        totalI += Math.abs(nodeCurrentOut.get(segNode) ?? 0);
   783→   783→      }
   784→   784→      current = totalI;
   785→   785→    } else if (comp.type === 'transformer') {
   786→   786→      const p1 = terms.find((t) => t.terminalId === 'p1')?.nodeId ?? 0;
   787→   787→      current = nodeCurrentOut.get(p1) ?? 0;
   788→   788→    } else if (comp.type === 'potentiometer') {
   789→   789→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   790→   790→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   791→   791→      const r = Math.max(1e-6, comp.parameters.resistance as number);
   792→   792→      current = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   793→   793→    } else if (comp.type === 'fuse') {
   794→   794→      const a = terms.find((t) => t.terminalId === 'a')?.nodeId ?? 0;
   795→   795→      const b = terms.find((t) => t.terminalId === 'b')?.nodeId ?? 0;
   796→   796→      const r = Math.max(1e-6, comp.parameters.resistance as number);
   797→   797→      current = (sim.nodeVoltage[a] - sim.nodeVoltage[b]) / r;
   798→   798→    }
   799→   799→    // For all other types (logic gates, power symbols, etc.), current = 0
   800→   800→
   801→   801→    result.set(comp.id, current);
   802→   802→  }
   803→   803→
   804→   804→  return result;
   805→   805→}

   807→   807→/**
   808→   808→ * Run one simulation step.
   809→   809→ * - Stamps all components (with Newton iteration for non-linear ones; for simplicity we do fixed iterations).
   810→   810→ * - Solves the linear system.
   811→   811→ * - Calls `step` on each plugin for stateful logic (555, MCU, etc.).
   812→   812→ */
   813→   813→export interface StepResult {
   814→   814→  sim: SimContext;
   815→   815→  branchCurrentSize: number;
   816→   816→  nodeMap: NodeMap;
   817→   817→}
   818→   818→
   819→   819→export interface PrevState {
   820→   820→  nodeVoltage: Float64Array;
   821→   821→  branchCurrent: Float64Array;
   822→   822→  time: number;
   823→   823→  /** persistent state map, survives across steps. Plugins store their internal state here. */
   824→   824→  state: Record<string, any>;
   825→   825→}
   826→   826→
   827→   827→export function simulateStep(
   828→   828→  components: CircuitComponent[],
   829→   829→  wires: Wire[],
   830→   830→  plugins: Map<string, ComponentPlugin>,
   831→   831→  prev?: PrevState,
   832→   832→  dt: number = 1e-4,
   833→   833→  simOptions?: { initialConditions?: Record<string, number>; nodeSets?: Record<string, number> },
   834→   834→): StepResult | null {
   835→   835→  const nodeMap = buildNodeMap(components, wires, plugins);
   836→   836→  const numNodes = nodeMap.numNodes; // includes ground (0)
   837→   837→  const maxExtras = components.length * 4 + 8;
   838→   838→
   839→   839→  // Use the sparse solver for circuits > 80 nodes — much faster for big designs.
   840→   840→  // Below that threshold, the dense solver wins (less overhead per stamp).
   841→   841→  const useSparse = shouldUseSparseSolver(numNodes - 1, maxExtras);
   842→   842→  const sparseSys = useSparse ? createSparseMnaSystem(numNodes - 1, maxExtras) : null;
   843→   843→  const sys = useSparse ? asMnaSystem(sparseSys!) : createMnaSystem(numNodes - 1, maxExtras);
   844→   844→
   845→   845→  const time = prev && prev.nodeVoltage.length > 0 ? prev.time + dt : 0;
   846→   846→
   847→   847→  // initialize SimContext with previous voltages (for companion models).
   848→   848→  // If prev has no data yet (first step), allocate fresh arrays sized to numNodes.
   849→   849→  const hasPrev = prev && prev.nodeVoltage.length > 0;
   850→   850→  const nodeVoltage = hasPrev ? Float64Array.from(prev.nodeVoltage) : new Float64Array(numNodes);
   851→   851→
   852→   852→  // Apply .IC (initial conditions) on the very first step (no prev) — these override
   853→   853→  // the default 0V initialization and are used when `uic` is true.
   854→   854→  // Apply .NODESET as a hint for the DC solver (initial guess).
   855→   855→  if (!hasPrev && simOptions) {
   856→   856→    // .IC — overrides node voltages at t=0
   857→   857→    if (simOptions.initialConditions) {
   858→   858→      for (const [termKey, voltage] of Object.entries(simOptions.initialConditions)) {
   859→   859→        const nodeId = nodeMap.terminalNode.get(termKey);
   860→   860→        if (nodeId != null && nodeId > 0 && nodeId - 1 < nodeVoltage.length) {
   861→   861→          nodeVoltage[nodeId - 1] = voltage;
   862→   862→        }
   863→   863→      }
   864→   864→    }
   865→   865→    // .NODESET — same as .IC but only used as an initial guess for the DC solver
   866→   866→    // (it gets overwritten by the solve, but helps convergence).
   867→   867→    // For the transient engine, .NODESET and .IC behave the same way at t=0.
   868→   868→    if (simOptions.nodeSets) {
   869→   869→      for (const [termKey, voltage] of Object.entries(simOptions.nodeSets)) {
   870→   870→        const nodeId = nodeMap.terminalNode.get(termKey);
   871→   871→        if (nodeId != null && nodeId > 0 && nodeId - 1 < nodeVoltage.length) {
   872→   872→          // Only apply if not already set by .IC
   873→   873→          if (simOptions.initialConditions?.[termKey] === undefined) {
   874→   874→            nodeVoltage[nodeId - 1] = voltage;
   875→   875→          }
   876→   876→        }
   877→   877→      }
   878→   878→    }
   879→   879→  }
   880→   880→
   881→   881→  const branchCurrent = hasPrev ? Float64Array.from(prev.branchCurrent) : new Float64Array(maxExtras);
   882→   882→  // PERSISTENT state: reuse the same state object across steps so plugins (capacitors,
   883→   883→  // inductors, 555, MCU) keep their memory. Created once per session.
   884→   884→  const stateMap: Record<string, any> = prev ? prev.state : {};
   885→   885→  // expose per-component state objects (also persistent via comp.simState)
   886→   886→  for (const comp of components) {
   887→   887→    if (!comp.simState) comp.simState = {};
   888→   888→    if (!stateMap[comp.id]) stateMap[comp.id] = comp.simState;
   889→   889→  }
   890→   890→
   891→   891→  const sim: SimContext = {
   892→   892→    nodeVoltage,
   893→   893→    branchCurrent,
   894→   894→    state: stateMap,
   895→   895→    time,
   896→   896→    dt,
   897→   897→  };
   898→   898→
   899→   899→  // First pass: let plugins initialize their simState (e.g., capacitor voltage)
   900→   900→  for (const comp of components) {
   901→   901→    const plugin = plugins.get(comp.type);
   902→   902→    if (!plugin) continue;
   903→   903→    if (!comp.simState) comp.simState = {};
   904→   904→    if (plugin.step && !comp.simState.__inited) {
   905→   905→      comp.simState.__inited = true;
   906→   906→    }
   907→   907→  }
   908→   908→
   909→   909→  // Stamp all components
   910→   910→  sys.nextExtra = numNodes - 1; // reset extra counter; extra vars start at index (numNodes-1)
   911→   911→  for (const comp of components) {
   912→   912→    const plugin = plugins.get(comp.type);
   913→   913→    if (!plugin || !plugin.stamp) continue;
   914→   914→    const terminals = getTerminalsForComponent(comp, plugin, nodeMap);
   915→   915→    try {
   916→   916→      plugin.stamp(comp.parameters, terminals, sys, sim);
   917→   917→    } catch (e) {
   918→   918→      console.error(`stamp error in ${comp.type} (${comp.id}):`, e);
   919→   919→    }
   920→   920→  }
   921→   921→
   922→   922→  // resize system: we may have allocated more extra vars than used.
   923→   923→  // build a smaller system to avoid singular cols
   924→   924→  const actualSize = sys.nextExtra;
   925→   925→  if (actualSize < sys.size) {
   926→   926→    // shrink
   927→   927→    const newA = new Float64Array(actualSize * actualSize);
   928→   928→    const newZ = new Float64Array(actualSize);
   929→   929→    for (let r = 0; r < actualSize; r++) {
   930→   930→      for (let c = 0; c < actualSize; c++) {
   931→   931→        newA[r * actualSize + c] = sys.A[r * sys.size + c];
   932→   932→      }
   933→   933→      newZ[r] = sys.z[r];
   934→   934→    }
   935→   935→    sys.A = newA;
   936→   936→    sys.z = newZ;
   937→   937→    sys.size = actualSize;
   938→   938→    sys.numExtra = actualSize - (numNodes - 1);
   939→   939→  }
   940→   940→
   941→   941→  const x = useSparse && sparseSys ? solveSparse(sparseSys) : solveMna(sys);
   942→   942→  if (!x) {
   943→   943→    return null;
   944→   944→  }
   945→   945→
   946→   946→  // copy results back into sim.nodeVoltage / branchCurrent
   947→   947→  for (let i = 0; i < numNodes; i++) {
   948→   948→    sim.nodeVoltage[i] = i === 0 ? 0 : x[i - 1];
   949→   949→  }
   950→   950→  for (let i = 0; i < sys.numExtra; i++) {
   951→   951→    sim.branchCurrent[i] = x[numNodes - 1 + i];
   952→   952→  }
   953→   953→
   954→   954→  // call step() for stateful components
   955→   955→  for (const comp of components) {
   956→   956→    const plugin = plugins.get(comp.type);
   957→   957→    if (!plugin || !plugin.step) continue;
   958→   958→    const terminals = getTerminalsForComponent(comp, plugin, nodeMap);
   959→   959→    try {
   960→   960→      plugin.step(comp.parameters, terminals, sim, comp);
   961→   961→    } catch (e) {
   962→   962→      console.error(`step error in ${comp.type} (${comp.id}):`, e);
   963→   963→    }
   964→   964→  }
   965→   965→
   966→   966→  return { sim, branchCurrentSize: sys.numExtra, nodeMap };
   967→   967→}
   968→   968→
   969→   969→/**
   970→   970→ * Solve a pure DC operating point (no time stepping).
   971→   971→ * Iterates a few times to handle non-linear components (diodes, transistors).
   972→   972→ */
   973→   973→export function solveDC(
   974→   974→  components: CircuitComponent[],
   975→   975→  wires: Wire[],
   976→   976→  plugins: Map<string, ComponentPlugin>,
   977→   977→  maxIter: number = 50,
   978→   978→): SimContext | null {
   979→   979→  // initialize simState
   980→   980→  for (const comp of components) {
   981→   981→    if (!comp.simState) comp.simState = {};
   982→   982→  }
   983→   983→
   984→   984→  let prev: PrevState | undefined;
   985→   985→  let result: SimContext | null = null;
   986→   986→  for (let iter = 0; iter < maxIter; iter++) {
   987→   987→    const r = simulateStep(components, wires, plugins, prev, 1e-6);
   988→   988→    if (!r) return null;
   989→   989→    result = r.sim;
   990→   990→    // check convergence
   991→   991→    if (prev) {
   992→   992→      let maxDelta = 0;
   993→   993→      for (let i = 0; i < r.sim.nodeVoltage.length; i++) {
   994→   994→        const d = Math.abs(r.sim.nodeVoltage[i] - prev.nodeVoltage[i]);
   995→   995→        if (d > maxDelta) maxDelta = d;
   996→   996→      }
   997→   997→      prev = {
   998→   998→        nodeVoltage: r.sim.nodeVoltage,
   999→   999→        branchCurrent: r.sim.branchCurrent,
  1000→  1000→        time: r.sim.time,
  1001→  1001→        state: r.sim.state,
  1002→  1002→      };
  1003→  1003→      if (maxDelta < 1e-6 && iter > 0) break;
  1004→  1004→    } else {
  1005→  1005→      prev = {
  1006→  1006→        nodeVoltage: r.sim.nodeVoltage,
  1007→  1007→        branchCurrent: r.sim.branchCurrent,
  1008→  1008→        time: r.sim.time,
  1009→  1009→        state: r.sim.state,
  1010→  1010→      };
  1011→  1011→    }
  1012→  1012→  }
  1013→  1013→  return result;
  1014→  1014→}
